+    /**
+     * Validate a requested log file path to prevent path traversal / arbitrary file access
+     * (CWE-022). Only genuine task log files are ever served here: they always end with the
+     * ".log" suffix and never contain a parent-directory ("..") segment. Any other path
+     * (e.g. "/etc/passwd" or "../../etc/passwd.log") is rejected.
+     *
+     * @param path the user-controlled log file path
+     */
+    private void checkLogPath(String path) {
+        if (path == null || path.trim().isEmpty()
+                || path.contains("..")
+                || !path.endsWith(".log")) {
+            logger.warn("Illegal log file path rejected: {}", path);
+            throw new IllegalArgumentException("Illegal log file path: " + path);
+        }
+    }
