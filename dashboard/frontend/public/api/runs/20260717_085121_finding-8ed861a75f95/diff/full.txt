+package io.undertow.server.handlers.resource;
+
+import java.io.File;
+import java.io.FileOutputStream;
+import java.io.IOException;
+
+import org.junit.After;
+import org.junit.Assert;
+import org.junit.Before;
+import org.junit.Test;
+
+/**
+ * Tests that {@link FileResourceManager} does not allow escaping the configured
+ * base directory via path traversal sequences, including sequences that rely on
+ * the platform specific ('\') separator on Windows.
+ *
+ * @author Stuart Douglas
+ */
+public class FileResourceManagerTestCase {
+
+    private File rootDir;
+    private File baseDir;
+    private File secretFile;
+
+    @Before
+    public void setup() throws IOException {
+        rootDir = new File(System.getProperty("java.io.tmpdir"), "frm-test-" + System.nanoTime());
+        baseDir = new File(rootDir, "base");
+        baseDir.mkdirs();
+
+        secretFile = new File(rootDir, "secret.txt");
+        writeFile(secretFile, "secret");
+
+        File allowed = new File(baseDir, "allowed.txt");
+        writeFile(allowed, "allowed");
+    }
+
+    private static void writeFile(File file, String contents) throws IOException {
+        FileOutputStream out = new FileOutputStream(file);
+        try {
+            out.write(contents.getBytes("UTF-8"));
+        } finally {
+            out.close();
+        }
+    }
+
+    @After
+    public void cleanup() {
+        deleteRecursive(rootDir);
+    }
+
+    private static void deleteRecursive(File file) {
+        if (file == null || !file.exists()) {
+            return;
+        }
+        File[] children = file.listFiles();
+        if (children != null) {
+            for (File child : children) {
+                deleteRecursive(child);
+            }
+        }
+        file.delete();
+    }
+
+    @Test
+    public void testResourceWithinBaseIsServed() throws Exception {
+        FileResourceManager manager = new FileResourceManager(baseDir, 1024);
+        Resource resource = manager.getResource("/allowed.txt");
+        Assert.assertNotNull(resource);
+    }
+
+    @Test
+    public void testDotDotTraversalIsBlocked() throws Exception {
+        FileResourceManager manager = new FileResourceManager(baseDir, 1024);
+        Resource resource = manager.getResource("/../secret.txt");
+        Assert.assertNull(resource);
+    }
+
+    @Test
+    public void testBackslashTraversalIsBlocked() throws Exception {
+        FileResourceManager manager = new FileResourceManager(baseDir, 1024);
+        //this mimics a request path that bypassed '/'-based canonicalization upstream
+        //(e.g. on Windows, where java.io.File also treats '\' as a separator)
+        Resource resource = manager.getResource("/..\\secret.txt");
+        Assert.assertNull(resource);
+    }
+}
