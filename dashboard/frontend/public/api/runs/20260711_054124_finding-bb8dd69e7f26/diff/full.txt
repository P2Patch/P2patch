+    /**
+     * Verifies that the resolved resource is located inside the configured base directory.
+     *
+     * <p>{@link UriPath#canonical(String)} only treats '/' as a path separator, so traversal
+     * sequences that rely on other separators (for example backslashes on some platforms) can
+     * survive canonicalization and later be resolved against the file system, escaping the base
+     * directory. This defense compares fully canonicalized file paths to guarantee containment
+     * regardless of how the request path was encoded.
+     *
+     * @param resource the resolved external resource
+     * @return {@code true} if the resource resides within the base directory, {@code false} otherwise
+     * @throws IOException if the canonical file paths cannot be resolved
+     */
+    private boolean isWithinBaseDirectory(ExternalResource resource) throws IOException {
+        File base = new File(baseResource).getCanonicalFile();
+        File target = new File(resource.getPath()).getCanonicalFile();
+
+        for (File current = target; current != null; current = current.getParentFile()) {
+            if (current.equals(base)) {
+                return true;
+            }
+        }
+        return false;
+    }
