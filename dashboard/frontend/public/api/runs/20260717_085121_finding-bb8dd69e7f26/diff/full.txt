+    /**
+     * Verifies that a resolved resource path stays within the given base resource path once both
+     * are canonicalized. This guards against path traversal ("..", mixed/platform-specific
+     * separators, etc.) that survives the URI-level normalization performed upstream (e.g.
+     * {@link UriPath#canonical(String)}), by re-checking at the actual filesystem boundary right
+     * before the resource is served.
+     * <p>Both paths are normalized to have a leading slash before comparison, since some
+     * resources (e.g. {@link ClassPathResource#getPath()}) report their path without one, which
+     * would otherwise cause a false mismatch against an absolute base resource.
+     *
+     * @param baseResource the configured resource root
+     * @param resourcePath the fully-combined candidate resource path to validate
+     * @return true if resourcePath resolves to baseResource itself or a location beneath it
+     * @throws IOException if the canonical path cannot be determined
+     */
+    protected static boolean isPathUnderBase(String baseResource, String resourcePath) throws IOException {
+        String base = new File(withLeadingSlash(baseResource)).getCanonicalPath();
+        String candidate = new File(withLeadingSlash(resourcePath)).getCanonicalPath();
+        return candidate.equals(base) || candidate.startsWith(base + File.separator);
+    }
+
+    private static String withLeadingSlash(String path) {
+        return path.startsWith(SLASH) ? path : SLASH + path;
+    }
