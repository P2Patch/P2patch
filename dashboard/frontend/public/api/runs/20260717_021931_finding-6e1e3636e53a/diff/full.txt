+    /**
+     * Load a class by name and ensure it is assignable to the expected base type
+     * <b>before</b> any instantiation is attempted. This is a hard boundary check
+     * against reflective code injection (CWE-094): class names coming from
+     * configuration files, REST/JSON payloads or HTTP parameters must never be
+     * used to construct arbitrary, attacker-chosen classes. Rejecting the class
+     * here means its constructor (and any side effect it may have) never runs
+     * when it does not match the expected contract.
+     *
+     * @param className
+     *      fully qualified class name, potentially attacker-controlled
+     * @param expectedType
+     *      base type/interface the resolved class must implement or extend
+     * @return
+     *      the resolved class, guaranteed assignable to {@code expectedType}
+     * @throws ClassNotFoundException
+     *      if the class cannot be found
+     */
+    public static <T> Class<? extends T> loadClass(String className, Class<T> expectedType) throws ClassNotFoundException {
+        Class<?> clazz = (classLoader == null) ? Class.forName(className) : classLoader.loadClass(className);
+        if (!expectedType.isAssignableFrom(clazz)) {
+            throw new IllegalArgumentException("Class '" + className + "' is not a valid " + expectedType.getName());
+        }
+        return clazz.asSubclass(expectedType);
+    }
