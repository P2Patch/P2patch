+    /**
+     * Determines whether the resolved extraction target is contained within the
+     * destination directory. This guards against Zip Slip / path traversal
+     * (CWE-22) where a crafted archive entry name (e.g. containing {@code ../})
+     * would otherwise write outside of the intended extraction directory.
+     *
+     * @param file the file that would be written for the archive entry
+     * @param destination the intended extraction directory
+     * @return {@code true} if the file resolves to a location inside the
+     * destination directory; otherwise {@code false}
+     */
+    private static boolean isInsideDestination(File file, File destination) {
+        try {
+            final String canonicalDest = destination.getCanonicalPath();
+            final String canonicalFile = file.getCanonicalPath();
+            return canonicalFile.equals(canonicalDest)
+                    || canonicalFile.startsWith(canonicalDest + File.separator);
+        } catch (IOException ex) {
+            LOGGER.debug("Unable to canonicalize extraction path", ex);
+            return false;
+        }
+    }

+    /**
+     * Regression test for CWE-022 (Zip Slip / path traversal). A crafted archive

+     * rejected and must not result in a file being written outside of the
+     * destination directory.

+    @Test
+    public void testZipSlipIsBlocked() throws Exception {
+        assumeFalse(isPreviouslyLoaded("org.owasp.dependencycheck.analyzer.ArchiveAnalyzer"));
+
+        final File sandbox = Files.createTempDirectory("dc-zipslip-test").toFile();
+        final File destination = new File(sandbox, "extract-here");
+        assertTrue(destination.mkdirs());
+
+        final File escapedTarget = new File(sandbox, "PWNED_outside_destination.txt");
+        final String maliciousEntryName = "../" + escapedTarget.getName();
+
+        final File maliciousZip = new File(sandbox, "malicious.zip");
+        try (ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(maliciousZip))) {
+            final ZipEntry entry = new ZipEntry(maliciousEntryName);
+            zos.putNextEntry(entry);
+            zos.write("owned-by-zip-slip".getBytes("UTF-8"));
+            zos.closeEntry();
+        }
+
+        final ArchiveAnalyzer analyzer = new ArchiveAnalyzer();
+        analyzer.initialize(getSettings());
+
+        final Engine engine = new Engine(Engine.Mode.EVIDENCE_COLLECTION, getSettings()) {
+            @Override
+            public boolean accept(File file) {
+                return true;
+            }
+        };
+
+        final Method extractFiles = ArchiveAnalyzer.class.getDeclaredMethod(
+                "extractFiles", File.class, File.class, Engine.class);
+        extractFiles.setAccessible(true);
+
+        try {
+            // extractFiles logs and swallows the ArchiveExtractionException; the
+            // security guarantee is that the traversing entry is never written.
+            extractFiles.invoke(analyzer, maliciousZip, destination, engine);
