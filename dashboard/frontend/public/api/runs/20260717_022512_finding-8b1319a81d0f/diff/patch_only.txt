+    /**
+     * Resolves {@code entryName} against {@code canonicalDirPath}, following any symlink already present on disk
+     * (typically created by an earlier entry of the very same archive) at each path segment, even when that
+     * symlink is dangling, i.e. its target does not exist (yet). Unlike {@link File#getCanonicalFile()}, this
+     * does not require the resolved location to exist, so a malicious archive cannot hide a path traversal
+     * behind a not-yet-materialized symlink target.
+     *
+     * @param dereferenceLastSegment whether an existing on-disk symlink at the final path segment should be
+     *                               followed too. This must be {@code false} when the current entry is itself
+     *                               (re)creating a symlink at that location, since it is about to replace
+     *                               whatever is currently there rather than read or write through it.
+     */
+    private static Path resolveAgainstExistingSymlinks( final Path canonicalDirPath, final String entryName,
+                                                         final boolean dereferenceLastSegment )
+        throws IOException
+    {
+        final String[] segments = entryName.replace( '\\', '/' ).split( "/" );
+        Path resolved = canonicalDirPath;
+        for ( int i = 0; i < segments.length; i++ )
+        {
+            final String segment = segments[i];
+            if ( segment.isEmpty() || ".".equals( segment ) )
+            {
+                continue;
+            }
+
+            resolved = resolved.resolve( segment );
+
+            final boolean isLastSegment = i == segments.length - 1;
+            if ( ( dereferenceLastSegment || !isLastSegment ) && Files.isSymbolicLink( resolved ) )
+            {
+                final Path linkTarget = Files.readSymbolicLink( resolved );
+                resolved = linkTarget.isAbsolute()
+                    ? linkTarget.normalize()
+                    : resolved.getParent().resolve( linkTarget ).normalize();
+            }
+            else
+            {
+                resolved = resolved.normalize();
+            }
+
+            // Bail out as soon as the path escapes, so that a later segment can't step back "inside" and
+            // hide the fact that it briefly left the destination directory via a symlink.
+            if ( !resolved.startsWith( canonicalDirPath ) )
+            {
+                return resolved;
+            }
+        }
+        return resolved;
+    }
