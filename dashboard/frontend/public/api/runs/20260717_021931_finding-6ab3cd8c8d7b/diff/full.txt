+    /**
+     * Check whether the given directory path, expressed as its "/" separated segments, can be
+     * traversal or not, e.g. a segment of ".." that would allow escaping the intended base
+     * directory. Unlike {@link #directoryTraversal(String)}, this accepts values that legitimately
+     * contain the "/" separator (multi-level directory paths).
+     *
+     * @param pathSegments the path split on "/"
+     * @return whether the directory path could be traversal or not
+     */
+    public static boolean directoryTraversal(String[] pathSegments) {
+        if (pathSegments == null) {
+            return false;
+        }
+        for (String segment : pathSegments) {
+            if (segment == null || segment.isEmpty()) {
+                continue;
+            }
+            if (directoryTraversal(segment)) {
+                return true;
+            }
+        }
+        return false;
+    }
