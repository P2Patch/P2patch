+package com.alibaba.oneagent.utils;
+
+import java.io.File;
+import java.io.FileOutputStream;
+import java.io.IOException;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+import org.junit.Assert;
+import org.junit.Test;
+
+/**
+ * Regression tests for zip-slip protection in {@link IOUtils#unzip(String, String)}.
+ */
+public class IOUtilsTest {
+
+    @Test
+    public void testUnzipRejectsPathTraversalEntry() throws IOException {
+        File tmpDir = new File(System.getProperty("java.io.tmpdir"),
+                "ioutils-zipslip-test-" + System.nanoTime());
+        Assert.assertTrue(tmpDir.mkdirs());
+
+        File destDir = new File(tmpDir, "dest");
+        Assert.assertTrue(destDir.mkdirs());
+
+        File maliciousZip = new File(tmpDir, "evil.zip");
+        FileOutputStream fos = new FileOutputStream(maliciousZip);
+        try {
+            ZipOutputStream zos = new ZipOutputStream(fos);
+            try {
+                // entry name attempts to escape the destination directory
+                ZipEntry entry = new ZipEntry("../../evil.txt");
+                zos.putNextEntry(entry);
+                zos.write("pwned".getBytes("UTF-8"));
+                zos.closeEntry();
+            } finally {
+                zos.close();
+            }
+        } finally {
+            fos.close();
+        }
+
+        try {
+            IOUtils.unzip(maliciousZip.getAbsolutePath(), destDir.getAbsolutePath());
+            Assert.fail("Expected IOException due to zip entry path traversal");
+        } catch (IOException expected) {
+            // expected: traversal must be rejected
+        }
+
+        File escapedFile = new File(tmpDir, "evil.txt");
+        Assert.assertFalse("Extraction must not create files outside the destination directory",
+                escapedFile.exists());
+    }
+
+    @Test
+    public void testUnzipExtractsNormalEntryWithinDestination() throws IOException {
+        File tmpDir = new File(System.getProperty("java.io.tmpdir"),
+                "ioutils-zip-normal-test-" + System.nanoTime());
+        Assert.assertTrue(tmpDir.mkdirs());
+
+        File destDir = new File(tmpDir, "dest");
+        Assert.assertTrue(destDir.mkdirs());
+
+        File zipFile = new File(tmpDir, "normal.zip");
+        FileOutputStream fos = new FileOutputStream(zipFile);
+        try {
+            ZipOutputStream zos = new ZipOutputStream(fos);
+            try {
+                ZipEntry entry = new ZipEntry("sub/hello.txt");
+                zos.putNextEntry(entry);
+                zos.write("hello".getBytes("UTF-8"));
+                zos.closeEntry();
+            } finally {
+                zos.close();
+            }
+        } finally {
+            fos.close();
+        }
+
+        IOUtils.unzip(zipFile.getAbsolutePath(), destDir.getAbsolutePath());
+
+        File extracted = new File(destDir, "sub/hello.txt");
+        Assert.assertTrue("Normal entries should still be extracted correctly", extracted.exists());
+    }
+}
