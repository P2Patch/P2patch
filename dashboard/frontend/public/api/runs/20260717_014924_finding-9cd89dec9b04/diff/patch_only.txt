+  /**
+   * Resolves a ZIP entry name against the extraction target directory, guarding against
+   * "Zip Slip" path traversal (entry names containing {@code ../}, absolute paths, or other
+   * sequences that would otherwise let an entry be written outside of {@code outputDir}).
+   *
+   * @param outputDir
+   *          the directory the entry must be extracted into.
+   * @param name
+   *          the (already mapped) entry name.
+   * @return the resolved destination file, guaranteed to be located within {@code outputDir}.
+   * @throws ZipException
+   *           if the resolved destination would fall outside of {@code outputDir}.
+   */
+  private static File resolveDestinationFile(File outputDir, String name) {
+    File file = new File(outputDir, name);
+    final String outputDirPath;
+    final String filePath;
+    try {
+      outputDirPath = outputDir.getCanonicalPath();
+      filePath = file.getCanonicalPath();
+    }
+    catch (IOException e) {
+      throw ZipExceptionUtil.rethrow(e);
+    }
+    if (!filePath.equals(outputDirPath) && !filePath.startsWith(outputDirPath + File.separator)) {
+      throw new ZipException("Zip entry '" + name + "' is outside of the target dir: " + outputDir);
+    }
+    return file;
+  }
