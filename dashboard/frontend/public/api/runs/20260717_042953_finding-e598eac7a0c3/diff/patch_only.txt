+     * Confine a caller-supplied log file path to the server's configured task-log root
+     * directory. This is the shared trust-boundary check for every log path that arrives
+     * over the wire (GET_LOG_BYTES_REQUEST, VIEW_WHOLE_LOG_REQUEST, ROLL_VIEW_LOG_REQUEST
+     * and REMOVE_TAK_LOG_REQUEST) so none of them can be used to read or delete arbitrary
+     * files outside the log directory (CWE-22 path traversal).

+     * @param filePath caller-supplied file path
+     * @throws IllegalArgumentException if the path is empty or resolves outside the log root
+     */
+    private void checkPathSecurity(String filePath) {
+        if (filePath == null || filePath.isEmpty()) {
+            throw new IllegalArgumentException("filePath must not be empty");
+        }
+        String logBase = LogUtils.getTaskLogBasePath();
+        try {

+            String targetCanonicalPath = new File(filePath).getCanonicalPath();

+                    && !targetCanonicalPath.startsWith(logBaseCanonicalPath + File.separator)) {
+                throw new IllegalArgumentException(
+                        String.format("filePath [%s] is not within the authorized log directory", filePath));
+            }
