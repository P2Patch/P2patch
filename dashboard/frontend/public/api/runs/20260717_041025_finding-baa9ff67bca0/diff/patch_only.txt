+     * prefix that swallows the traversal token. Without the fix,
+     * {@code normalize("//../secret.txt")} returned the string unchanged
+     * instead of {@code null}, even though the equivalent single-separator
+     * form {@code normalize("/../secret.txt")} correctly returned null.

+    @Test
+    public void testNormalize_rejectsDoubleSeparatorDotDotPrefix() throws Exception {
+        assertEquals(null, FilenameUtils.normalize("/../secret.txt"));
