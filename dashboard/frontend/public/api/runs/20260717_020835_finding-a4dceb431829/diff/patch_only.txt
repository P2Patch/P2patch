+  /**
+   * Resolves a tar archive entry's target file against the destination directory, rejecting
+   * entries whose name would cause the resulting file to escape that directory (e.g. via {@code
+   * ../} sequences or absolute paths), preventing path/zip-slip traversal when extracting
+   * untrusted archives received from a pod.
+   *
+   * @param destinationDir the directory extraction is confined to
+   * @param entryName the raw name of the archive entry
+   * @return the resolved, validated destination file
+   * @throws IOException if the entry would resolve outside of {@code destinationDir}
+   */
+  static File resolveArchiveEntry(File destinationDir, String entryName) throws IOException {
+    File f = new File(destinationDir, entryName);
+    String destDirCanonicalPath = destinationDir.getCanonicalPath();
+    String targetCanonicalPath = f.getCanonicalPath();
+    if (!targetCanonicalPath.equals(destDirCanonicalPath)
+        && !targetCanonicalPath.startsWith(destDirCanonicalPath + File.separator)) {
+      throw new IOException(
+          "Entry with illegal path (path traversal) in tar archive: " + entryName);
+    }
+    return f;
+  }
