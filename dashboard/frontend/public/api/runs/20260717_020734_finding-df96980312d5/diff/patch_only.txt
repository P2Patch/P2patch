+   /**
+    * Resolve a zip entry name against the target extraction directory,
+    * ensuring the resulting path cannot escape that directory
+    * (protects against "zip slip" path traversal via entry names such as
+    * "../../etc/passwd" or absolute paths).
+    *
+    * @param dir target extraction directory
+    * @param entryName raw name taken from the zip entry
+    * @return File instance guaranteed to be located within {@code dir}
+    */
+   private static File validateZipEntryPath(File dir, String entryName) throws IOException
+   {
+      String canonicalDirPath = dir.getCanonicalPath();
+      File file = new File(dir, entryName);
+      String canonicalFilePath = file.getCanonicalPath();
+
+      if (!(canonicalFilePath + File.separator).startsWith(canonicalDirPath + File.separator) && !canonicalFilePath.equals(canonicalDirPath))
+      {
+         throw new IOException("Zip entry is outside of the target directory: " + entryName);
+      }
+
+      return file;
+   }

+/**
+ * Tests to ensure that zip file extraction cannot write files outside
+ * of the target temporary directory (zip slip / path traversal).

+public class InputStreamHelperTest

+   /**
+    * A zip entry using "../" segments to escape the extraction directory
+    * must be rejected rather than silently written outside of it.
+    */
+   @Test public void testZipSlipRelativeTraversalIsRejected() throws Exception
+   {
+      InputStream stream = createZipStream("../../tmp/mpxj-zip-slip-relative.txt", "pwned");
+      try
+      {
+         InputStreamHelper.writeZipStreamToTempDir(stream);
+         fail("Expected an IOException to be raised for a path traversal zip entry");
+      }
+
+      catch (IOException ex)
+      {
+         // expected
+      }
+
+      assertFalse("File must not be written outside of the extraction directory", new File(System.getProperty("java.io.tmpdir"), "mpxj-zip-slip-relative.txt").exists());
+   }
+
+   /**
+    * A zip entry using enough "../" segments to walk past the root of the
+    * filesystem and back down into an absolute target path must still be
+    * rejected, not just simple single-level traversal.

+   @Test public void testZipSlipDeepTraversalToAbsolutePathIsRejected() throws Exception
+   {
+      File target = File.createTempFile("mpxj-zip-slip-absolute", ".txt");

+      InputStream stream = createZipStream("../../../../../../../../" + target.getAbsolutePath(), "pwned");
+      try
+      {

+      }
+
+      catch (IOException ex)
+      {
+         // expected
+      }
+
+      assertFalse("File must not be written outside of the extraction directory", target.exists());
+   }

+   @Test public void testNormalZipEntryIsExtracted() throws Exception

+   /**
+    * Build an in-memory zip stream containing a single entry.
+    *
+    * @param entryName name of the zip entry
+    * @param content content of the zip entry
+    * @return InputStream representing the zip file
+    */
+   private InputStream createZipStream(String entryName, String content) throws IOException
+   {
+      ByteArrayOutputStream bytes = new ByteArrayOutputStream();
+      ZipOutputStream zip = new ZipOutputStream(bytes);
+      zip.putNextEntry(new ZipEntry(entryName));
+      zip.write(content.getBytes());
+      zip.closeEntry();
+      zip.close();
+      return new ByteArrayInputStream(bytes.toByteArray());
+   }
