+import java.io.File;
+import java.io.FileOutputStream;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+/**
+ * Tests that {@link Expand} refuses to extract zip entries whose name would
+ * resolve outside of the requested destination directory ("Zip Slip" /
+ * CWE-22 path traversal).
+ */
+public class ExpandTest
+    extends FileBasedTestCase
+{
+    public void testExpandRejectsZipEntryEscapingDestinationDirectoryViaTraversal()
+        throws Exception
+    {
+        File destDir = new File( getTestDirectory(), "expand-dest-" + System.identityHashCode( this ) );
+        destDir.mkdirs();
+
+        File zipFile = new File( getTestDirectory(), "traversal-" + System.identityHashCode( this ) + ".zip" );
+        createZipWithEntry( zipFile, "../evil.txt", "pwned" );
+
+        Expand expand = new Expand();
+        expand.setSrc( zipFile );
+        expand.setDest( destDir );
+
+        try
+        {
+            expand.execute();
+            fail( "Expected Expand.execute() to reject a zip entry that traverses outside the destination directory" );
+        }
+        catch ( Exception expected )
+        {
+            // expected: entry escaping the destination directory must be rejected
+        }
+
+        File escaped = new File( destDir.getParentFile(), "evil.txt" );
+        assertFalse( "Zip entry must not be written outside the destination directory", escaped.exists() );
+    }
+
+    public void testExpandAllowsNormalZipEntry()
+        throws Exception
+    {
+        File destDir = new File( getTestDirectory(), "expand-dest-ok-" + System.identityHashCode( this ) );
+        destDir.mkdirs();
+
+        File zipFile = new File( getTestDirectory(), "normal-" + System.identityHashCode( this ) + ".zip" );
+        createZipWithEntry( zipFile, "safe/inner.txt", "hello" );
+
+        Expand expand = new Expand();
+        expand.setSrc( zipFile );
+        expand.setDest( destDir );
+        expand.execute();
+
+        File expected = new File( destDir, "safe/inner.txt" );
+        assertTrue( "Normal zip entries must still be extracted into the destination directory", expected.exists() );
+    }
+
+    private void createZipWithEntry( File zipFile, String entryName, String content )
+        throws Exception
+    {
+        FileOutputStream fos = new FileOutputStream( zipFile );
+        ZipOutputStream zos = new ZipOutputStream( fos );
+        try
+        {
+            ZipEntry entry = new ZipEntry( entryName );
+            zos.putNextEntry( entry );
+            zos.write( content.getBytes( "UTF-8" ) );
+            zos.closeEntry();
+        }
+        finally
+        {
+            zos.close();
+        }
+    }
+}
