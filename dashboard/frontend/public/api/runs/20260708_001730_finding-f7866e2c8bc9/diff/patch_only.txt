+     * @param destination the destination directory the entry must stay within

+    {
+        try
+        {
+            final File canonicalDestination = destination.getCanonicalFile();
+            File current = f.getCanonicalFile();
+            while ( current != null )
+            {
+                if ( current.equals( canonicalDestination ) )
+                {
+                    return true;
+                }
+                current = current.getParentFile();
+            }

+package org.codehaus.plexus.archiver.tar;
+
+import java.io.File;
+import java.io.FileOutputStream;
+import java.io.OutputStream;
+import org.apache.commons.compress.archivers.tar.TarArchiveEntry;
+import org.apache.commons.compress.archivers.tar.TarArchiveOutputStream;
+import org.codehaus.plexus.PlexusTestCase;
+import org.codehaus.plexus.archiver.ArchiverException;
+import org.codehaus.plexus.archiver.UnArchiver;
+import org.codehaus.plexus.util.FileUtils;
+
+/**
+ * Regression test for CWE-022 path traversal ("Zip Slip") in
+ * {@link org.codehaus.plexus.archiver.AbstractUnArchiver#extractFile}. A tar
+ * entry whose name traverses out of the destination directory via {@code ../}
+ * must not be written outside the destination.
+ */
+public class TarUnArchiverPathTraversalTest extends PlexusTestCase
+{
+
+    public void testEntryEscapingDestinationIsRejected() throws Exception
+    {
+        File baseDir = new File( getBasedir(), "target/tar-path-traversal-tests" );
+        File destDir = new File( baseDir, "dest" );
+        File maliciousTar = new File( baseDir, "malicious.tar" );
+
+        FileUtils.deleteDirectory( baseDir );
+        destDir.mkdirs();
+
+        // Build a tar archive containing an entry that escapes the destination.
+        String escapingEntry = "../pov-escaped-marker.txt";
+        TarArchiveOutputStream tos = null;
+        try
+        {
+            OutputStream fos = new FileOutputStream( maliciousTar );
+            tos = new TarArchiveOutputStream( fos );
+            tos.setLongFileMode( TarArchiveOutputStream.LONGFILE_GNU );
+            byte[] content = "owned".getBytes( "UTF-8" );
+            TarArchiveEntry entry = new TarArchiveEntry( escapingEntry );
+            entry.setSize( content.length );
+            tos.putArchiveEntry( entry );
+            tos.write( content );
+            tos.closeArchiveEntry();
+            tos.finish();
+        }
+        finally
+        {
+            if ( tos != null )
+            {
+                tos.close();
+            }
+        }
+
+        File escaped = new File( baseDir, "pov-escaped-marker.txt" );
+        assertFalse( "Precondition: escaped marker must not exist yet", escaped.exists() );
+
+        TarUnArchiver unArchiver = (TarUnArchiver) lookup( UnArchiver.ROLE, "tar" );
+        unArchiver.setSourceFile( maliciousTar );
+        unArchiver.setDestDirectory( destDir );
+
+        try
+        {
+            unArchiver.extract();
+            fail( "Extraction of a path-traversing entry should have failed" );
+        }
+        catch ( ArchiverException expected )
+        {
+            // expected: traversal is rejected
+        }
+
+        assertFalse( "Path-traversing entry escaped the destination directory",
+                     escaped.exists() );
+    }
+
+}
