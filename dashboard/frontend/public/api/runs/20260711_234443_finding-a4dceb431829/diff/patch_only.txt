+   * Builds a base64-encoded, gzip-compressed tar stream (matching what the pod-side {@code tar cz -
+   * | base64} command emits) containing the given (name, content) entries.
+   */
+  private static InputStream buildEncodedTar(String[] names, byte[][] contents) throws IOException {
+    ByteArrayOutputStream base64Out = new ByteArrayOutputStream();
+    try (TarArchiveOutputStream tar =
+        new TarArchiveOutputStream(
+            new GzipCompressorOutputStream(new Base64OutputStream(base64Out, true, 0, null)))) {
+      tar.setLongFileMode(TarArchiveOutputStream.LONGFILE_GNU);
+      for (int i = 0; i < names.length; i++) {
+        TarArchiveEntry entry = new TarArchiveEntry(names[i]);
+        entry.setSize(contents[i].length);
+        tar.putArchiveEntry(entry);
+        tar.write(contents[i]);
+        tar.closeArchiveEntry();
+      }
+    }
+    return new ByteArrayInputStream(base64Out.toByteArray());
+  }
