+     * invalid path segment. A segment is considered invalid if it consists of
+     * more than two consecutive dots (e.g. {@code ...}). Such segments are not
+     * normalized away by {@link ResourceUtil#normalize(String)} and can be used
+     * for path traversal (SLING-12233 / CVE-2024-23673).
+     *
+     * @param path the resource type or script path to check (may be {@code null})
+     * @return {@code true} if the path contains an invalid segment
+     */
+    static boolean isInvalidPath(final String path) {
+        if (path == null || path.length() == 0) {
+            return false;
+        }
+        int dotCount = 0;
+        for (int i = 0; i < path.length(); i++) {
+            final char c = path.charAt(i);
+            if (c == '/') {
+                if (dotCount > 2) {
+                    return true;
+                }
+                dotCount = 0;
+            } else if (c == '.') {

+            } else {

+            }

+/**
+ * Regression test for SLING-12233 / CVE-2024-23673 (CWE-022 path traversal).
+ * Resource types / script paths that contain a segment with more than two
+ * consecutive dots (e.g. {@code ...}) must be rejected because such segments
+ * are not normalized away and can be abused for path traversal.
+ */
+public class SlingServletResolverInvalidPathTest {
+
+    @Test
+    public void rejectsTripleDotSegment() {
+        assertTrue(SlingServletResolver.isInvalidPath("/libs/proxy/.../uploaded.servlet"));
+    }
+
+    @Test
+    public void rejectsTrailingTripleDotSegment() {
+        assertTrue(SlingServletResolver.isInvalidPath("/libs/proxy/..."));
+    }
+
+    @Test
+    public void rejectsManyDotsSegment() {
+        assertTrue(SlingServletResolver.isInvalidPath("/apps/...../foo"));
+    }
+
+    @Test
+    public void allowsRegularPaths() {
+        assertFalse(SlingServletResolver.isInvalidPath("/libs/foo/bar.servlet"));
+        assertFalse(SlingServletResolver.isInvalidPath("foo/bar"));
+    }
+
+    @Test
+    public void allowsSingleAndDoubleDotSegments() {
+        // single dot and double dot are handled/normalized elsewhere
+        assertFalse(SlingServletResolver.isInvalidPath("/libs/./foo"));
+        assertFalse(SlingServletResolver.isInvalidPath("/libs/../foo"));
+    }
+
+    @Test
+    public void allowsDotsInsideSegment() {
+        assertFalse(SlingServletResolver.isInvalidPath("/libs/foo.bar.baz.servlet"));
+    }
+
+    @Test
+    public void handlesNullAndEmpty() {
+        assertFalse(SlingServletResolver.isInvalidPath(null));
+        assertFalse(SlingServletResolver.isInvalidPath(""));
+    }
