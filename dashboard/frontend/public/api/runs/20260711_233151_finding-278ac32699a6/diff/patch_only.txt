+/**
+ * Verifies that {@link DirectoryTraversal} correctly rejects paths that escape the
+ * configured static files root, including the sibling-folder prefix bypass where a
+ * resolved path merely shares the root folder name as a string prefix.
+ */
+public class DirectoryTraversalTest {
+
+    @Test
+    public void allowsFileInsideConfiguredExternalFolder() {
+        String root = new File(System.getProperty("java.io.tmpdir"), "pub").getAbsolutePath();
+        StaticFilesFolder.externalConfiguredTo(root);
+
+        // A regular file inside the configured folder must be accepted.
+        DirectoryTraversal.protectAgainstForExternal(Paths.get(root, "index.html").toString());
+    }
+
+    @Test(expected = DirectoryTraversal.DirectoryTraversalDetection.class)
+    public void rejectsSiblingFolderSharingRootAsPrefix() {
+        String base = System.getProperty("java.io.tmpdir");
+        String root = new File(base, "pub").getAbsolutePath();
+        StaticFilesFolder.externalConfiguredTo(root);
+
+        // "pub-secret" shares "pub" as a plain string prefix but is NOT inside "pub".
+        // A naive startsWith check would wrongly accept it (path traversal bypass).
+        String siblingFile = new File(base, "pub-secret" + File.separator + "passwd").getAbsolutePath();
+        DirectoryTraversal.protectAgainstForExternal(siblingFile);
+    }
+
+    @Test(expected = DirectoryTraversal.DirectoryTraversalDetection.class)
