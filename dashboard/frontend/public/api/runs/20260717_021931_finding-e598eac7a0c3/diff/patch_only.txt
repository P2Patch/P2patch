+    /**
+     * Verify that an externally supplied log file path is confined to the configured
+     * task log root directory. This guards against path traversal (e.g. "../../etc/passwd",
+     * absolute paths outside of the log root, or symlink-based escapes) since these paths
+     * are taken directly from remote {@link Command} payloads and are otherwise untrusted.
+     *
+     * @param filePath externally supplied file path
+     * @return true if the path is missing or resolves outside of the log root, i.e. illegal
+     */
+    private boolean isLogPathIllegal(String filePath) {
+        if (StringUtils.isBlank(filePath)) {
+            return true;
+        }
+        String logBasePath = LogUtils.getTaskLogBasePath();
+        if (StringUtils.isBlank(logBasePath)) {
+            logger.warn("Log base path is not configured, deny log path request: {}", filePath);
+            return true;
+        }
+        try {
+            Path base = Paths.get(logBasePath).toAbsolutePath().normalize();
+            Path target = Paths.get(filePath).toAbsolutePath().normalize();
+            if (!target.startsWith(base)) {
+                logger.warn("Log path: {} is outside of the log base path: {}, deny request", filePath, logBasePath);
+                return true;
+            }
+            return false;
+        } catch (Exception e) {
+            logger.warn("Failed to validate log path: {}", filePath, e);
+            return true;
+        }
+    }

+    /**
+     * get the root directory that all task logs are written under.
+     * used to confine externally supplied log paths (e.g. from remote log view/download
+     * requests) so that they cannot escape outside of the configured log directory.
+     */
+    public static String getTaskLogBasePath() {
+        // Optional.map will be skipped if null
+        return Optional.of(LoggerFactory.getILoggerFactory())
+                .map(e -> (AppenderAttachable<ILoggingEvent>) (e.getLogger("ROOT")))
+                .map(e -> (SiftingAppender) (e.getAppender("TASKLOGFILE")))
+                .map(e -> ((TaskLogDiscriminator) (e.getDiscriminator())))
+                .map(TaskLogDiscriminator::getLogBase)
+                .map(e -> Paths.get(e).toAbsolutePath().normalize())
+                .map(Path::toString)
+                .orElse("");
+    }
