+  /**
+   * Resolves a tar entry name against the extraction destination, guaranteeing that the
+   * resolved path is equal to, or a descendant of, the canonical destination path. This
+   * guards against "tar slip"/zip-slip path traversal (CWE-22) where a malicious or
+   * compromised container returns archive entries with names such as
+   * "../../etc/cron.d/evil" or an absolute path, which would otherwise let the extraction
+   * loop create directories or write files outside of {@code destinationDir}.
+   *
+   * @throws IOException if the entry name resolves to a path outside of destinationDir
+   */
+  private static File resolveEntry(File destinationDir, String entryName) throws IOException {
+    File f = new File(destinationDir, entryName);
+    String destinationCanonicalPath = destinationDir.getCanonicalPath();
+    String targetCanonicalPath = f.getCanonicalPath();
+    if (!targetCanonicalPath.equals(destinationCanonicalPath)
+        && !targetCanonicalPath.startsWith(destinationCanonicalPath + File.separator)) {
+      throw new IOException(
+          "Entry "
+              + entryName
+              + " is outside of the target destination directory "
+              + destinationCanonicalPath
+              + " (resolved to "
+              + targetCanonicalPath
+              + "); refusing to extract");
+    }
+    return f;
+  }

+  private static byte[] buildMaliciousTarGzBase64(
