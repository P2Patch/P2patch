+    /**
+     * Check whether the given multi-segment directory path (e.g. "/a/b/c") can escape its intended
+     * root via path traversal. Unlike {@link #directoryTraversal(String)} -- which validates a single
+     * path segment / leaf name and rejects any "/" -- this method validates a full, "/"-separated
+     * directory path such as the resource center's "currentDir" parameter, by rejecting ".." segments
+     * and backslashes (which are not treated as separators when building Hadoop paths, but could be
+     * used to smuggle a traversal segment past naive checks).
+     *
+     * @param dirPath String type of directory path
+     * @return true if the directory path contains a traversal attempt or is invalid
+     */
+    public static boolean directoryPathTraversal(String dirPath){
+        if (dirPath == null || dirPath.contains("\\")) {
+            return true;
+        }
+        for (String segment : dirPath.split(FOLDER_SEPARATOR, -1)) {
+            if ("..".equals(segment.trim())) {
+                return true;
+            }
+        }
+        return false;
+    }
