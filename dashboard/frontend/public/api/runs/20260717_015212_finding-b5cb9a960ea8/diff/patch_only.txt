+    /**
+     * Validates that the given extraction target resides within the
+     * expected destination directory. This guards against maliciously
+     * crafted archive entry names (e.g. containing {@code ../}) that
+     * would otherwise allow files to be written outside of the intended
+     * extraction directory (a "Zip Slip" / path traversal attack).
+     *
+     * @param destination the directory files are expected to be extracted
+     * into
+     * @param file the resolved extraction target for an archive entry
+     * @param entryName the raw archive entry name, used for error
+     * reporting
+     * @throws AnalysisException thrown if the resolved extraction target
+     * falls outside of the destination directory
+     */
+    private static void validateExtractionTarget(File destination, File file, String entryName) throws AnalysisException {
+        final String destinationPath;
+        final String filePath;
+        try {
+            destinationPath = destination.getCanonicalPath();
+            filePath = file.getCanonicalPath();
+        } catch (IOException ex) {
+            throw new AnalysisException(String.format("Unable to validate the extraction path for entry '%s'.", entryName), ex);
+        }
+        if (!filePath.equals(destinationPath) && !filePath.startsWith(destinationPath + File.separator)) {
+            throw new AnalysisException(String.format(
+                    "Archive entry '%s' attempts to extract outside of the target directory; extraction blocked.", entryName));
+        }
+    }
