+package com.alibaba.oneagent.utils;
+
+import java.io.File;
+import java.io.FileOutputStream;
+import java.io.IOException;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+import org.junit.Assert;
+import org.junit.Test;
+
+/**
+ * Regression test for the Zip Slip (CWE-022 path traversal) fix in
+ * {@link IOUtils#unzip(String, String)}.
+ */
+public class IOUtilsUnzipTest {
+
+    private static File newTempDir(String prefix) throws IOException {
+        File dir = File.createTempFile(prefix, "");
+        Assert.assertTrue(dir.delete());
+        Assert.assertTrue(dir.mkdirs());
+        dir.deleteOnExit();
+        return dir;
+    }
+
+    @Test
+    public void testUnzipNormalEntry() throws IOException {
+        File work = newTempDir("oneagent-unzip-normal");
+        File zipFile = new File(work, "good.zip");
+        ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zipFile));
+        try {
+            zos.putNextEntry(new ZipEntry("sub/hello.txt"));
+            zos.write("hello".getBytes("UTF-8"));
+            zos.closeEntry();
+        } finally {
+            zos.close();
+        }
+
+        File extract = new File(work, "extract");
+        IOUtils.unzip(zipFile.getAbsolutePath(), extract.getAbsolutePath());
+
+        File extracted = new File(extract, "sub/hello.txt");
+        Assert.assertTrue("Normal entry should be extracted", extracted.isFile());
+    }
+
+    @Test
+    public void testUnzipRejectsPathTraversal() throws IOException {
+        File work = newTempDir("oneagent-unzip-evil");
+        File zipFile = new File(work, "evil.zip");
+        ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zipFile));
+        try {
+            zos.putNextEntry(new ZipEntry("../../evil.txt"));
+            zos.write("pwned".getBytes("UTF-8"));
+            zos.closeEntry();
+        } finally {
+            zos.close();
+        }
+
+        File extract = new File(work, "extract");
+        try {
+            IOUtils.unzip(zipFile.getAbsolutePath(), extract.getAbsolutePath());
+            Assert.fail("Expected IOException for path traversal entry");
+        } catch (IOException expected) {
+            // expected
+        }
+
+        File escaped = new File(work.getParentFile(), "evil.txt");
+        Assert.assertFalse("Traversal entry must not escape target directory", escaped.exists());
+    }
+}
