+package net.sf.mpxj.junit;
+
+import static org.junit.Assert.*;
+
+import java.io.ByteArrayInputStream;
+import java.io.ByteArrayOutputStream;
+import java.io.File;
+import java.io.IOException;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+import net.sf.mpxj.common.FileHelper;
+import net.sf.mpxj.common.InputStreamHelper;
+
+import org.junit.Test;
+
+/**
+ * Tests to exercise InputStreamHelper, in particular protection against
+ * Zip Slip (CWE-022) path traversal when extracting zip streams.
+ */
+public class InputStreamHelperTest
+{
+   /**
+    * Ensure a zip entry whose name attempts to traverse outside of the
+    * extraction directory is rejected and does not write a file to the
+    * parent directory.
+    */
+   @Test public void testZipSlipEntryRejected() throws Exception
+   {
+      byte[] zipBytes = createZip("../mpxj_zipslip_regression.txt", "malicious");
+
+      File dir = null;
+      File escaped = null;
+      try
+      {
+         try
+         {
+            dir = InputStreamHelper.writeZipStreamToTempDir(new ByteArrayInputStream(zipBytes));
+         }
+
+         catch (IOException ex)
+         {
+            // Rejecting the traversal entry with an exception is acceptable.
+         }
+
+         // The malicious entry must never be written outside the extraction dir.
+         File tempRoot = File.createTempFile("mpxj", "probe").getParentFile();
+         escaped = new File(tempRoot, "mpxj_zipslip_regression.txt");
+         assertFalse("Zip Slip entry escaped the extraction directory", escaped.exists());
+      }
+
+      finally
+      {
+         if (dir != null)
+         {
+            FileHelper.deleteQuietly(dir);
+         }
+         if (escaped != null && escaped.exists())
+         {
+            FileHelper.deleteQuietly(escaped);
+         }
+      }
+   }
+
+   /**
+    * Ensure a well-formed zip entry is still extracted correctly.
+    */
+   @Test public void testValidEntryExtracted() throws Exception
+   {
+      byte[] zipBytes = createZip("nested/data.txt", "hello");
+
+      File dir = InputStreamHelper.writeZipStreamToTempDir(new ByteArrayInputStream(zipBytes));
+      try
+      {
+         File extracted = new File(dir, "nested/data.txt");
+         assertTrue("Valid zip entry was not extracted", extracted.exists());
+      }
+
+      finally
+      {
+         FileHelper.deleteQuietly(dir);
+      }
+   }
+
+   /**
+    * Build an in-memory zip file containing a single entry.
+    *
+    * @param entryName name of the zip entry
+    * @param content entry content
+    * @return zip file bytes
+    */
+   private byte[] createZip(String entryName, String content) throws IOException
+   {
+      ByteArrayOutputStream baos = new ByteArrayOutputStream();
+      ZipOutputStream zos = new ZipOutputStream(baos);
+      ZipEntry entry = new ZipEntry(entryName);
+      zos.putNextEntry(entry);
+      zos.write(content.getBytes("UTF-8"));
+      zos.closeEntry();
+      zos.close();
+      return baos.toByteArray();
+   }
+}
