+    /**
+     * check whether the given directory path attempts a path traversal (contains a ".." segment).
+     * The parent directory legitimately contains path separators, so {@link FileUtils#directoryTraversal(String)}
+     * (which rejects any separator) cannot be used here; instead each path segment is inspected.
+     *
+     * @param currentDir the user supplied parent directory
+     * @return true if the path contains a traversal segment and must be rejected
+     */
+    private static boolean checkDirectoryTraversal(String currentDir) {
+        if (StringUtils.isEmpty(currentDir)) {
+            return false;
+        }
+        for (String segment : currentDir.split("/")) {
+            if ("..".equals(segment.trim())) {
+                return true;
+            }
