+	/**
+	 * Guards against "Zip Slip" path traversal by ensuring the resolved output file
+	 * stays within the requested output directory. A malicious archive can contain
+	 * entry names such as "../../evil.txt" which would otherwise cause files to be
+	 * written outside the intended extraction directory.
+	 */
+	private void assertValidOutputPath(String outPath, File file) throws ZipException {
+		try {
+			File outDir = new File(outPath).getCanonicalFile();
+			String canonicalOutDir = outDir.getPath();
+			String canonicalFile = file.getCanonicalPath();
+
+			if (!canonicalFile.equals(canonicalOutDir)
+					&& !canonicalFile.startsWith(canonicalOutDir + File.separator)) {
+				throw new ZipException("illegal file name that breaks out of the target directory: "
+						+ file.getName());
+			}
+		} catch (IOException e) {
+			throw new ZipException(e);
