+	 * safe to write directly into the "clean" CSS text produced by
+	 * {@link org.owasp.validator.css.CssHandler}. That text is written
+	 * verbatim into DOM attribute/text node values by
+	 * {@code AntiSamyDOMScanner} (e.g. the "style" attribute, or the text
+	 * content of a &lt;style&gt; tag) with no further HTML-neutralization,
+	 * and is reachable by any caller of the public, documented
+	 * {@code CleanResults.getCleanXMLDocumentFragment()} API - not only by
+	 * consumers that happen to route the DOM back through AntiSamy's own
+	 * HTML serializer. Unlike {@link #lexicalValueToString(LexicalUnit)}
+	 * (which returns the raw, decoded token value and is used purely for
+	 * policy validation), this method HTML-entity-encodes any '"', ''',
+	 * '&lt;', '&gt;' or '&amp;' character present in a string/identifier or
+	 * URI value so that CSS-derived, attacker-controlled content can never
+	 * introduce a raw HTML/XML-significant character into the DOM.
+	 *
+	 * @param lu
+	 *            the lexical unit to convert
+	 * @return an HTML-safe <code>String</code> representation of the given
+	 *         lexical unit, suitable for inclusion in generated CSS output
+	 */
+	public String lexicalValueToOutputString(LexicalUnit lu) {
+		switch (lu.getLexicalUnitType()) {
+		case LexicalUnit.SAC_STRING_VALUE:
+		case LexicalUnit.SAC_IDENT: {
+			String stringValue = lu.getStringValue();
+			String escaped = htmlEscapeSensitiveChars(stringValue);
+			if (!escaped.equals(stringValue)) {
+				// The value contains characters that are significant to an
+				// HTML parser. Never wrap already-escaped content in raw
+				// quotes here - those quotes would themselves be usable to
+				// break out of a surrounding HTML attribute value.
+				return escaped;
+			}
+			if (stringValue.indexOf(' ') != -1) {
+				return "\"" + stringValue + "\"";
+			}
+			return stringValue;
+		}
+		case LexicalUnit.SAC_URI:
+			// this is a URL
+			return "url(" + htmlEscapeSensitiveChars(lu.getStringValue()) + ")";
+		default:
+			return lexicalValueToString(lu);
+		}
+	}

+	/**
+	 * Escapes any character that is significant to an HTML parser
+	 * ('"', ''', '&lt;', '&gt;', '&amp;') using standard numeric/named HTML
+	 * entities. Every other character (including CSS syntax characters such
+	 * as ':', ';', '-', '(', ')', ',' and whitespace) is left untouched so
+	 * that ordinary CSS structure is preserved.
+	 *
+	 * @param value
+	 *            the raw, decoded CSS token value
+	 * @return the value with HTML-sensitive characters entity-encoded; the
+	 *         same String instance is returned unchanged if no such
+	 *         characters are present
+	 */
+	private static String htmlEscapeSensitiveChars(String value) {
+		StringBuilder sb = null;
+		for (int i = 0; i < value.length(); i++) {
+			char ch = value.charAt(i);
+			String entity;
+			switch (ch) {
+			case '"':
+				entity = "&#34;";
+				break;
+			case '\'':
+				entity = "&#39;";
+				break;
+			case '<':
+				entity = "&lt;";
+				break;
+			case '>':
+				entity = "&gt;";
+				break;
+			case '&':
+				entity = "&amp;";
+				break;
+			default:
+				entity = null;
+			}
+
+			if (entity != null) {
+				if (sb == null) {
+					sb = new StringBuilder(value.length() + 16);
+					sb.append(value, 0, i);
+				}
+				sb.append(entity);
+			} else if (sb != null) {
+				sb.append(ch);
+			}
+		}
+		return sb == null ? value : sb.toString();
+	}

+	 * Neutralizes any literal '&lt;' character in a CSS selector or
+	 * attribute-condition string that is about to be written into generated
+	 * CSS output. Selector text can legitimately contain structurally
+	 * significant characters (for example the '&gt;' child combinator) that
+	 * must be preserved verbatim, so - unlike
+	 * {@link #lexicalValueToOutputString(LexicalUnit)} - only '&lt;' is
+	 * neutralized here; it is the one character that could otherwise allow
+	 * CSS-derived selector text written into the text content of a
+	 * &lt;style&gt; tag to be confused with the start of a closing
+	 * &lt;/style&gt; tag (or any other markup) by an HTML parser.
+	 *
+	 * @param selectorText
+	 *            the raw selector or condition text (as produced by
+	 *            {@code Selector#toString()}/{@code Condition#toString()})
+	 * @return the selector text with any literal '&lt;' HTML-entity-encoded
+	 */
+	public static String escapeSelectorForOutput(String selectorText) {
+		if (selectorText.indexOf('<') == -1) {
+			return selectorText;
+		}
+		StringBuilder sb = new StringBuilder(selectorText.length() + 16);
+		for (int i = 0; i < selectorText.length(); i++) {
+			char ch = selectorText.charAt(i);
+			if (ch == '<') {
+				sb.append("&lt;");
+			} else {
+				sb.append(ch);
+			}
+		}
+		return sb.toString();
+	}

+    private static String getFirstElementAttr(org.w3c.dom.Node root, String tagName, String attrName) {
+        org.w3c.dom.Element el = findFirstElement(root, tagName);
+        return el == null ? null : el.getAttribute(attrName);
+    }
+
+    private static String getFirstElementText(org.w3c.dom.Node root, String tagName) {
+        org.w3c.dom.Element el = findFirstElement(root, tagName);
+        if (el == null || el.getFirstChild() == null) {
+            return null;
+        }
+        return el.getFirstChild().getNodeValue();
+    }
+
+    private static org.w3c.dom.Element findFirstElement(org.w3c.dom.Node node, String tagName) {
+        if (node instanceof org.w3c.dom.Element && tagName.equalsIgnoreCase(((org.w3c.dom.Element) node).getTagName())) {
+            return (org.w3c.dom.Element) node;
+        }
+        org.w3c.dom.NodeList children = node.getChildNodes();
+        for (int i = 0; i < children.getLength(); i++) {
+            org.w3c.dom.Element found = findFirstElement(children.item(i), tagName);
+            if (found != null) {
+                return found;
+            }
+        }
+        return null;
+    }
