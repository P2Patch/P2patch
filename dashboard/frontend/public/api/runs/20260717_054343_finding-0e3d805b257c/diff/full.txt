+/*
+ * PoV for CWE-22 (Path Traversal) in the Apache Sling Servlets Resolver
+ * script-resolution machinery flagged by the finder alert.
+ *
+ * Root cause: org.apache.sling.servlets.resolver.internal.helper
+ * .NamedScriptResourceCollector#getWeightedResources(Set, Resource) builds an
+ * intermediate resource path by concatenating the caller-supplied
+ * "script name" directly onto the (trusted) location path:
+ *
+ *   current = getResource(resolver, location.getPath() + '/' +
+ *                          this.scriptName.substring(0, pos), useResourceCaching);
+ *
+ * Unlike the sibling branch a few lines above it in the very same method
+ * (the "extension" branch, which normalizes the path with
+ * ResourceUtil.normalize(...) and gates it through
+ * SlingServletResolver.isPathAllowed(...) before resolving it), this branch
+ * performs no normalization and no allow-list check on the path it builds
+ * before resolving it via the (real, package-visible) AbstractResourceCollector
+ * static helpers, which bottom out in ResourceResolver#getResource(String) --
+ * the same generic sink flagged repeatedly throughout the alert's trace
+ * (AbstractResourceCollector.getResourceOrNull -> ScriptResourceResolver ->
+ * MergingServletResourceProvider -> ServletResourceProvider -> ServletResource
+ * -> SlingServletResolver -> ScriptResource).
+ *
+ * "scriptName" is attacker/caller-influenced input: it is the literal
+ * "scriptName" argument of the two PUBLIC ServletResolver API methods
+ * SlingServletResolver#resolveServlet(Resource, String) and
+ * #resolveServlet(ResourceResolver, String), which route (when passed a
+ * request-less caller, i.e. any consumer that is not the request dispatcher
+ * itself -- e.g. SlingScriptHelper#getScript()-style "named script"
+ * lookups used by includes/templates) straight into
+ * NamedScriptResourceCollector.create(scriptNameOrResourceType, resource, ...).
+ *
+ * This test drives the real, unmodified NamedScriptResourceCollector /
+ * AbstractResourceCollector production code (no product code is patched or
+ * stubbed) and shows that a scriptName containing "../" segments causes the
+ * collector to resolve and return a script resource that lives completely
+ * outside the restricted directory it is supposed to be confined to (the
+ * resource-type-derived script folder of the resource being rendered,
+ * e.g. "/apps/myapp/comp"), reaching an unrelated, sensitive resource
+ * ("/secret/admin/leak.esp") elsewhere in the repository.
+ *
+ * This test PASSES (exit 0) on the vulnerable code, because it asserts that
+ * the out-of-tree resource IS found -- i.e. it asserts the presence of the
+ * vulnerable behavior. After a correct fix (normalizing + isPathAllowed
+ * -checking the intermediate path the same way the sibling branch already
+ * does, and/or rejecting scriptName values that escape the location subtree)
+ * the traversal result must no longer be resolvable, so this same assertion
+ * FAILS (non-zero exit), which is how a downstream patch verification step
+ * detects that the vulnerability has been closed.
+ */

+/**
+ * Regression test for CWE-22 (path traversal) in
+ * {@link NamedScriptResourceCollector#getWeightedResources(java.util.Set, Resource)}.
+ * <p>
+ * A caller-supplied "script name" containing "../" segments must not be able
+ * to escape the resource's own resourceType script folder (here
+ * "/apps/foo/bar") and reach an unrelated resource elsewhere in the
+ * repository, even when no {@code executionPaths} allow-list is configured
+ * (the common, unrestricted default).
+ */
+public class NamedScriptResourceCollectorTest extends HelperTestBase {
+
+    public void testScriptNameCannotEscapeLocationSubtreeViaDotDotSegments() {
+        // a legitimate script sitting in the resource type's own script folder
+        // ("/apps/" + resourceTypePath, i.e. "/apps/foo/bar")
+        String legitPath = "/apps/" + resourceTypePath + "/legit.esp";
+        addOrReplaceResource(resourceResolver, legitPath, "nt:file");
+
+        // a sensitive resource that lives completely outside of the
+        // resourceType's own script tree
+        String secretPath = "/secret/admin/leak.esp";
+        addOrReplaceResource(resourceResolver, secretPath, "nt:file");
+
+        // "/apps/foo/bar" has 3 path segments (apps, foo, bar), so 3 "../"
+        // segments are enough to climb out to the repository root before
+        // descending into "secret/admin/leak"
+        String maliciousScriptName = "../../../secret/admin/leak";
+
+        NamedScriptResourceCollector collector = NamedScriptResourceCollector.create(
+                maliciousScriptName, resource, null /* executionPaths: unconfigured/unrestricted */, false);
+
+        Collection<Resource> found = collector.getServlets(resourceResolver, Collections.emptyList());
+        for (Resource r : found) {
+            assertFalse("must not resolve a script outside the resourceType's own script tree, found: " + r.getPath(),
+                    r.getPath().startsWith("/secret"));
+        }
+
+        // sanity: the legitimate, in-tree script is unaffected by the fix
+        assertNotNull(resourceResolver.getResource(legitPath));
+    }
