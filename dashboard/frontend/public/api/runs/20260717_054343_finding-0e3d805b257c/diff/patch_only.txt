+/**
+ * Regression test for CWE-22 (path traversal) in
+ * {@link NamedScriptResourceCollector#getWeightedResources(java.util.Set, Resource)}.
+ * <p>
+ * A caller-supplied "script name" containing "../" segments must not be able
+ * to escape the resource's own resourceType script folder (here
+ * "/apps/foo/bar") and reach an unrelated resource elsewhere in the
+ * repository, even when no {@code executionPaths} allow-list is configured
+ * (the common, unrestricted default).
+ */
+public class NamedScriptResourceCollectorTest extends HelperTestBase {
+
+    public void testScriptNameCannotEscapeLocationSubtreeViaDotDotSegments() {
+        // a legitimate script sitting in the resource type's own script folder
+        // ("/apps/" + resourceTypePath, i.e. "/apps/foo/bar")
+        String legitPath = "/apps/" + resourceTypePath + "/legit.esp";
+        addOrReplaceResource(resourceResolver, legitPath, "nt:file");
+
+        // a sensitive resource that lives completely outside of the
+        // resourceType's own script tree
+        String secretPath = "/secret/admin/leak.esp";
+        addOrReplaceResource(resourceResolver, secretPath, "nt:file");
+
+        // "/apps/foo/bar" has 3 path segments (apps, foo, bar), so 3 "../"
+        // segments are enough to climb out to the repository root before
+        // descending into "secret/admin/leak"
+        String maliciousScriptName = "../../../secret/admin/leak";
+
+        NamedScriptResourceCollector collector = NamedScriptResourceCollector.create(
+                maliciousScriptName, resource, null /* executionPaths: unconfigured/unrestricted */, false);
+
+        Collection<Resource> found = collector.getServlets(resourceResolver, Collections.emptyList());
+        for (Resource r : found) {
+            assertFalse("must not resolve a script outside the resourceType's own script tree, found: " + r.getPath(),
+                    r.getPath().startsWith("/secret"));
+        }
+
+        // sanity: the legitimate, in-tree script is unaffected by the fix
+        assertNotNull(resourceResolver.getResource(legitPath));
+    }
