+  /**
+   * Resolves {@code entryName} against {@code targetDir} and verifies that the resulting
+   * path is contained within {@code targetDir}. This guards against "Zip Slip" path
+   * traversal attacks where a malicious ZIP entry name (e.g. containing {@code ../} segments
+   * or an absolute path) would otherwise cause extraction outside of the intended directory.
+   *
+   * @param targetDir
+   *          the directory entries are expected to be extracted into.
+   * @param entryName
+   *          the (already name-mapped) entry name to resolve, purely for error reporting.
+   * @param destFile
+   *          the candidate destination file, typically {@code new File(targetDir, entryName)}
+   *          or an intermediate path built up from path segments of the entry name.
+   * @return {@code destFile}, for convenient chaining.
+   * @throws ZipException
+   *           if {@code destFile} would resolve outside of {@code targetDir}.
+   */
+  private static File checkDestinationInsideTargetDir(File targetDir, String entryName, File destFile) {
+    try {
+      String targetDirPath = targetDir.getCanonicalPath() + File.separator;
+      String destFilePath = destFile.getCanonicalPath();
+      if (!destFilePath.equals(targetDir.getCanonicalPath()) && !destFilePath.startsWith(targetDirPath)) {
+        throw new ZipException("Extracting entry '" + entryName + "' would create a file outside of the target directory '" + targetDir + "' (Zip Slip)");
+      }
+    }
+    catch (IOException e) {
+      throw ZipExceptionUtil.rethrow(e);
+    }
+    return destFile;
+  }

+  private static File zipWithEntry(String entryName, byte[] contents) throws IOException {
+    File zip = File.createTempFile("zip-slip-src", ".zip");
+    ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zip));
+    try {
+      zos.putNextEntry(new ZipEntry(entryName));
+      zos.write(contents);
+      zos.closeEntry();
+    }
+    finally {
+      IOUtils.closeQuietly(zos);
+    }
+    return zip;
+  }
