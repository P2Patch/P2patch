+  /**
+   * Verifies that {@code file} (already resolved relative to some directory nested under
+   * {@code outputDir}) is actually contained within {@code outputDir}. Used by callers that
+   * build up destination paths incrementally (e.g. per path segment).
+   *
+   * @throws ZipException if the resolved path escapes {@code outputDir}.
+   */
+  private static File checkEntryDestination(File outputDir, File file, String name) throws IOException {
+    String canonicalOutputPath = outputDir.getCanonicalPath();
+    String canonicalDestPath = file.getCanonicalPath();
+    if (!canonicalDestPath.equals(canonicalOutputPath) && !canonicalDestPath.startsWith(canonicalOutputPath + File.separator)) {
+      throw new ZipException("Zip entry '" + name + "' is outside of the target directory '" + outputDir + "' (path traversal / zip-slip attempt)");
+    }
+    return file;
+  }

+  private static File createTempDir(String prefix) throws IOException {
+    File dest = File.createTempFile(prefix, null);
+    if (!dest.delete()) {
+      throw new IOException("Could not delete temp file: " + dest.getAbsolutePath());
+    }
+    if (!dest.mkdir()) {
+      throw new IOException("Could not create temp directory: " + dest.getAbsolutePath());
+    }
+    return dest;
+  }

+  private static File createZipWithEntry(String entryName, String content) throws IOException {
+    File zip = File.createTempFile("zip-slip-test", ".zip");
+    zip.deleteOnExit();
+    ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zip));
+    try {
+      ZipEntry entry = new ZipEntry(entryName);
+      zos.putNextEntry(entry);
+      zos.write(content.getBytes("UTF-8"));
+      zos.closeEntry();
+    }
+    finally {
+      IOUtils.closeQuietly(zos);
+    }
+    return zip;
+  }
