+    /**
+     * Reject paths with more than two consecutive dots in any single path segment.
+     * This blocks traversal sequences beyond the standard ".." (e.g. "...", "....")
+     * which are not handled by {@link ResourceUtil#normalize(String)} but which some
+     * resource providers may still resolve as parent-directory references.
+     * @param path The path
+     * @return {@code true} if the path is invalid
+     */
+    public static boolean isInvalidPath(final String path) {
+        int index = 0;
+        while (index < path.length()) {
+            int charCount = 0;
+            int dotCount = 0;
+            // count dots (".") and total chars in each path segment (between two '/')
+            while (index < path.length() && path.charAt(index) != '/') {
+                if (path.charAt(index) == '.') {
+                    dotCount++;
+                }
+                charCount++;
+                index++;
+            }
+            // if all chars are dots (".") and there are more than two dots, then the path is rejected
+            if (charCount > 2 && dotCount == charCount) {
+                return true;
+            }
+            index++;
+        }
+        return false;
+    }
