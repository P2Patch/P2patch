+     * Guard against path traversal in attacker-controlled log paths.
+     * The request path must be an absolute path that does not contain any
+     * parent-directory ("..") segment, otherwise the request is rejected.
+     *
+     * @param path the requested log file path
+     */
+    private void checkPathSecurity(String path) {
+        if (path == null || path.trim().isEmpty()) {
+            throw new IllegalArgumentException("Illegal file path: path is empty");
+        }
+        String normalized = Paths.get(path).normalize().toString();
+        if (normalized.contains("..") || path.contains("..")) {
