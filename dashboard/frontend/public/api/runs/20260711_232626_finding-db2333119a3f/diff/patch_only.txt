+import java.util.List;
+
+import junit.framework.TestCase;
+
+/**
+ * Tests that the Windows cmd.exe shell neutralizes command-injection
+ * metacharacters (CWE-078) by quoting the executable and arguments that
+ * contain them.
+ */
+public class CmdShellTest
+    extends TestCase
+{
+    private String lastElement( List<String> commandLine )
+    {
+        return commandLine.get( commandLine.size() - 1 );
+    }
+
+    /**
+     * A plain executable without metacharacters must keep the existing behavior:
+     * it is wrapped once by CmdShell but not otherwise quoted.
+     */
+    public void testPlainExecutableIsNotOverQuoted()
+    {
+        Shell sh = new CmdShell();
+        sh.setExecutable( "chmod" );
+
+        List<String> commandLine = sh.getShellCommandLine( new String[]{} );
+
+        assertEquals( "\"chmod\"", lastElement( commandLine ) );
+    }
+
+    /**
+     * An executable containing a cmd.exe command separator must be quoted so it
+     * cannot break out of the command and inject an additional command.
+     */
+    public void testExecutableWithMetacharacterIsQuoted()
+    {
+        Shell sh = new CmdShell();
+        sh.setExecutable( "ping&calc" );
+
+        List<String> commandLine = sh.getShellCommandLine( new String[]{} );
+
+        // The inner token is quoted, so the '&' stays inside quotes instead of
+        // being interpreted by cmd.exe as a command separator.
+        assertEquals( "\"\"ping&calc\"\"", lastElement( commandLine ) );
+    }
+
+    /**
+     * An argument containing cmd.exe metacharacters must likewise be quoted.
+     */
+    public void testArgumentWithMetacharacterIsQuoted()
+    {
+        Shell sh = new CmdShell();
+        sh.setExecutable( "ping" );
+
+        List<String> commandLine = sh.getShellCommandLine( new String[]{ "a|whoami" } );
+
+        assertEquals( "\"ping \"a|whoami\"\"", lastElement( commandLine ) );
+    }
+}
