+    /**
+     * Verifies that a resolved resource path is actually contained within its configured
+     * base resource path, once both are fully normalized the same way the underlying
+     * {@code Resource} implementations normalize them (see {@link StringUtils#cleanPath}).
+     * <p>
+     * This guards against path traversal payloads (e.g. mixed/encoded separators such as
+     * "\.." on some platforms) that {@link UriPath#canonical} may not recognize as ".."
+     * segments, but which {@link StringUtils#cleanPath} does resolve when the resource is
+     * finally opened - potentially escaping the configured root. Any such payload is
+     * rejected here at the trust boundary, rather than relying solely on path parsing.
+     *
+     * @param basePath     the configured base resource path
+     * @param resourcePath the fully resolved resource path (base + requested path)
+     * @return true if resourcePath is base itself or a descendant of base
+     */
+    protected static boolean isPathUnderBase(String basePath, String resourcePath) {
+        if (basePath == null || resourcePath == null) {
+            return false;
+        }
+
+        String normalizedBase = StringUtils.cleanPath(basePath);
+        String normalizedResource = StringUtils.cleanPath(resourcePath);
+
+        if (!normalizedBase.endsWith(SLASH)) {
+            normalizedBase += SLASH;
+        }
+
+        return normalizedResource.equals(normalizedBase.substring(0, normalizedBase.length() - 1))
+                || normalizedResource.startsWith(normalizedBase);
+    }
