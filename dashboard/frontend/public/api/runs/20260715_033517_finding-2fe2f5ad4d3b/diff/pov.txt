+import org.owasp.validator.html.AntiSamy;
+import org.owasp.validator.html.Policy;
+
+import java.net.URL;
+
+/**
+ * Proof-of-Vulnerability for CWE-079 (XSS) in OWASP AntiSamy 1.6.5 —
+ * "STYLE tag smuggling" (CVE-2022-28367 / CVE-2022-29577).
+ *
+ * AntiSamyDOMScanner.processStyleTag() only scans the FIRST child node of a
+ * <style> element with the CSS scanner (CssScanner.scanStyleSheet) and leaves
+ * any additional child nodes untouched. Malformed markup such as
+ *   <style/>b<![cdata[</style><a href=javascript:alert(1)>test
+ * is parsed by the HTML parser into a <style> element that owns MULTIPLE child
+ * nodes. Only the first ("b") gets CSS-sanitized; the smuggled
+ * "</style><a href=javascript:alert(1)>test" survives and is emitted verbatim
+ * inside the raw (CDATA) text content of the serialized <style> element, which
+ * a browser re-parses as live HTML — a cross-site-scripting vector.
+ *
+ * This POV drives the real public API (AntiSamy.scan, DOM mode) and inspects the
+ * cleaned HTML. It exits 0 when the smuggled javascript: payload leaks into the
+ * output (vulnerable / unpatched) and exits 1 when the output no longer contains
+ * it (correctly patched — all child nodes are scanned and the smuggled tag is
+ * neutralized).
+ */
+public class Pov {
+
+    public static void main(String[] args) throws Exception {
+        URL policyUrl = Pov.class.getResource("/antisamy.xml");
+        if (policyUrl == null) {
+            System.err.println("POV ERROR: could not locate antisamy.xml policy on the classpath");
+            System.exit(2);
+        }
+        Policy policy = Policy.getInstance(policyUrl).cloneWithDirective(Policy.USE_XHTML, "true");
+        AntiSamy as = new AntiSamy();
+
+        // Smuggling payload taken from the upstream regression test for the fix.
+        String input = "<style/>b<![cdata[</style><a href=javascript:alert(1)>test";
+
+        String cleanHtml = as.scan(input, policy, AntiSamy.DOM).getCleanHTML();
+
+        System.out.println("INPUT      : " + input);
+        System.out.println("CLEAN HTML : " + cleanHtml);
+
+        // The smuggled javascript: URI must never appear in sanitized output.
+        boolean smuggled = cleanHtml.contains("javascript:alert(1)")
+                || cleanHtml.contains("href=javascript");
+
+        if (smuggled) {
+            System.out.println("VULNERABLE: smuggled 'javascript:' XSS payload survived AntiSamy sanitization "
+                    + "inside the <style> element content.");
+            System.exit(0);
+        } else {
+            System.out.println("SAFE: smuggled payload was neutralized by AntiSamy (patched).");
+            System.exit(1);
+        }
+    }
+}

+#!/usr/bin/env bash
+# Proof-of-Vulnerability runner for finding-2fe2f5ad4d3b (AntiSamy STYLE tag smuggling / CWE-079).
+# Exits 0 when the vulnerable behavior is reproduced on the unpatched code,
+# and non-zero after a correct patch neutralizes the smuggled payload.
+set -euo pipefail
+
+cd "$(dirname "$0")/../.."   # -> project root (/workspace/repo)
+
+# Build the product classes and resolve the runtime classpath.
+mvn -B -q -DskipTests package
+mvn -B -q dependency:build-classpath -Dmdep.outputFile=.security-pipeline/pov/cp.txt
+
+CP="target/classes:$(cat .security-pipeline/pov/cp.txt)"
+
+# Compile and run the POV against the real AntiSamy public API.
+javac -cp "$CP" -d .security-pipeline/pov .security-pipeline/pov/Pov.java
+java -cp ".security-pipeline/pov:$CP" Pov
