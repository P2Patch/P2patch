+import java.io.File;
+import java.io.FileOutputStream;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+import junit.framework.TestCase;
+
+/**
+ * Tests for {@link Expand}, in particular the "Zip Slip" path traversal
+ * protection (CWE-022).
+ */
+public class ExpandTest
+    extends TestCase
+{
+    /**
+     * A zip entry whose name contains a path traversal sequence must not be
+     * written outside of the destination directory.
+     */
+    public void testZipSlipEntryIsRejected()
+        throws Exception
+    {
+        File base = File.createTempFile( "zipslip-test", "" );
+        base.delete();
+        base.mkdirs();
+
+        File dest = new File( base, "dest" );
+        dest.mkdirs();
+
+        File zip = new File( base, "malicious.zip" );
+        ZipOutputStream zos = new ZipOutputStream( new FileOutputStream( zip ) );
+        try
+        {
+            zos.putNextEntry( new ZipEntry( "../escaped.txt" ) );
+            zos.write( "owned".getBytes( "UTF-8" ) );
+            zos.closeEntry();
+        }
+        finally
+        {
+            zos.close();
+        }
+
+        Expand expand = new Expand();
+        expand.setSrc( zip );
+        expand.setDest( dest );
+
+        try
+        {
+            expand.execute();
+        }
+        catch ( Exception expected )
+        {
+            // Extraction is allowed to fail loudly for a malicious entry.
+        }
+
+        File escaped = new File( base, "escaped.txt" );
+        assertFalse( "Zip Slip: entry was written outside the destination directory",
+                     escaped.exists() );
+
+        FileUtils.deleteDirectory( base );
+        zip.delete();
+    }
+
+    /**
+     * A normal entry must still be extracted into the destination directory.
+     */
+    public void testNormalEntryIsExtracted()
+        throws Exception
+    {
+        File base = File.createTempFile( "zipok-test", "" );
+        base.delete();
+        base.mkdirs();
+
+        File dest = new File( base, "dest" );
+        dest.mkdirs();
+
+        File zip = new File( base, "ok.zip" );
+        ZipOutputStream zos = new ZipOutputStream( new FileOutputStream( zip ) );
+        try
+        {
+            zos.putNextEntry( new ZipEntry( "sub/good.txt" ) );
+            zos.write( "hello".getBytes( "UTF-8" ) );
+            zos.closeEntry();
+        }
+        finally
+        {
+            zos.close();
+        }
+
+        Expand expand = new Expand();
+        expand.setSrc( zip );
+        expand.setDest( dest );
+        expand.execute();
+
+        File extracted = new File( dest, "sub/good.txt" );
+        assertTrue( "Normal entry should be extracted into the destination directory",
+                    extracted.exists() );
+
+        FileUtils.deleteDirectory( base );
+        zip.delete();
+    }
+}
