+     * @param expectedType

+     * @return

+     */
+    public static <T> Class<? extends T> loadClass(String className, Class<T> expectedType) {
+        if (className == null) {
+            throw new IllegalArgumentException("Class name cannot be null");
+        }
+        try {
+            Class<?> rawClazz = (classLoader == null) ? Class.forName(className) : classLoader.loadClass(className);
+            if (!expectedType.isAssignableFrom(rawClazz)) {
+                throw new IllegalArgumentException("Class '" + className + "' does not implement/extend expected type '"

+            }
+            return rawClazz.asSubclass(expectedType);
+        } catch (ClassNotFoundException cnfe) {
