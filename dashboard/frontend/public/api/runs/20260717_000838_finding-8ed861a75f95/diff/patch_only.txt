+package io.undertow.server.handlers.resource;
+
+import org.junit.After;
+import org.junit.Assert;
+import org.junit.Before;
+import org.junit.Test;
+
+import java.io.File;
+import java.io.FileOutputStream;
+import java.io.IOException;
+
+/**
+ * Regression test for CWE-022 path traversal: a request path containing a backslash must never
+ * be resolved by {@link FileResourceManager}, since on platforms where {@code java.io.File}
+ * treats '\' as a name separator (e.g. Windows) it could otherwise be used to escape the
+ * configured resource base directory even though upstream callers only canonicalize '/'.
+ *
+ * @author Stuart Douglas
+ */
+public class FileResourceManagerTest {
+
+    private File rootDir;
+    private File servedDir;
+    private File secretFile;
+
+    @Before
+    public void setup() throws IOException {
+        rootDir = new File(System.getProperty("java.io.tmpdir"), "undertow-frm-test-" + System.nanoTime());
+        servedDir = new File(rootDir, "served");
+        servedDir.mkdirs();
+
+        secretFile = new File(rootDir, "secret.txt");
+        writeFile(secretFile, "top secret");
+
+        writeFile(new File(servedDir, "hello.txt"), "hello world");
+    }
+
+    @After
+    public void cleanup() {
+        deleteRecursively(rootDir);
+    }
+
+    @Test
+    public void testNormalResourceIsServed() throws IOException {
+        FileResourceManager manager = new FileResourceManager(servedDir, 1024);
+        Resource resource = manager.getResource("hello.txt");
+        Assert.assertNotNull(resource);
+        Assert.assertEquals(servedDir.getCanonicalFile(), resource.getFile().getParentFile().getCanonicalFile());
+    }
+
+    @Test
+    public void testBackslashTraversalIsBlocked() throws IOException {
+        FileResourceManager manager = new FileResourceManager(servedDir, 1024);
+        Assert.assertNull(manager.getResource("..\\secret.txt"));
+        Assert.assertNull(manager.getResource("\\..\\secret.txt"));
+        Assert.assertNull(manager.getResource("..\\..\\secret.txt"));
+        Assert.assertNull(manager.getResource("subdir\\..\\..\\secret.txt"));
+    }
+
+    /**
+     * On java.io.File implementations where '\' is a name separator (e.g. WinNTFileSystem on
+     * Windows), a path like "..\\secret.txt" is resolved as a traversal sequence that escapes
+     * the resource base directory - the exact CWE-022 behaviour reported for
+     * FileResourceManager#getResource. That OS-specific separator handling can't be exercised
+     * on a Linux test runner, so this test proves the guard is unconditional (not gated on the
+     * current platform's separator) by using a request path that, if the backslash were instead
+     * treated as a literal filename character (as it is on Linux), would successfully resolve to
+     * a real file. A correct fix must reject it regardless, since it can never legitimately be a
+     * two-segment resource path.
+     */
+    @Test
+    public void testPathContainingBackslashIsNeverResolvedEvenIfLiterallyPresentOnDisk() throws IOException {
+        File fileWithBackslashInName = new File(servedDir, "evil\\name.txt");
+        writeFile(fileWithBackslashInName, "should never be served");
+        Assert.assertTrue(fileWithBackslashInName.isFile());
+
+        FileResourceManager manager = new FileResourceManager(servedDir, 1024);
+        Assert.assertNull(manager.getResource("evil\\name.txt"));
+    }
+
+    private static void writeFile(final File f, final String contents) throws IOException {
+        FileOutputStream out = new FileOutputStream(f);
+        try {
+            out.write(contents.getBytes("UTF-8"));
+        } finally {
+            out.close();
+        }
+    }
+
+    private static void deleteRecursively(final File file) {
+        if (file == null || !file.exists()) {
+            return;
+        }
+        File[] children = file.listFiles();
+        if (children != null) {
+            for (File child : children) {
+                deleteRecursively(child);
+            }
+        }
+        file.delete();
+    }
+}
