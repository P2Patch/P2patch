+   /**

+    * extraction directory. This prevents "Zip Slip" style path traversal
+    * attacks where a malicious zip entry name (e.g. containing "../"

+    */

+   {

+      String fileCanonicalPath = file.getCanonicalPath();

+/**
+ * Tests to ensure that InputStreamHelper.writeZipStreamToTempDir does not
+ * allow zip entries to escape the extraction directory (Zip Slip / CWE-22).
+ */
+public class InputStreamHelperTest
+{
+   /**
+    * A zip entry using "../" to escape the extraction directory must be
+    * rejected, and must not result in a file being written outside the
+    * temporary extraction directory.
+    */
+   @Test public void testSimpleTraversalFileEntryIsRejected() throws Exception
+   {
+      assertEscapeIsBlocked("../evil.txt", false);
+   }

+   /**
+    * A zip entry which is not itself prefixed with "../" but whose parent
+    * path (via nested ".." segments) still escapes the extraction directory
+    * must also be rejected. This covers the parent-directory creation code
+    * path (mkdirsQuietly(parent)). The archive first creates a real
+    * sub-directory (so the ".." segments resolve against a directory which
+    * genuinely exists on disk), then attempts to traverse back out of it,
+    * mirroring the ordering-dependent escape the fix must also cover.
+    */
+   @Test public void testNestedTraversalFileEntryIsRejected() throws Exception
+   {
+      byte[] zipBytes = createMultiEntryZip(new String[]{"sub/", "sub/../../evil.txt"}, new String[]{null, "zip-slip-payload"}, new boolean[]{true, false});
+
+      File dir = null;
+      File target = null;
+      try
+      {
+         dir = InputStreamHelper.writeZipStreamToTempDir(new ByteArrayInputStream(zipBytes));
+
+         target = new File(dir, "sub/../../evil.txt");
+         String dirCanonical = dir.getCanonicalPath();
+         String targetCanonical = target.getCanonicalPath();
+         boolean isContained = targetCanonical.equals(dirCanonical) || targetCanonical.startsWith(dirCanonical + File.separator);
+         assertTrue("test entry name should attempt to escape the extraction directory", !isContained);
+
+         assertFalse("zip entry should not have been able to escape the extraction directory: " + targetCanonical, new File(targetCanonical).exists());
+      }
+      catch (IOException ex)
+      {
+         // Throwing is also an acceptable way to reject the malicious entry.
+      }
+      finally
+      {
+         if (dir != null)
+         {
+            FileHelper.deleteQuietly(dir);
+         }
+
+         if (target != null && target.exists())
+         {
+            FileHelper.deleteQuietly(target);
+         }
+      }

+   /**
+    * Build an in-memory zip file with a single entry.
+    *
+    * @param entryName name of the zip entry
+    * @param content content to write for a file entry (ignored for directory entries)
+    * @param isDirectory true to create a directory entry
+    * @return zip file bytes
+    */
+   private byte[] createZip(String entryName, String content, boolean isDirectory) throws IOException
+   {
+      ByteArrayOutputStream baos = new ByteArrayOutputStream();
+      try (ZipOutputStream zos = new ZipOutputStream(baos))
+      {
+         String name = isDirectory && !entryName.endsWith("/") ? entryName + "/" : entryName;
+         ZipEntry entry = new ZipEntry(name);
+         zos.putNextEntry(entry);
+         if (!isDirectory && content != null)
+         {
+            zos.write(content.getBytes("UTF-8"));
+         }
+         zos.closeEntry();
+      }
+      return baos.toByteArray();
+   }
+
+   /**
+    * Build an in-memory zip file containing multiple entries, processed in
+    * the supplied order.
+    *
+    * @param entryNames names of the zip entries
+    * @param contents content to write for each entry (null for directory entries)
+    * @param isDirectory whether each entry is a directory entry
+    * @return zip file bytes
+    */
+   private byte[] createMultiEntryZip(String[] entryNames, String[] contents, boolean[] isDirectory) throws IOException
+   {
+      ByteArrayOutputStream baos = new ByteArrayOutputStream();
+      try (ZipOutputStream zos = new ZipOutputStream(baos))
+      {
+         for (int index = 0; index < entryNames.length; index++)
+         {
+            String entryName = entryNames[index];
+            boolean directory = isDirectory[index];
+            String name = directory && !entryName.endsWith("/") ? entryName + "/" : entryName;
+            ZipEntry entry = new ZipEntry(name);
+            zos.putNextEntry(entry);
+            if (!directory && contents[index] != null)
+            {
+               zos.write(contents[index].getBytes("UTF-8"));
+            }
+            zos.closeEntry();
+         }
+      }
+      return baos.toByteArray();
+   }
