+    /**
+     * Checks that a resolved candidate path is still contained within the configured base
+     * resource path, after normalization.
+     * <p>
+     * This is a defense-in-depth check against path traversal: {@link UriPath#canonical(String)}
+     * only recognizes '/' as a path separator, so request paths that smuggle traversal via
+     * other separators (e.g. backslashes) or via more '..' segments than there are leading
+     * path elements to consume are passed through unmodified. {@link StringUtils#cleanPath}
+     * (used to build the final {@code File}/classpath lookup) does normalize those separators
+     * and collapse '..' segments, but it never rejects a path whose leftover '..' segments
+     * would walk above the root it was concatenated onto - it just keeps them. Comparing the
+     * normalized candidate against the normalized base closes that gap.
+     *
+     * @param base      the configured base resource path
+     * @param candidate the candidate path (base concatenated with the requested path)
+     * @return true if candidate, once normalized, is base or a descendant of base
+     */
+    protected static boolean isPathUnderBase(String base, String candidate) {
+        if (base == null || candidate == null) {
+            return false;
+        }
+
+        String normalizedBase = StringUtils.cleanPath(base);
+        String normalizedCandidate = StringUtils.cleanPath(candidate);
+
+        // Normalize leading slash presence too, since callers may pass either
+        // filesystem-style paths (leading '/') or classpath-relative paths
+        // (leading '/' stripped, e.g. ClassPathResource#getPath()).
+        if (!normalizedBase.startsWith(SLASH)) {
+            normalizedBase = SLASH + normalizedBase;
+        }
+        if (!normalizedCandidate.startsWith(SLASH)) {
+            normalizedCandidate = SLASH + normalizedCandidate;
+        }
+        if (!normalizedBase.endsWith(SLASH)) {
+            normalizedBase += SLASH;
+        }
+        if (!normalizedCandidate.endsWith(SLASH)) {
+            normalizedCandidate += SLASH;
+        }
+
+        return normalizedCandidate.startsWith(normalizedBase);
+    }
