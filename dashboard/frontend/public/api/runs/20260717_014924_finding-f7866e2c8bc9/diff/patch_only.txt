+    /**
+     * Verifies that the resolved extraction target {@code f} is actually located inside {@code dir}, guarding
+     * against path traversal / "zip-slip" style archive entries (e.g. names containing {@code ../} sequences or
+     * absolute paths) that would otherwise resolve outside of the intended destination directory.
+     */
+    private boolean isInsideDestination( final File dir, final File f )
+        throws IOException
+    {
+        final String destDirPath = dir.getCanonicalPath() + File.separator;
+        final String entryPath = f.getCanonicalPath();
+        return entryPath.equals( dir.getCanonicalPath() ) || entryPath.startsWith( destDirPath );
+    }
