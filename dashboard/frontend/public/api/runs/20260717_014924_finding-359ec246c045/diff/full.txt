+	/**
+	 * Resolves fileName against outPath and verifies that the resulting
+	 * canonical path is still located within outPath. This guards against
+	 * "zip slip" / path traversal attacks where a zip entry name contains
+	 * sequences such as "../" (or an absolute path) that would otherwise
+	 * cause the extracted file to be written, its parent directories
+	 * created, or an existing file overwritten/deleted outside of the
+	 * intended extraction root.
+	 *
+	 * @return the verified, canonical output {@link File}
+	 * @throws ZipException if the resolved path escapes outPath
+	 */
+	public static File getSanitizedOutputFile(String outPath, String fileName) throws ZipException {
+		if (!isStringNotNullAndNotEmpty(outPath)) {
+			throw new ZipException("invalid output path");
+		}
+
+		if (!isStringNotNullAndNotEmpty(fileName)) {
+			throw new ZipException("invalid entry name");
+		}
+
+		try {
+			File outDirFile = new File(outPath);
+			File targetFile = new File(outPath, fileName);
+
+			String canonicalOutDir = outDirFile.getCanonicalPath();
+			String canonicalTarget = targetFile.getCanonicalPath();
+
+			if (!canonicalTarget.equals(canonicalOutDir) &&
+					!canonicalTarget.startsWith(canonicalOutDir + File.separator)) {
+				throw new ZipException("Illegal file entry: " + fileName +
+						". Entry is outside of the target directory");
+			}
+
+			return targetFile;
+		} catch (java.io.IOException e) {
+			throw new ZipException("Cannot resolve canonical path for entry: " + fileName, e);
+		}
+	}

+package net.lingala.zip4j.unzip;
+
+import static org.junit.Assert.assertFalse;
+import static org.junit.Assert.assertTrue;
+import static org.junit.Assert.fail;
+
+import java.io.File;
+import java.io.FileOutputStream;
+import java.io.IOException;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+import net.lingala.zip4j.core.ZipFile;
+import net.lingala.zip4j.exception.ZipException;
+
+import org.junit.After;
+import org.junit.Before;
+import org.junit.Test;
+
+/**
+ * Regression tests for the "zip slip" path traversal vulnerability
+ * (CWE-022): a malicious zip entry name containing "../" sequences must
+ * not be able to write, create directories for, or delete/overwrite
+ * files outside of the requested extraction directory.
+ */
+public class ZipSlipTest {
+
+	private File tempRoot;
+	private File extractDir;
+	private File maliciousMarker;
+	private File maliciousDir;
+
+	@Before
+	public void setUp() throws IOException {
+		tempRoot = File.createTempFile("zip4j-ziptest", "");
+		tempRoot.delete();
+		tempRoot.mkdirs();
+
+		extractDir = new File(tempRoot, "extract-here");
+		extractDir.mkdirs();
+
+		// Sibling locations outside of extractDir that a "../" traversal would target.
+		maliciousMarker = new File(tempRoot, "evil.txt");
+		if (maliciousMarker.exists()) {
+			maliciousMarker.delete();
+		}
+		maliciousDir = new File(tempRoot, "evil-dir");
+		if (maliciousDir.exists()) {
+			maliciousDir.delete();
+		}
+	}
+
+	@After
+	public void tearDown() {
+		deleteRecursively(tempRoot);
+	}
+
+	@Test
+	public void extractAllRejectsZipEntryThatTraversesOutsideDestination() throws Exception {
+		File maliciousZip = createZipWithEntry(new File(tempRoot, "malicious.zip"), "../evil.txt", "pwned");
+
+		ZipFile zipFile = new ZipFile(maliciousZip);
+		try {
+			zipFile.extractAll(extractDir.getAbsolutePath());
+			fail("Expected a ZipException to be thrown for a path traversal entry, but extraction succeeded");
+		} catch (ZipException expected) {
+			// expected: traversal must be rejected
+		}
+
+		assertFalse("Entry must not be written outside of the extraction directory", maliciousMarker.exists());
+	}
+
+	@Test
+	public void extractAllRejectsTraversalDirectoryEntry() throws Exception {
+		// A directory entry (name ends with "/") exercises the separate
+		// mkdirs() sink used for directory entries, as opposed to the file
+		// write sink exercised above.
+		File maliciousZip = createZipWithEntry(new File(tempRoot, "malicious-dir.zip"), "../evil-dir/", "");
+
+		ZipFile zipFile = new ZipFile(maliciousZip);
+		try {
+			zipFile.extractAll(extractDir.getAbsolutePath());
+			fail("Expected a ZipException to be thrown for a path traversal directory entry, but extraction succeeded");
+		} catch (ZipException expected) {
+			// expected: traversal must be rejected
+		}
+
+		assertFalse("Directory must not be created outside of the extraction directory", maliciousDir.exists());
+	}
+
+	@Test
+	public void extractAllStillWorksForWellBehavedEntries() throws Exception {
+		File goodZip = createZipWithEntry(new File(tempRoot, "good.zip"), "subdir/hello.txt", "hello");
+
+		ZipFile zipFile = new ZipFile(goodZip);
+		zipFile.extractAll(extractDir.getAbsolutePath());
+
+		File extracted = new File(extractDir, "subdir/hello.txt");
+		assertTrue("Legitimate entries must still extract correctly", extracted.exists());
+	}
+
+	private static File createZipWithEntry(File zipFile, String entryName, String content) throws IOException {
+		ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zipFile));
+		try {
+			zos.putNextEntry(new ZipEntry(entryName));
+			zos.write(content.getBytes("UTF-8"));
+			zos.closeEntry();
+		} finally {
+			zos.close();
+		}
+		return zipFile;
+	}
+
+	private static void deleteRecursively(File file) {
+		if (file == null || !file.exists()) {
+			return;
+		}
+		File[] children = file.listFiles();
+		if (children != null) {
+			for (File child : children) {
+				deleteRecursively(child);
+			}
+		}
+		file.delete();
+	}
+}
