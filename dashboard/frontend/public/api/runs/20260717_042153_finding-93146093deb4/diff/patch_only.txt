+    /**
+     * Verifies that {@code target} is equal to, or a proper descendant of,
+     * {@code root} once both paths are canonicalized. A plain
+     * {@code String#startsWith} comparison of canonical paths is not sufficient
+     * because it has no path-separator boundary, so a sibling directory whose
+     * name merely extends the root's name (e.g. root {@code "extract"} vs
+     * sibling {@code "extractEVIL"}) would incorrectly be treated as contained.
+     */
+    private static void checkContainment(File root, File target)
+        throws IOException
+    {
+        String canonicalRoot = root.getCanonicalPath();
+        String canonicalTarget = target.getCanonicalPath();
+        if (!canonicalTarget.equals(canonicalRoot)
+            && !canonicalTarget.startsWith(canonicalRoot + File.separator))
+        {
+            throw new IOException("JAR resource cannot contain paths with .. characters: " + target);
